/* ═══════════════════════════════════════════════════════ Smart Shopper — Cloudflare Worker Proxy (v5) Multi-stage product search + strong deduplication. ═══════════════════════════════════════════════════════ */ const API_BASE = "https://api-sg.aliexpress.com/sync"; const { keyword, page = 1, pageSize = 50, currency = "USD" } = body; if (!keyword) return json({ ok: false, error: "keyword required" }, env, 400); const params = { method: "aliexpress.affiliate.product.query", app_key: env.ALI_APP_KEY, timestamp: Date.now(), format: "json", v: "2.0", sign_method: "md5", keywords: keyword, page_no: String(page), page_size: String(pageSize), currency: currency, target_currency: currency, target_language: "EN", ship_to_country: "US", tracking_id: env.ALI_TRACKING_ID, }; if (!productId) return json({ ok: false, error: "productId required" }, env, 400); const params = { method: "aliexpress.affiliate.productdetail.get", app_key: env.ALI_APP_KEY, timestamp: Date.now(), format: "json", v: "2.0", sign_method: "md5", product_ids: String(productId), currency: currency, target_currency: currency, target_language: "EN", ship_to_country: "US", tracking_id: env.ALI_TRACKING_ID, }; const result = await callAliExpress(params, env); return json({ ok: true, data: result }, env); } /** * ⭐ البحث الذكي المتعدد المراحل * يقبل: keywords (مصفوفة) — يبحث بها كلها ويجمع النتائج */ async function handleFindSame(request, env) { const body = await request.json(); const { keywords = [], currency = "USD", pageSize = 50 } = body; if (!Array.isArray(keywords) || keywords.length === 0) { return json({ ok: false, error: "keywords array required" }, env, 400); } console.log(`[Worker] find-same with ${keywords.length} queries`); const allProducts = []; const seenIds = new Set(); // نبحث بالتوازي const promises = keywords.slice(0, 5).map(async (kw) => { try { const params = { method: "aliexpress.affiliate.product.query", app_key: env.ALI_APP_KEY, timestamp: Date.now(), format: "json", v: "2.0", sign_method: "md5", keywords: kw, page_no: "1", page_size: String(pageSize), currency: currency, target_currency: currency, target_language: "EN", ship_to_country: "US", tracking_id: env.ALI_TRACKING_ID, }; const result = await callAliExpress(params, env); const products = extractProducts(result); console.log(`[Worker] "${kw}" → ${products.length}`); return { keyword: kw, products }; } catch (e) { console.log(`[Worker] "${kw}" failed: ${e.message}`); return { keyword: kw, products: [], error: e.message }; } }); const results = await Promise.all(promises); // نجمع + نزيل التكرار for (const r of results) { for (const p of r.products) { const id = String(p.product_id || ""); if (!id || seenIds.has(id)) continue; seenIds.add(id); allProducts.push(p); } } console.log(`[Worker] Total unique: ${allProducts.length}`); return json({ ok: true, data: { total: allProducts.length, products: allProducts, queries: keywords, } }, env); } function extractProducts(apiResult) { try { const resp = apiResult?.aliexpress_affiliate_product_query_response; const products = resp?.resp_result?.result?.products?.product || []; return Array.isArray(products) ? products : [products]; } catch (_) { return []; } } async function handleLink(request, env) { const body = await request.json(); const { urls = [] } = body; if (!Array.isArray(urls) || urls.length === 0) { return json({ ok: false, error: "urls required" }, env, 400);
