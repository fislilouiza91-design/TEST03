export default { async fetch(request, env, ctx) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Cache-Control": "no-store" }; if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders }); const url = new URL(request.url); if (url.pathname === '/search-db') { let title = url.searchParams.get('q') || ''; const pid = url.searchParams.get('pid') || ''; const page = parseInt(url.searchParams.get('page')) || 1; const existingIds = url.searchParams.get('existingIds') || ''; try { if (!env.TURSO_URL || !env.TURSO_TOKEN) throw new Error("Missing Turso config"); // ⚡ 1. إنشاء مفتاح فريد للكاش (يعتمد على رقم المنتج لضمان الدقة) const cacheKey = pid ? `search_v2_${pid}_page_${page}` : `search_v2_title_${title}_page_${page}`; // ⚡ 2. التحقق من الكاش أولاً (إذا كان موجوداً، نرجعه فوراً دون استهلاك قاعدة البيانات) if (env.SEARCH_CACHE) { const cachedData = await env.SEARCH_CACHE.get(cacheKey); if (cachedData) { console.log("🟢 Found in KV Cache!"); return new Response(cachedData, { headers: { ...corsHeaders, 'Content-Type': 'application/json', 'X-Cache': 'HIT' } }); } } console.log("🔴 Not in cache, querying DB..."); let englishTitle = title; if (pid) { try { const dbTitle = await getDbTitle(pid, env); if (dbTitle) englishTitle = dbTitle; } catch (e) {} } const result = await searchTursoDB(englishTitle, page, existingIds, env); // ⚡ 3. حفظ النتيجة في الذاكرة (لمدة 3 أيام = 259,200 ثانية) if (env.SEARCH_CACHE && result.items.length > 0) { ctx.waitUntil( env.SEARCH_CACHE.put(cacheKey, JSON.stringify(result), { expirationTtl: 259200 }) ); } return new Response(JSON.stringify(result), { headers: { ...corsHeaders, 'Content-Type': 'application/json', 'X-Cache': 'MISS' } }); } catch (e) { return new Response(JSON.stringify({ ok: false, error: e.message }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } } return new Response('Smart-Turso KV API is running.', { status: 200, headers: corsHeaders }); } } // دالة جلب العنوان async function getDbTitle(pid, env) { const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const res = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: 'SELECT title FROM products WHERE product_id = ? LIMIT 1', args: [{ type: 'text', value: pid }] } }, { type: 'close' }] }) }); const data = await res.json(); if (data.results[0].type === 'error') return null; const rows = data.results[0].response.result.rows; return rows.length > 0 ? rows[0][0].value : null; } // دالة البحث في القاعدة (نفس الكود السابق) async function searchTursoDB(title, page, existingIds, env) { const stopWords = new Set([ 'for', 'the', 'and', 'with', 'from', 'new', 'hot', 'best', 'free', 'high', 'pro', 'sale', 'sales', 'shipping', 'pcs', 'piece', 'set', 'plus', 'original', 'official', 'good', 'quality', 'smart', 'portable', 'wireless', 'digital', 'type', 'model', 'color', 'size', 'men', 'women', 'kids', 'baby', 'home', 'car', 'phone', 'case', 'made', 'material', 'style', 'fashion', 'design', 'item', 'top', 'super', 'ultra', 'mini', 'max', 'premium', 'fast', 'pack', 'box', 'quick', 'release', 'replacement', 'gift', 'drop', 'ship', 'factory', 'direct', 'wholesale', 'dropshipping', 'performance', 'sports', 'racing', 'gas', 'cc', '4x', '10x', '100', 'mah', 'large', 'capacity', 'charging', 'charger', 'external', 'device', 'storage', 'capsule', 'cable', 'adapter', 'cover', 'protector', 'glass', 'iphone', 'samsung', 'xiaomi', 'huawei', 'redmi', 'poco', 'oppo', 'vivo', 'realme' ]); const allWords = title.toLowerCase().replace(/[^\w\s]/g, ' ').split(/\s+/); let meaningfulWords = allWords.filter(w => w.length >= 3 && !stopWords.has(w) && !/^\d+$/.test(w)); const modelNumbers = allWords.filter(w => /\d+/.test(w)); let keywords = meaningfulWords.slice(0, 4); if (modelNumbers.length > 0) keywords.push(modelNumbers[0]); keywords = [...new Set(keywords)]; if (keywords.length === 0) return { ok: true, items: [], hasMore: false }; const ftsQuery = keywords.map(k => `"${k}"*`).join(' OR '); const queryArgs = [{ type: 'text', value: ftsQuery }]; let excludeCondition = ''; if (existingIds && existingIds.trim() !== '') { const ids = existingIds.split(','); excludeCondition = `AND product_id NOT IN (${ids.map(() => '?').join(',')})`; ids.forEach(id => queryArgs.push({ type: 'text', value: id })); } const LIMIT = 60; const OFFSET = (page - 1) * LIMIT; const sql = ` SELECT product_id as id, title, image_url as img, current_price_usd as price, original_price_usd as oldPrice, rating, orders_count as sold FROM products WHERE fts_match(title, ?) ${excludeCondition} ORDER BY rowid DESC LIMIT ${LIMIT} OFFSET ${OFFSET} `; const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const tursoRes = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: sql, args: queryArgs } }, { type: 'close' }] }) }); if (!tursoRes.ok) { const errText = await tursoRes.text(); throw new Error(`Turso HTTP Error: ${tursoRes.status} - ${errText}`); } const data = await tursoRes.json(); const result = data.results[0]; if (result.type === 'error') throw new Error(result.error.message); const cols = result.response.result.cols.map(c => c.name); const rows = result.response.result.rows; const items = rows.map(row => { const obj = {}; cols.forEach((col, i) => { obj[col] = row[i].value; }); const discount = obj.oldPrice && obj.price ? Math.round(((obj.oldPrice - obj.price) / obj.oldPrice) * 100) : 0; return { id: obj.id, title: obj.title, img: obj.img || '', price: obj.price ? String(obj.price) : '', oldPrice: obj.oldPrice ? String(obj.oldPrice) : '', discount: discount > 0 ? String(discount) : '', rating: obj.rating ? String(obj.rating) : '', sold: obj.sold ? String(obj.sold) : '', link: `https://www.aliexpress.com/item/${obj.id}.html` }; }); return { ok: true, items: items, hasMore: items.length === LIMIT }; }
