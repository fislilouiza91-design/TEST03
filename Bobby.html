if (links && links.length > 0) { affiliateLink = links[0].promotion_link; } } if (affiliateLink) { return new Response(JSON.stringify({ success: true, affiliate_url: affiliateLink }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
