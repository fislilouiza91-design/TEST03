import { createClient } from '@libsql/client'; import fs from 'fs'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const FILE = process.env.CATEGORIES_FILE || './config/master_categories.json'; async function main() { console.log('📥 استيراد الكلمات...\n'); if (!fs.existsSync(FILE)) { console.error(`❌ الملف غير موجود: ${FILE}`); process.exit(1); } let content = fs.readFileSync(FILE, 'utf8').replace(/^\uFEFF/, ''); content = content.replace(/"id"\s*\{\s*"id"\s*:\s*(\d+)/g, '"id": $1'); // إصلاح تلقائي let data; try { data = JSON.parse(content); } catch (e) { console.error(`❌ JSON خطأ: ${e.message}`); process.exit(1); } const categories = data.categories || []; let insertedKw = 0; const now = Date.now(); let counter = 0; for (const cat of categories) { const catName = String(cat.name || '').trim(); if (!catName) continue; await db.execute({ sql: `INSERT OR IGNORE INTO categories (name_en, full_path, level, created_at, updated_at) VALUES (?, ?, 1, ?, ?)`, args: [catName, catName, now, now] }); for (const sub of cat.subcategories || []) { for (const kw of (sub.keywords || [])) { const keyword = String(kw).trim().toLowerCase(); if (!keyword || keyword.length < 3) continue; counter++; try { const r = await db.execute({ sql: `INSERT OR IGNORE INTO keywords_pipeline (keyword, category, source, priority, enabled, run_count, success_count, fail_count, last_run_at, created_at) VALUES (?, ?, 'master_category', 4, 1, 0, 0, 0, 0, ?)`, args: [keyword, catName, now + counter], }); if (Number(r.rowsAffected) > 0) insertedKw++; } catch (err) {} } } console.log(` ✅ تم استيراد فئة: [${catName}]`); } console.log(`\n📊 إجمالي الكلمات المضافة: ${insertedKw}`); } main().catch(console.error);
