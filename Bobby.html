let attachedTabId = null; async function attachDebugger(tabId) { return new Promise((resolve, reject) => { chrome.debugger.attach({ tabId }, "1.3", () => { if (chrome.runtime.lastError) { if (chrome.runtime.lastError.message.includes("Another debugger is already attached")) { resolve(); } else { reject(new Error(chrome.runtime.lastError.message)); } } else { attachedTabId = tabId; resolve(); } }); }); } async function detachDebugger() { if (attachedTabId) { return new Promise((resolve) => { chrome.debugger.detach({ tabId: attachedTabId }, () => { attachedTabId = null; resolve(); }); }); } } async function fetchSimilarProducts() { const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true }); if (!tab?.url?.match(/\/item\/(\d+)\.htm/)) return { error: 'NOT_PRODUCT_PAGE' }; try { await attachDebugger(tab.id); await chrome.debugger.sendCommand({ tabId: tab.id }, "Network.enable"); let capturedArrays = []; const onEvent = async (source, method, params) => { if (method === "Network.responseReceived") { const url = params.response.url; const type = params.type; // نركز فقط على طلبات الـ XHR أو Fetch التي تحتوي على كلمات مفتاحية if ((type === 'XHR' || type === 'Fetch') && (url.includes('recommend') || url.includes('itemList') || url.includes('aeglodetailweb') || url.includes('query'))) { // ننتظر حتى يكتمل تحميل الاستجابة بالكامل const checkComplete = setInterval(async () => { try { const body = await chrome.debugger.sendCommand( { tabId: tab.id }, "Network.getResponseBody", { requestId: params.requestId } ); if (body && body.body) { clearInterval(checkComplete); let json = JSON.parse(body.body); const findProductsArray = (obj) => { if (!obj) return null; if (Array.isArray(obj)) { if (obj.length > 0 && (obj[0].productId || obj[0].itemId || obj[0].id)) return obj; for (let item of obj) { const found = findProductsArray(item); if (found) return found; } } else if (typeof obj === 'object') { for (let key in obj) { const found = findProductsArray(obj[key]); if (found) return found; } } return null; }; let productsArray = findProductsArray(json); if (productsArray && productsArray.length > 0) { capturedArrays.push(productsArray); } } } catch (e) { // الاستجابة قد لا تكون جاهزة بعد، سنحاول في الدورة التالية } }, 500); } } }; chrome.debugger.onEvent.addListener(onEvent); // تحفيز الموقع لتحميل البيانات عبر التمرير لأسفل await chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => { window.scrollBy(0, 2000); } }); // الانتظار لمدة 6 ثوانٍ لجمع أكبر قدر ممكن من الطلبات await new Promise(r => setTimeout(r, 6000)); chrome.debugger.onEvent.removeListener(onEvent); await detachDebugger(); if (capturedArrays.length === 0) { return { error: 'NO_DATA_CAPTURED: لم يتم اعتراض أي بيانات. جرب التمرير بيدك وإعادة المحاولة.' }; } // اختيار المصفوفة التي تحتوي على أكبر عدد من المنتجات (القائمة الحقيقية) capturedArrays.sort((a, b) => b.length - a.length); let rawProducts = capturedArrays[0]; // إزالة التكرار (إذا ظهر نفس المنتج بأكثر من طلب) const uniqueProducts = []; const seenIds = new Set(); for (let p of rawProducts) { const id = p.productId || p.itemId || p.id; if (id && !seenIds.has(id)) { seenIds.add(id); uniqueProducts.push(p); } } const products = uniqueProducts.map(p => { const priceNum = p?.prices?.salePrice?.minPrice || p?.prices?.salePrice?.formattedPrice || p?.price?.formattedPrice || p?.price?.minPrice || p?.price || '0'; return { productId: p.productId || p.id || p.itemId, title: { displayTitle: p.title?.displayTitle || p.title || p.titleText || 'Product' }, image: { imgUrl: p.image?.imgUrl || p.imageUrl || p.imgUrl || p.image || '' }, prices: { salePrice: { formattedPrice: typeof priceNum === 'string' && (priceNum.includes('₽') || priceNum.includes('руб')) ? priceNum : `${priceNum} ₽`, minPrice: p?.prices?.salePrice?.minPrice || p?.price?.minPrice || 0 } }, trade: { tradeCount: p.trade?.tradeCount || p.tradeCount || p.sold || 0 }, evaluation: { starRating: p.evaluation?.starRating || p.starRating || p.rating || 0 } }; }); return { products, streamId: null, ret: 'SUCCESS', host: location.host }; } catch (error) { await detachDebugger(); return { error: error.message }; } } chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (msg.action === 'FETCH_SIMILAR') { fetchSimilarProducts() .then(sendResponse) .catch(err => sendResponse({ error: err.message })); return true; } });
