// src/smart-pipeline.js - يعتمد على استخراج البيانات من صفحة البحث مع التحديث الذكي (Upsert) import fetch from 'node-fetch'; import crypto from 'crypto'; import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const WORKER_URL = 'https://smart-shopper.smailsuit11.workers.dev'; // ═══════════════════════════════════════════ // الإعدادات // ═══════════════════════════════════════════ const ENRICHMENT_KEYWORDS = 30; const PAGES_PER_KEYWORD = 3; const COMMON_WORDS = new Set([ 'for','the','and','with','from','new','hot','best','free','high','pro', 'sale','shipping','pcs','piece','set','plus','original','official','good', 'quality','smart','portable','wireless','digital','type','model','color', 'size','men','women','kids','baby','without','use','used','home', 'car','phone','case','made','material','style','fashion','design','item', 'top','super','ultra','mini','max','premium','fast','pack','box', ]); function toNum(v) { if (v == null) return 0; if (typeof v === 'bigint') return Number(v); if (typeof v === 'number') return Number.isFinite(v) ? v : 0; const n = Number(v); return Number.isFinite(n) ? n : 0; } function extractPriceNum(s) { const m = String(s || '').replace(/,/g, '').match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } // ═══════════════════════════════════════════ // 1. اختيار فئة عشوائية // ═══════════════════════════════════════════ async function pickRandomCategory() { try { const res = await db.execute(`SELECT DISTINCT category FROM keywords_pipeline WHERE enabled = 1 AND category IS NOT NULL AND category != ''`); if (res.rows.length === 0) return null; const cats = res.rows.map(r => String(r.category)); const saved = await db.execute(`SELECT value FROM system_state WHERE key = 'category_cycle'`); let used = []; if (saved.rows.length > 0) { try { const state = JSON.parse(saved.rows[0].value); if ((Date.now() - (state.savedAt || 0)) / (1000 * 60 * 60 * 24) < 7) { used = Array.isArray(state.used) ? state.used : []; } } catch (e) {} } let remaining = cats.filter(c => !used.includes(c)); if (remaining.length === 0) { used = []; remaining = cats; } const picked = remaining[Math.floor(Math.random() * remaining.length)]; used.push(picked); await db.execute({ sql: `INSERT OR REPLACE INTO system_state (key, value, updated_at) VALUES ('category_cycle', ?, ?)`, args: [JSON.stringify({ used, lastPicked: picked, totalCats: cats.length, savedAt: Date.now() }), Date.now()], }); console.log(`\n🎲 الفئة المختارة: [${picked}]`); return picked; } catch (e) { return null; } } // ═══════════════════════════════════════════ // 2. اختيار الكلمات المفتاحية // ═══════════════════════════════════════════ async function pickKeywordsFromDB(count, category) { if (!category) return []; const all = await db.execute({ sql: `SELECT keyword, category FROM keywords_pipeline WHERE enabled = 1 AND category = ? AND (last_run_at = 0 OR last_run_at IS NULL) ORDER BY priority ASC, run_count ASC, RANDOM() LIMIT ?`, args: [category, count], }); return all.rows; } async function markProcessed(keyword, success) { await db.execute({ sql: `UPDATE keywords_pipeline SET last_run_at = ?, run_count = run_count + 1, success_count = success_count + ?, fail_count = fail_count + ? WHERE keyword = ?`, args: [Date.now(), success ? 1 : 0, success ? 0 : 1, keyword], }); } // ═══════════════════════════════════════════ // 3. دوال مساعدة // ═══════════════════════════════════════════ function generateImageHash(url) { if (!url) return null; const m = url.match(/\/([^/]+\.(jpg|jpeg|png|webp))/i); if (m) return crypto.createHash('md5').update(m[1]).digest('hex').substring(0, 16); return null; } function extractTokens(title) { if (!title) return ''; const stop = new Set(['for','the','and','with','from','new','hot','best','free','high','pro','sale','shipping','pcs','piece','set']); return title.toLowerCase().replace(/[^\w\s-]/g, ' ').split(/\s+/).filter(w => w.length > 3 && !stop.has(w) && !/^\d+$/.test(w)).slice(0, 12).join(' '); } // ═══════════════════════════════════════════ // 4. حفظ وتحديث البيانات بذكاء (Upsert) // ═══════════════════════════════════════════ async function saveProductsBatch(products, keyword, category) { if (!products || products.length === 0) return 0; const now = Date.now(); const productStmts = []; const imageStmts = []; const priceStmts = []; const detailStmts = []; const derivedKeywords = new Map(); // لحل مشكلة التكرار في نفس الدفعة (مثلاً المنتج ظهر في صفحة 1 و 2) const seenIds = new Set(); for (const p of products) { if (!p.id || !p.title || p.title.length < 5) continue; if (seenIds.has(p.id)) continue; // تخطي المنتجات المكررة في نفس الدفعة seenIds.add(p.id); const tokens = extractTokens(p.title); const words = tokens.split(' ').filter(w => w.length >= 4 && !COMMON_WORDS.has(w)); const priceNum = extractPriceNum(p.price); const oldPriceNum = extractPriceNum(p.oldPrice); // استخراج كلمات مفتاحية جديدة for (let i = 0; i < words.length - 1; i++) { const bigram = `${words[i]} ${words[i + 1]}`; if (bigram.length >= 10 && bigram.length <= 35) { derivedKeywords.set(bigram, (derivedKeywords.get(bigram) || 0) + 1); } } // 1. جدول المنتجات: إدخال جديد أو تحديث السعر والتقييم إذا كان موجوداً (Upsert) productStmts.push({ sql: `INSERT INTO products (product_id, title, title_tokens, category_path, current_price_usd, original_price_usd, rating, orders_count, image_url, image_hash, source_keyword, source_type, first_seen_at, last_updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'enrichment', ?, ?) ON CONFLICT(product_id) DO UPDATE SET current_price_usd = excluded.current_price_usd, original_price_usd = excluded.original_price_usd, rating = excluded.rating, orders_count = excluded.orders_count, last_updated_at = excluded.last_updated_at`, args: [p.id, p.title, tokens, category || '', priceNum, oldPriceNum, parseFloat(p.rating) || 0, p.sold || '', p.img || '', generateImageHash(p.img), keyword, now, now], }); // 2. جدول التفاصيل detailStmts.push({ sql: `INSERT INTO product_details (product_id, currency, in_stock, updated_at) VALUES (?, 'USD', 1, ?) ON CONFLICT(product_id) DO UPDATE SET updated_at = excluded.updated_at`, args: [p.id, now] }); // 3. جدول الصور (يتجاهل إذا كانت الصورة موجودة مسبقاً) if (p.img) { imageStmts.push({ sql: `INSERT OR IGNORE INTO product_images (product_id, image_url, image_hash, position, is_main, created_at) VALUES (?, ?, ?, 0, 1, ?)`, args: [p.id, p.img, generateImageHash(p.img), now] }); } // 4. جدول تاريخ الأسعار (يتجاهل التكرار لنفس اللحظة) if (priceNum > 0) { priceStmts.push({ sql: `INSERT OR IGNORE INTO price_history (product_id, price_usd, original_price_usd, recorded_at, source) VALUES (?, ?, ?, ?, 'search_page')`, args: [p.id, priceNum, oldPriceNum, now] }); } } let inserted = 0; try { // تنفيذ الاستعلامات if (productStmts.length > 0) { const res = await db.batch(productStmts, 'write'); // نحسب المنتجات التي تم إدخالها *جديداً* فقط (وليس التي تم تحديثها) for (const r of res) if (toNum(r.rowsAffected) > 0) inserted++; } if (detailStmts.length > 0) await db.batch(detailStmts, 'write'); if (imageStmts.length > 0) await db.batch(imageStmts, 'write'); if (priceStmts.length > 0) await db.batch(priceStmts, 'write'); } catch (e) { console.error(` ❌ DB Error: ${e.message.substring(0, 60)}`); } // حفظ الكلمات المفتاحية الجديدة const sortedKeywords = [...derivedKeywords.entries()].filter(([_, f]) => f >= 2).sort((a, b) => b[1] - a[1]).slice(0, 5); for (const [kw] of sortedKeywords) { await db.execute({ sql: `INSERT OR IGNORE INTO keywords_pipeline (keyword, category, source, priority, created_at) VALUES (?, ?, 'auto_derived', 7, ?)`, args: [kw, category || 'General', now], }); } return inserted; } // ═══════════════════════════════════════════ // 5. 🔥 لب الإضافة (Cloudflare Worker + Parser) // ═══════════════════════════════════════════ async function fetchPage(baseUrl, page) { try { const aliexpressUrl = baseUrl + '&page=' + page; const workerEndpoint = `${WORKER_URL}/fetch?url=${encodeURIComponent(aliexpressUrl)}`; const res = await fetch(workerEndpoint); if (!res.ok) return []; const html = await res.text(); if (html.length < 5000) return []; let items = parseItems(html); if (!items.length) items = fallbackIds(html); return items; } catch (e) { return []; } } function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'boolean') return v ? 'true' : ''; if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'name', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price', 'desc', 'label']; for (const k of keys) { if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); } for (const k in v) { if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; } return ''; } return String(v); } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { if (!o || typeof o !== 'object') continue; const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = clean(extractText(rawTitle)); let img = ''; if (o.image) { if (typeof o.image === 'string') img = o.image; else img = o.image.imgUrl || o.image.imageUrl || o.image.url || ''; } if (!img && o.picUrl) { if (typeof o.picUrl === 'string') img = o.picUrl; else img = o.picUrl.imgUrl || o.picUrl.url || ''; } if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const orig = p.originalPrice || {}; const price = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const oldPrice = extractText(orig.formatedPrice || orig.minPrice || ''); const discount = extractText(o.discount || sale.discount || ''); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit similaire', img, price, oldPrice, discount, rating, sold, link: 'https://www.aliexpress.com/item/' + id + '.html' }); } catch (e) {} } return out; } function fallbackIds(html) { const re = /"productId"\s*:\s*"(\d{6,})"/g; const seen = new Set(); const out = []; let m; while ((m = re.exec(html))) { if (seen.has(m[1])) continue; seen.add(m[1]); out.push({ id: m[1], title: 'Produit', img: '', price: '', oldPrice: '', discount: '', rating: '', sold: '' }); } return out; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function clean(s) { return String(s).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); } // ═══════════════════════════════════════════ // 6. المحرك الرئيسي (Main) // ═══════════════════════════════════════════ async function main() { console.log('🧠 Smart Pipeline (Worker Edition) - بداية جديدة\n'); const category = await pickRandomCategory(); if (!category) { console.log('⚠️ لا توجد فئات في قاعدة البيانات.'); process.exit(1); } const keywords = await pickKeywordsFromDB(ENRICHMENT_KEYWORDS, category); console.log(`\n📋 تم سحب ${keywords.length} كلمة مفتاحية من فئة [${category}]\n`); let totalSaved = 0; for (const row of keywords) { console.log(`\n🔍 البحث عن: "${row.keyword}"`); let keywordSaved = 0; for (let page = 1; page <= PAGES_PER_KEYWORD; page++) { try { const baseUrl = 'https://www.aliexpress.com/wholesale?SearchText=' + encodeURIComponent(row.keyword.slice(0, 90)) + '&catId=0'; const products = await fetchPage(baseUrl, page); if (!products || products.length === 0) break; const saved = await saveProductsBatch(products, row.keyword, category); console.log(` ✅ صفحة ${page}: تحليل ${products.length}، منتجات جديدة ${saved} في DB.`); keywordSaved += saved; await new Promise(r => setTimeout(r, 2000)); } catch (error) { console.error(` ⚠️ خطأ في الصفحة ${page}:`, error.message); } } totalSaved += keywordSaved; await markProcessed(row.keyword, keywordSaved > 0); } // الإحصائيات النهائية const stats = await db.execute(` SELECT (SELECT COUNT(*) FROM products) as total_products, (SELECT COUNT(*) FROM product_images) as total_images, (SELECT COUNT(*) FROM price_history) as total_prices `); const s = stats.rows[0]; console.log(`\n📊 النتيجة النهائية:`); console.log(` 📦 إجمالي المنتجات: ${toNum(s.total_products)}`); console.log(` 🖼️ إجمالي الصور المسجلة: ${toNum(s.total_images)}`); console.log(` 📉 سجلات الأسعار: ${toNum(s.total_prices)}`); console.log(` ➕ منتجات جديدة أضيفت الآن: ${totalSaved}`); } main().catch(err => { console.error('💥 خطأ قاتل:', err); process.exit(1); });
