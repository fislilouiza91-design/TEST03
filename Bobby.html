function scanPageForProducts(sortByPopular = false) { if (document.title.toLowerCase().includes('security') || document.title.toLowerCase().includes('captcha')) return 'CAPTCHA'; let products = []; const seen = new Set(); const links = document.querySelectorAll('a[href*="/item/"]'); for (const link of links) { const match = link.href.match(/\/item\/(\d+)\.html/); if (!match) continue; const pid = match[1]; if (seen.has(pid)) continue; // 1. جلب الصورة let img = ''; const imgEls = link.querySelectorAll('img'); for (let el of imgEls) { let src = el.getAttribute('src') || el.getAttribute('data-src') || el.getAttribute('srcset') || ''; if (src.startsWith('//')) src = 'https:' + src; if (src && (src.includes('ae01.alicdn.com') || src.includes('ae04.alicdn.com') || src.includes('.jpg') || src.includes('.webp'))) { if (!src.includes('tps') && !src.includes('banner') && !src.includes('logo')) { img = src.split(' ')[0].split(',')[0]; break; } } } if (!img) { const firstImg = link.querySelector('img'); if (firstImg) { img = firstImg.src || firstImg.getAttribute('data-src') || ''; if (img.startsWith('//')) img = 'https:' + img; } } if (!img) continue; // 2. استخراج النصوص let fullText = link.innerText || ''; let textLines = fullText.split('\n').map(l => l.trim()).filter(l => l.length > 0); // 3. العنوان (نجلبه من خصائص العناصر أو من أطول سطر) let title = link.getAttribute('title') || link.querySelector('[class*="title"]')?.innerText || ''; if (!title || title.length < 5) { title = textLines.reduce((a, b) => a.length > b.length ? a : b, ''); } if (!title) continue; title = title.replace(/\n/g, ' ').trim().substring(0, 80); // 4. السعر (يدعم NT$ وجميع العملات) let price = '—'; const priceEl = link.querySelector('[class*="price"], [class*="Price"]'); if (priceEl && priceEl.innerText.match(/\d/)) { price = priceEl.innerText.trim().split('\n')[0]; } else { const priceMatch = fullText.match(/(NT\$|US \$|\$|€|£|د\.ج|DA|SAR|AED|руб|zł|kr|CHF|¥)\s*(\d+[.,]?\d*)/i) || fullText.match(/(\d+[.,]?\d*)\s*(NT\$|US \$|\$|€|£|د\.ج|DA|SAR|AED|руб|zł|kr|CHF|¥)/i); if (priceMatch) price = priceMatch[0]; } // 5. عدد المبيعات (يدعم صيغة الأقواس مثل (1000-) أو (500+) أو كلمة sold) let soldCount = 0; const soldRegex = /\((\d+(?:[.,]\d+)?[kK]?)[^\)]*\)/; // يلتقط (1000-) أو (500+) let soldMatch = fullText.match(soldRegex); if (soldMatch) { let numStr = soldMatch[1].toLowerCase().replace(',', '.'); soldCount = numStr.includes('k') ? parseFloat(numStr) * 1000 : parseFloat(numStr); } else { const soldTextRegex = /(\d+(?:[.,]\d+)?[kK]?)\s*(?:sold|vendus|مباع|продано|verkauft|venduti|vendidos)/i; let soldTextMatch = fullText.match(soldTextRegex); if (soldTextMatch) { let numStr = soldTextMatch[1].toLowerCase().replace(',', '.'); soldCount = numStr.includes('k') ? parseFloat(numStr) * 1000 : parseFloat(numStr); } } // 6. التقييم (يبحث عن أرقام مثل 4.5، أو يقرأ أسماء صور النجوم) let rating = 0; const ratingTextRegex = /([0-5][.,]\d)/; let ratingTextMatch = fullText.match(ratingTextRegex); if (ratingTextMatch) { // التأكد من أن الرقم ليس جزءاً من السعر if (!new RegExp(`([a-zA-Z$€£])\\s*${ratingTextMatch[1]}`).test(fullText)) { rating = parseFloat(ratingTextMatch[1].replace(',', '.')); } } // إذا لم يجد رقماً نصياً، يقرأ صورة النجمة (مثال: star_4_5.png) if (rating === 0) { const starImg = link.querySelector('img[src*="star"], [class*="star"] img'); if (starImg) { const src = starImg.src || ''; const starMatch = src.match(/star[_-]?(\d)[_-]?(\d)?/i); if (starMatch) { rating = parseFloat(starMatch[1] + (starMatch[2] ? '.' + starMatch[2] : '')); } } } // 7. الشحن المجاني let isFreeShipping = /free shipping|livraison gratuite|envío gratis|شحن مجاني|spedizione gratuita|kostenloser versand|免运费|免運費/i.test(fullText); seen.add(pid); products.push({ productId: pid, title: { displayTitle: title }, image: { imgUrl: img }, prices: { salePrice: { formattedPrice: price } }, evaluation: { starRating: rating }, trade: { realTradeCount: soldCount }, _isLocal: true, _isFreeShipping: isFreeShipping, url: link.href }); } if (sortByPopular) products.sort((a, b) => b.trade.realTradeCount - a.trade.realTradeCount); return products.slice(0, 20); }
