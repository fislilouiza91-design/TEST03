// src/apify-fetcher.js - المرحلة 1: Apify + استخراج الكلمات البديلة import fetch from 'node-fetch'; import crypto from 'crypto'; import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const APIFY_TOKEN = process.env.APIFY_TOKEN; const KEYWORDS_PER_RUN = 4; const MAX_ITEMS_PER_KEYWORD = 20; // ═══════════════════════════════════════════ // دوال مساعدة // ═══════════════════════════════════════════ function generateImageHash(url) { if (!url) return null; const m = url.match(/\/([^/]+\.(jpg|jpeg|png|webp))/i); if (m) return crypto.createHash('md5').update(m[1]).digest('hex').substring(0, 16); return null; } function extractTokens(title) { if (!title) return ''; const stop = new Set(['for','the','and','with','from','new','hot','best','free','high','pro','sale','shipping','pcs','piece','set']); return title.toLowerCase() .replace(/[^\w\s-]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !stop.has(w) && !/^\d+$/.test(w)) .slice(0, 10) .join(' '); } function detectCategory(categoryPath) { if (!categoryPath || !Array.isArray(categoryPath) || categoryPath.length === 0) { return 'general'; } const rootId = String(categoryPath[0]); const rootMap = { '44': 'electronics', '3': 'home-kitchen', '100001540': 'fashion-women', '100001541': 'fashion-men', '66': 'beauty', '18': 'sports', '26': 'toys-kids', '34': 'automotive', '36': 'pets', '1420': 'tools', '509': 'watches', '1511': 'jewelry', '200000348': 'phones', '7': 'computer', }; return rootMap[rootId] || 'general'; } // ═══════════════════════════════════════════ // استدعاء Apify Actor // ═══════════════════════════════════════════ async function runApifyActor(keyword) { console.log(` 🎭 Apify Actor...`); const actorId = 'apt_marble~aliexpress-search-scraper'; const runUrl = `https://api.apify.com/v2/acts/${actorId}/run-sync-get-dataset-items?token=${APIFY_TOKEN}`; const inputBody = { keyword: keyword, country: 'us', currency: 'USD', sortOrder: 'best_match', maxItems: MAX_ITEMS_PER_KEYWORD, }; const response = await fetch(runUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(inputBody), timeout: 300000, }); if (!response.ok) { const err = await response.text(); throw new Error(`Apify ${response.status}: ${err.substring(0, 200)}`); } const items = await response.json(); console.log(` 📥 Apify: ${Array.isArray(items) ? items.length : 0} عنصر`); if (Array.isArray(items) && items.length > 0) { console.log(` 🔍 نموذج: ${items[0].title?.substring(0, 50)}...`); } return items; } // ═══════════════════════════════════════════ // حفظ المنتجات في Turso // ═══════════════════════════════════════════ async function saveToTurso(keyword, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { if (!p.product_id || !p.title) continue; if (p.title.length < 15) continue; try { const imageHash = generateImageHash(p.image_url); const tokens = extractTokens(p.title); const category = detectCategory(p.category_path); const trustScore = (p.rating >= 4.8 ? 40 : p.rating >= 4.5 ? 30 : p.rating >= 4.0 ? 20 : 10) + (p.orders_count >= 10000 ? 30 : p.orders_count >= 1000 ? 20 : p.orders_count >= 100 ? 10 : 0) + (p.is_free_shipping ? 10 : 0); const valueScore = (p.rating / 5.0) * 35 + Math.min(p.orders_count / 5000, 1.0) * 30 + Math.max(0, 1 - (p.current_price / 100.0)) * 20; await db.execute({ sql: `INSERT INTO products_current (product_id, keyword, language, title, current_price, original_price, orders_count, rating, image_url, seller_name, image_hash, title_tokens, category, is_free_shipping, trust_score, value_score, total_score, first_seen_at, last_updated_at, update_count) VALUES (?, ?, 'en', ?, ?, ?, ?, ?, ?, '', ?, ?, ?, ?, ?, ?, ?, ?, ?, 1) ON CONFLICT (product_id, keyword, language) DO UPDATE SET current_price = excluded.current_price, orders_count = excluded.orders_count, rating = excluded.rating, trust_score = excluded.trust_score, value_score = excluded.value_score, total_score = excluded.total_score, last_updated_at = excluded.last_updated_at, update_count = update_count + 1`, args: [ p.product_id, keyword, p.title, p.current_price, p.original_price, p.orders_count, p.rating, p.image_url, imageHash, tokens, category, p.is_free_shipping ? 1 : 0, trustScore, valueScore, Math.round(trustScore * 0.5 + valueScore * 0.5), now, now ], }); saved++; } catch (e) { console.error(` ❌ ${e.message.substring(0, 60)}`); } } return saved; } // ═══════════════════════════════════════════ // ⭐ استخراج الكلمات المفتاحية البديلة // ═══════════════════════════════════════════ async function extractDerivedKeywords(products) { const derived = new Map(); for (const p of products) { if (!p.title_tokens) continue; const tokens = p.title_tokens.split(' ').filter(t => t.length > 3); if (tokens.length < 2) continue; // توليد ثنائيات (Bigrams) من الكلمات for (let i = 0; i < tokens.length - 1; i++) { const bigram = `${tokens[i]} ${tokens[i + 1]}`; if (derived.has(bigram)) { derived.get(bigram).frequency++; } else { derived.set(bigram, { keyword: bigram, frequency: 1, category: p.category || 'general', }); } } } // احتفظ بالكلمات التي ظهرت مرتين أو أكثر const validated = [...derived.values()].filter(d => d.frequency >= 2); return validated; } async function saveDerivedKeywords(keywords) { let inserted = 0; for (const kw of keywords) { try { const result = await db.execute({ sql: `INSERT INTO keyword_rotation (keyword, category, subcategory, priority, enabled) VALUES (?, ?, 'derived', 5, 1) ON CONFLICT (keyword) DO NOTHING`, args: [kw.keyword, kw.category], }); inserted += result.rowsAffected || 0; } catch (e) {} } return inserted; } // ═══════════════════════════════════════════ // اختيار الكلمات الأصلية // ═══════════════════════════════════════════ async function pickNextKeywords(count) { const res = await db.execute({ sql: `SELECT keyword, category FROM keyword_rotation WHERE enabled = 1 AND (subcategory IS NULL OR subcategory != 'derived') ORDER BY CASE WHEN last_run_at = 0 THEN 0 ELSE 1 END, priority ASC, run_count ASC, last_run_at ASC LIMIT ?`, args: [count], }); return res.rows; } async function markProcessed(keyword, success) { await db.execute({ sql: `UPDATE keyword_rotation SET last_run_at = ?, run_count = run_count + 1, success_count = success_count + ?, fail_count = fail_count + ? WHERE keyword = ?`, args: [Date.now(), success ? 1 : 0, success ? 0 : 1, keyword], }); } // ═══════════════════════════════════════════ // الدالة الرئيسية // ═══════════════════════════════════════════ async function main() { const start = Date.now(); console.log('🎭 المرحلة 1: Apify (كلمات أصلية + استخراج كلمات بديلة)...\n'); if (!APIFY_TOKEN) { console.error('❌ APIFY_TOKEN غير موجود!'); process.exit(1); } const keywords = await pickNextKeywords(KEYWORDS_PER_RUN); console.log(`📋 سيتم معالجة ${keywords.length} كلمة أصلية\n`); let totalSaved = 0; let successCount = 0; let failCount = 0; for (const row of keywords) { console.log(`\n🔍 "${row.keyword}" [${row.category}]`); try { const items = await runApifyActor(row.keyword); if (!Array.isArray(items) || items.length === 0) { console.log(` ⚠️ 0 نتيجة`); failCount++; await markProcessed(row.keyword, false); continue; } const products = items.map(item => ({ product_id: item.productId ? String(item.productId) : null, title: item.title || '', current_price: parseFloat(item.price || 0), original_price: parseFloat(item.originalPrice || 0), image_url: item.imageUrl || '', orders_count: parseInt(item.soldCount || 0), rating: parseFloat(item.rating || 0), is_free_shipping: item.hasFreeShippingOffer === true, category_path: item.categoryPath || [], })).filter(p => p.product_id && p.title && p.title.length > 15); const saved = await saveToTurso(row.keyword, products); console.log(` ✅ ${saved} منتج مخزّن`); totalSaved += saved; if (saved > 0) successCount++; else failCount++; await markProcessed(row.keyword, saved > 0); await new Promise(r => setTimeout(r, 2000)); } catch (error) { console.log(` ❌ فشل: ${error.message}`); failCount++; await markProcessed(row.keyword, false); } } // ⭐ استخراج الكلمات البديلة من جميع المنتجات console.log('\n🔑 استخراج كلمات مفتاحية بديلة من المنتجات...'); const allProducts = await db.execute(` SELECT title_tokens, category FROM products_current WHERE title_tokens IS NOT NULL AND title_tokens != '' ORDER BY last_updated_at DESC LIMIT 300 `); const derived = await extractDerivedKeywords(allProducts.rows); const insertedDerived = await saveDerivedKeywords(derived); console.log(` ✅ ${insertedDerived} كلمة مفتاحية بديلة أُضيفت للدوران`); const mins = ((Date.now() - start) / 60000).toFixed(1); console.log(`\n📊 تقرير المرحلة 1 (${mins} دقيقة):`); console.log(` ✅ نجح: ${successCount}/${keywords.length}`); console.log(` ❌ فشل: ${failCount}`); console.log(` 📦 مخزّن: ${totalSaved} منتج`); console.log(` 🔑 كلمات بديلة جديدة: ${insertedDerived}`); console.log(` 💰 تكلفة Apify: $${(totalSaved * 0.0007).toFixed(4)}`); if (failCount > successCount) process.exit(1); } main().catch(err => { console.error('💥 خطأ قاتل:', err); process.exit(1); });
