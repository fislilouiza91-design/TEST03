export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } const url = new URL(request.url); let keyword = url.searchParams.get("q") || ""; const originalTitle = url.searchParams.get("originalTitle") || ""; const page = parseInt(url.searchParams.get("page") || "1"); if (page > 2) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { // 1. محاولة استخدام الذكاء الاصطناعي لاستخراج الكلمات if (originalTitle && env.AI) { try { const aiQuery = await env.AI.run('@cf/meta/llama-3-8b-instruct', { messages: [ { role: 'system', content: 'You are an AliExpress search expert. Extract ONLY the main product name, brand, and model as 2-4 English keywords. Do NOT include colors, sizes, quantities, or promotional words. Output ONLY the keywords separated by spaces.' }, { role: 'user', content: `Title: ${originalTitle}` } ], max_tokens: 20 }); if (aiQuery && aiQuery.response) { let cleanAI = aiQuery.response.trim().replace(/[^a-zA-Z0-9\s]/g, ''); if (cleanAI.length > 2) { keyword = cleanAI; console.log(`🤖 AI Extracted: [${keyword}]`); } } } catch (e) { console.log(`⚠️ AI failed, falling back to dynamic extraction. Error: ${e.message}`); } } // 2. إذا فشل الذكاء الاصطناعي، نستخدم الاحتياط الذكي (أول 4 كلمات) if (!keyword && originalTitle) { const cleanTitle = originalTitle.replace(/[^\w\s]/g, ' ').replace(/\s+/g, ' ').trim(); const rawWords = cleanTitle.split(' '); const stopWords = new Set(['the','and','for','with','new','hot','sale','free','shipping','original','high','quality','best','pcs','set','pack','piece','items','item','wholesale','dropshipping','compatible','fits','fit','suitable','accessories','accessory','universal','dual','purpose','large','screen','plug','creative','dedicated','student']); const usefulWords = rawWords.filter(w => { const lower = w.toLowerCase(); return w.length > 1 && !stopWords.has(lower) && !/^\d+$/.test(lower) && !/^\d+(mm|cm|inch)$/i.test(lower); }); keyword = usefulWords.slice(0, 4).join(' ') || rawWords.slice(0, 3).join(' '); console.log(`🛡️ Fallback Extracted: [${keyword}]`); } if (!keyword) { return new Response(JSON.stringify({ success: false, message: "لم يتم العثور على كلمات بحث." }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // 3. جلب المنتجات من AliExpress بالكلمة المحددة const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${env.SCRAPINGANT_API_KEY}&return_page_source=true&proxy_type=datacenter&browser=false&proxy_country=us`; const antResponse = await fetch(antUrl); if (!antResponse.ok) throw new Error(`ScrapingAnt Error: ${antResponse.status}`); const html = await antResponse.text(); if (html.includes('punish') || html.includes('captcha') || html.length < 5000) { return new Response(JSON.stringify({ success: false, message: "كابتشا أو حظر مؤقت." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // 4. استخراج المنتجات (بدون فلترة إضافية) let products = parseItems(html); if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لا توجد منتجات." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // إرجاع النتائج return new Response(JSON.stringify({ success: true, data: products, aiQuery: keyword }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // دوال الاستخراج // ========================================== function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const priceNum = extractPriceNumber(priceStr); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img: img, price: priceNum, rating: parseFloat(rating) || 0, soldStr: sold, url: `https://www.aliexpress.com/item/${id}.html` }); } catch (e) {} } return out; }
