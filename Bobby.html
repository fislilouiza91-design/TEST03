import { createClient } from '@libsql/client'; import fetch from 'node-fetch'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const MARKETS = [ { lang: 'en', code: 'US' }, { lang: 'ar', code: 'SA' }, { lang: 'zh', code: 'CN' }, ]; const KEYWORDS = [ 'bluetooth earphones', 'smart watch', 'wireless charger', 'phone case', 'laptop stand', 'usb c cable', 'power bank', 'gaming mouse', 'mechanical keyboard', 'webcam', ]; const API_KEY = process.env.OMKAR_API_KEY; const DELAY_BETWEEN_KEYWORDS = 4000; // زيادة التأخير لتقليل الضغط async function fetchFromOmkar(keyword, countryCode) { const url = `https://aliexpress-scraper-api.omkar.cloud/aliexpress/v2/search?` + `query=${encodeURIComponent(keyword)}&` + `sort_by=most_orders&` + `country_code=${countryCode}&` + `page=1`; const response = await fetch(url, { method: 'GET', headers: { 'API-Key': API_KEY, 'Accept': 'application/json', }, }); if (!response.ok) { const errText = await response.text(); throw new Error(`Omkar API error ${response.status}: ${errText}`); } const data = await response.json(); return data.results || []; } // دالة الجلب مع إعادة المحاولة التلقائية async function fetchWithRetry(keyword, countryCode, maxRetries = 3) { for (let attempt = 1; attempt <= maxRetries; attempt++) { try { return await fetchFromOmkar(keyword, countryCode); } catch (error) { const isRetryable = error.message.includes('502') || error.message.includes('503') || error.message.includes('504') || error.message.includes('upstream_unavailable') || error.message.includes('timeout'); if (!isRetryable || attempt === maxRetries) { throw error; } const waitSeconds = attempt * 15; // 15ث، 30ث، 45ث console.log(` ⏳ خطأ مؤقت (${attempt}/${maxRetries})، انتظار ${waitSeconds}ث...`); await new Promise(r => setTimeout(r, waitSeconds * 1000)); } } } async function saveToTurso(keyword, language, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { try { const productId = p.id || p.product_id; const title = p.title || ''; const price = parseFloat(p.pricing?.sale_price || p.sale_price || 0); const originalPrice = parseFloat(p.pricing?.original_price || p.original_price || 0); const orders = parseInt(p.orders_count || p.orders || 0); const rating = parseFloat(p.rating || 0); const image = p.image || ''; const sellerName = p.store?.name || p.seller?.name || ''; const sellerRating = parseFloat(p.store?.rating || p.seller?.rating || 0); const isFreeShipping = (p.shipping?.free === true || p.shipping?.free_shipping === true) ? 1 : 0; await db.execute({ sql: `INSERT INTO products_current (product_id, keyword, language, title, current_price, original_price, orders_count, rating, image_url, seller_name, seller_rating, is_free_shipping, first_seen_at, last_updated_at, update_count) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1) ON CONFLICT (product_id, keyword, language) DO UPDATE SET title = excluded.title, current_price = excluded.current_price, original_price = excluded.original_price, orders_count = excluded.orders_count, rating = excluded.rating, image_url = excluded.image_url, seller_name = excluded.seller_name, seller_rating = excluded.seller_rating, is_free_shipping = excluded.is_free_shipping, last_updated_at = excluded.last_updated_at, update_count = update_count + 1`, args: [productId, keyword, language, title, price, originalPrice, orders, rating, image, sellerName, sellerRating, isFreeShipping, now, now] }); await db.execute({ sql: `INSERT INTO price_history (product_id, keyword, price, orders_count, recorded_at, source) VALUES (?, ?, ?, ?, ?, ?)`, args: [productId, `${keyword} (${language})`, price, orders, now, 'omkar'] }); saved++; } catch (error) { console.error(` ❌ فشل حفظ ${p.id}:`, error.message); } } return saved; } async function main() { console.log('🚀 بدء الجلب متعدد اللغات...'); console.log(`📋 ${KEYWORDS.length} كلمة × ${MARKETS.length} سوق = ${KEYWORDS.length * MARKETS.length} طلب\n`); let totalSaved = 0; let successCount = 0; let failCount = 0; const totalJobs = KEYWORDS.length * MARKETS.length; for (const keyword of KEYWORDS) { for (const market of MARKETS) { try { console.log(`🔍 جلب: "${keyword}" بلغة [${market.lang}] (${market.code})`); const products = await fetchWithRetry(keyword, market.code); console.log(` 📦 استُلم ${products.length} منتج`); const saved = await saveToTurso(keyword, market.lang, products); console.log(` ✅ خُزّن ${saved} منتج`); totalSaved += saved; successCount++; await new Promise(r => setTimeout(r, DELAY_BETWEEN_KEYWORDS)); } catch (error) { console.error(` ❌ فشل "${keyword}" [${market.lang}]:`, error.message); failCount++; } } } console.log(`\n📊 ملخص:`); console.log(` ✅ نجح: ${successCount}/${totalJobs}`); console.log(` ❌ فشل: ${failCount}`); console.log(` 📦 إجمالي المنتجات المخزنة: ${totalSaved}`); // لا تفشل إذا فشل أقل من 50% if (failCount > successCount) { process.exit(1); } } main().catch(error => { console.error('💥 خطأ قاتل:', error); process.exit(1); });
