// popup/popup.js const state = { products: [], streamId: null, host: null, page: 1, sortMode: 'popular', // cheapest | popular | rating loading: false, hasMore: true, }; const $ = (s) => document.querySelector(s); const resultsEl = $('#results'); const loadMoreBtn = $('#loadMore'); const statusEl = $('#status'); async function fetchFromBackground({ page = 1, streamId = null } = {}) { return new Promise((resolve) => { chrome.runtime.sendMessage({ action: 'FETCH_SIMILAR', page, streamId }, resolve); }); } // دالة الترتيب (Client-side) - مطابقة لـ widget.js الأصلي function sortProductsClientSide(products, mode) { const arr = [...products]; if (mode === 'cheapest') { arr.sort((a, b) => { const pA = Number(a?.prices?.salePrice?.minPrice) || 999999; const pB = Number(b?.prices?.salePrice?.minPrice) || 999999; return pA - pB; }); } else if (mode === 'rating') { arr.sort((a, b) => { const rA = Number(a?.evaluation?.starRating) || 0; const rB = Number(b?.evaluation?.starRating) || 0; return rB - rA; }); } else { arr.sort((a, b) => { const tA = Number(a?.trade?.realTradeCount) || 0; const tB = Number(b?.trade?.realTradeCount) || 0; return tB - tA; }); } return arr; } function getRating(p) { return Number(p?.evaluation?.starRating) || Number(p?.evaluation?.ratingAvg) || 0; } function getTradeCount(p) { return Number(p?.trade?.realTradeCount) || Number(p?.trade?.tradeCount) || 0; } function render() { if (!state.products.length) { resultsEl.innerHTML = `

لا توجد منتجات مشابهة حالياً

`; return; } const sorted = sortProductsClientSide(state.products, state.sortMode); resultsEl.innerHTML = sorted.map((p) => { const rawImg = p?.image?.imgUrl || ''; const img = rawImg.startsWith('//') ? 'https:' + rawImg : rawImg; const price = p?.prices?.salePrice?.formattedPrice || '—'; const title = p?.title?.displayTitle || p?.title?.seoTitle || 'منتج'; const rating = getRating(p); const trades = getTradeCount(p); const url = `https://${state.host}/item/${p.productId}.html`; // ⚠️ ضع رابط العمولة الخاص بك هنا const aff = `https://YOUR-AFFILIATE-LINK.com/redirect?to=${encodeURIComponent(url)}`; return `
${escapeHtml(title)}
${price}
⭐ ${rating ? rating.toFixed(1) : '—'} 🔥 ${formatNumber(trades)}
`; }).join(''); } function escapeHtml(s) { const d = document.createElement('div'); d.textContent = s; return d.innerHTML; } function formatNumber(n) { if (n >= 1000000) return (n / 1000000).toFixed(1) + 'M'; if (n >= 1000) return (n / 1000).toFixed(1) + 'K'; return String(n); } async function loadInitial() { state.loading = true; state.page = 1; state.products = []; state.streamId = null; state.hasMore = true; resultsEl.innerHTML = `

جاري البحث...

`; loadMoreBtn.disabled = true; statusEl.textContent = ''; const res = await fetchFromBackground({ page: 1, streamId: null }); state.loading = false; if (res?.error) return showError(res.error); state.products = res.products || []; state.streamId = res.streamId || null; state.host = res.host || null; state.hasMore = state.products.length >= 20; if (!state.products.length) { resultsEl.innerHTML = `

لم نجد منتجات مشابهة لهذا المنتج

`; return; } render(); loadMoreBtn.disabled = !state.hasMore; statusEl.textContent = `${state.products.length} منتج`; } async function loadMore() { if (state.loading || !state.hasMore) return; state.loading = true; loadMoreBtn.disabled = true; loadMoreBtn.textContent = 'جاري التحميل...'; const nextPage = state.page + 1; const res = await fetchFromBackground({ page: nextPage, streamId: state.streamId }); state.loading = false; loadMoreBtn.textContent = 'تحميل المزيد'; if (res?.error) return showError(res.error); const more = res.products || []; if (more.length) { state.products = [...state.products, ...more]; state.streamId = res.streamId || state.streamId; state.page = nextPage; state.hasMore = more.length >= 20; render(); statusEl.textContent = `${state.products.length} منتج`; } else { state.hasMore = false; } loadMoreBtn.disabled = !state.hasMore; } function showError(msg) { const map = { 'NOT_PRODUCT_PAGE': 'افتح صفحة منتج في AliExpress أولاً', 'NO_TOKEN': 'لم يتم العثور على توكن. أعد تحميل صفحة AliExpress.', }; const friendly = map[msg] || (msg?.startsWith('API_') ? 'تم رفض الطلب من AliExpress. حاول لاحقاً.' : msg); resultsEl.innerHTML = `

⚠️ ${escapeHtml(friendly || 'حدث خطأ')}

`; loadMoreBtn.disabled = true; } document.querySelectorAll('.sort-btn').forEach(btn => { btn.addEventListener('click', () => { document.querySelectorAll('.sort-btn').forEach(b => b.classList.remove('active')); btn.classList.add('active'); state.sortMode = btn.dataset.sort; render(); // الترتيب يتم فوراً في الواجهة بدون إعادة طلب }); }); loadMoreBtn.addEventListener('click', loadMore); // الوضع الليلي const themeToggle = $('#themeToggle'); chrome.storage.local.get(['darkMode'], (r) => { if (r.darkMode) { document.body.classList.add('dark'); themeToggle.textContent = '☀️'; } }); themeToggle.addEventListener('click', () => { document.body.classList.toggle('dark'); const isDark = document.body.classList.contains('dark'); themeToggle.textContent = isDark ? '☀️' : '🌙'; chrome.storage.local.set({ darkMode: isDark }); }); loadInitial();
