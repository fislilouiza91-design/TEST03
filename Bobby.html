const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'high', 'good', '2023', '2024', '2025', '2026', 'best', 'top', 'pcs', 'set', 'pack', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'double', 'door', 'enclosed', 'large', 'angle', 'flip', 'cover', 'removable', 'easy', 'clean', 'waterproof', 'portable', 'multifunctional', 'design', 'filter' ]); chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; // في وضع الاختبار، نتخطى الكاش ونطلب دائماً من الـ API مباشرة if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); const q = encodeURIComponent(searchQuery); // الاتصال بالـ Cloudflare Worker fetch(`${WORKER_URL}?q=${q}&page=${page}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { // تنسيق البيانات لتناسب content.js const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.priceStr, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, // نجعلها true لكي تعرض في كل التبويبات finalScore: 100 })); // التحقق من وجود المزيد (إذا كان الـ Worker أعاد 20 منتج، نفترض أن هناك المزيد) const hasMore = items.length >= 20; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } }); function buildSmartSearchQuery(title) { if (!title) return ''; const cleanTitle = title.toLowerCase().replace(/[^a-z0-9\s\-]/g, ' '); const words = cleanTitle.split(/\s+/).filter(w => w.length > 1 && !STOP_WORDS.has(w)); const originalWords = title.split(/\s+/); let brand = ''; for (let w of originalWords) { let cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 8 && cleanW === cleanW.toUpperCase() && /^[A-Z]+$/.test(cleanW)) { brand = cleanW.toLowerCase(); break; } } const models = words.filter(w => { const hasLetters = /[a-z]/.test(w); const hasNumbers = /[0-9]/.test(w); return hasLetters && hasNumbers; }); const model = models[0] || ''; let category = ''; if (/flash drive|usb drive|usb flash|cle usb|флешка/i.test(cleanTitle)) category = 'usb flash drive'; else if (/multimeter|voltmeter|мультиметр/i.test(cleanTitle)) category = 'multimeter'; else if (/label maker|dymo|принтер этикеток/i.test(cleanTitle)) category = 'label maker'; else if (/case|cover|coque|housse|чехол|حافظة/i.test(cleanTitle)) category = 'case'; else if (/tape|label tape|ruban|sticker|etiquette/i.test(cleanTitle)) category = 'label tape'; else if (/sweeper|vacuum|пилосос/i.test(cleanTitle)) category = 'vacuum cleaner'; else if (/earbuds|headphone|casque|навушники/i.test(cleanTitle)) category = 'earbuds'; let queryParts = []; if (brand) queryParts.push(brand); if (model && model !== brand) queryParts.push(model); if (category) queryParts.push(category); if (queryParts.length > 0) return queryParts.join(' '); return words.slice(0, 3).join(' '); }
