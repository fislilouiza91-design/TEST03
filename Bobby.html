export default { async fetch(request, env, ctx) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders }); const url = new URL(request.url); if (url.pathname === '/search-db') { let title = url.searchParams.get('q') || ''; const pid = url.searchParams.get('pid') || ''; const page = parseInt(url.searchParams.get('page')) || 1; const existingIds = url.searchParams.get('existingIds') || ''; try { if (!env.TURSO_URL || !env.TURSO_TOKEN) throw new Error("Missing Turso config"); let englishTitle = ''; const isNonEnglish = /[^\x00-\x7F]/.test(title); // 1. جلب العنوان الإنجليزي الأصلي من قاعدة البيانات if (pid) { try { const dbTitle = await getDbTitle(pid, env); if (dbTitle) englishTitle = dbTitle; } catch (e) {} } // 2. إذا لم نجد المنتج في القاعدة، نترجم العنوان if (!englishTitle && isNonEnglish) { englishTitle = await translateToEnglish(title); } if (!englishTitle) englishTitle = title; // 3. البحث الدقيق const result = await searchTursoDB(englishTitle, page, existingIds, env); return new Response(JSON.stringify(result), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } catch (e) { return new Response(JSON.stringify({ ok: false, error: e.message }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } } return new Response('Smart-Turso API is running.', { status: 200, headers: corsHeaders }); } } async function getDbTitle(pid, env) { const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const res = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: 'SELECT title FROM products WHERE product_id = ? LIMIT 1', args: [{ type: 'text', value: pid }] } }, { type: 'close' }] }) }); const data = await res.json(); if (data.results[0].type === 'error') return null; const rows = data.results[0].response.result.rows; return rows.length > 0 ? rows[0][0].value : null; } async function translateToEnglish(text) { if (!text) return text; try { const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=en&dt=t&q=${encodeURIComponent(text)}`; const res = await fetch(url); if (res.ok) { const data = await res.json(); if (data && data[0] && data[0][0]) return data[0].map(item => item[0]).join(''); } } catch (e) {} return text; } async function searchTursoDB(title, page, existingIds, env) { const stopWords = new Set([ 'for', 'the', 'and', 'with', 'from', 'new', 'hot', 'best', 'free', 'high', 'pro', 'sale', 'sales', 'shipping', 'pcs', 'piece', 'set', 'plus', 'original', 'official', 'good', 'quality', 'smart', 'portable', 'wireless', 'digital', 'type', 'model', 'color', 'size', 'men', 'women', 'kids', 'baby', 'home', 'car', 'phone', 'case', 'made', 'material', 'style', 'fashion', 'design', 'item', 'top', 'super', 'ultra', 'mini', 'max', 'premium', 'fast', 'pack', 'box', 'quick', 'release', 'replacement', 'gift', 'drop', 'ship', 'factory', 'direct', 'wholesale', 'dropshipping', 'performance', 'sports', 'racing', 'gas', 'cc', '4x', '10x', '100', 'mah', 'large', 'capacity', 'charging', 'charger', 'external', 'device', 'storage', 'capsule', 'cable', 'adapter', 'cover', 'protector', 'glass', 'iphone', 'samsung', 'xiaomi', 'huawei', 'redmi', 'poco', 'oppo', 'vivo', 'realme' ]); const allWords = title.toLowerCase().replace(/[^\w\s]/g, ' ').split(/\s+/); let meaningfulWords = allWords.filter(w => w.length > 2 && !stopWords.has(w) && !/^\d+$/.test(w)); if (meaningfulWords.length === 0) { meaningfulWords = allWords.filter(w => w.length > 1).slice(0, 2); } if (meaningfulWords.length === 0) return { ok: true, items: [], hasMore: false }; // نأخذ أول 5 كلمات مهمة let keywords = meaningfulWords.slice(0, 5); keywords = keywords.map(k => k.endsWith('s') && k.length > 4 ? k.slice(0, -1) : k); keywords = [...new Set(keywords)]; const conditions = keywords.map(() => `title LIKE ?`).join(' OR '); const queryArgs = keywords.map(k => ({ type: 'text', value: `%${k}%` })); let excludeCondition = ''; if (existingIds && existingIds.trim() !== '') { const ids = existingIds.split(','); excludeCondition = `AND product_id NOT IN (${ids.map(() => '?').join(',')})`; ids.forEach(id => queryArgs.push({ type: 'text', value: id })); } // 🔥 السر: نجلب 300 منتج لإتاحة الفرصة للفرز الدقيق const FETCH_LIMIT = 300; const OFFSET = (page - 1) * 60; const sql = ` SELECT product_id as id, title, image_url as img, current_price_usd as price, original_price_usd as oldPrice, rating, orders_count as sold FROM products WHERE (${conditions}) ${excludeCondition} LIMIT ${FETCH_LIMIT} OFFSET ${OFFSET} `; const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const tursoRes = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: sql, args: queryArgs } }, { type: 'close' }] }) }); if (!tursoRes.ok) throw new Error(`Turso HTTP Error: ${tursoRes.status}`); const data = await tursoRes.json(); const result = data.results[0]; if (result.type === 'error') throw new Error(result.error.message); const cols = result.response.result.cols.map(c => c.name); let rows = result.response.result.rows.map(row => { const obj = {}; cols.forEach((col, i) => { obj[col] = row[i].value; }); return obj; }); // 🔥 نظام النقاط (Scoring) لجعل النتائج دقيقة جداً rows.forEach(item => { const itemTitle = (item.title || '').toLowerCase(); let score = 0; // 1. إعطاء نقاط لكل كلمة مطابقة keywords.forEach(kw => { if (itemTitle.includes(kw)) score += 10; }); // 2. إعطاء نقاط إضافية إذا كان العنوان يحتوي على "كل" الكلمات (تطابق تام) const matchAll = keywords.every(kw => itemTitle.includes(kw)); if (matchAll && keywords.length > 1) score += 50; // 3. خصم نقاط إذا كان المنتج إكسسواراً (لمنع الكابلات والحافظات) const badWords = ['cable', 'case', 'cover', 'adapter', 'charger', 'protector']; badWords.forEach(bad => { if (itemTitle.includes(bad) && !title.toLowerCase().includes(bad)) { score -= 40; } }); item.score = score; }); // ترتيب المنتجات حسب النقاط (الأعلى أولاً) rows.sort((a, b) => b.score - a.score); // فلترة المنتجات التي ليس لها أي نقاط (لم تتطابق مع أي كلمة) rows = rows.filter(item => item.score > 0); // نأخذ أفضل 60 نتيجة فقط rows = rows.slice(0, 60); const items = rows.map(obj => { const discount = obj.oldPrice && obj.price ? Math.round(((obj.oldPrice - obj.price) / obj.oldPrice) * 100) : 0; return { id: obj.id, title: obj.title, img: obj.img || '', price: obj.price ? String(obj.price) : '', oldPrice: obj.oldPrice ? String(obj.oldPrice) : '', discount: discount > 0 ? String(discount) : '', rating: obj.rating ? String(obj.rating) : '', sold: obj.sold ? String(obj.sold) : '', link: `https://www.aliexpress.com/item/${obj.id}.html` }; }); return { ok: true, items: items, hasMore: items.length === 60 }; }
