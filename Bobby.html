export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } const url = new URL(request.url); // 🔥 نقطة النهاية: فرز المنتجات بالذكاء الاصطناعي النصي والبصري (المرحلة الجديدة) if (request.method === "POST" && url.pathname === "/sort") { try { const body = await request.json(); const { originalTitle, originalImage, items } = body; if (!originalTitle || !items || !Array.isArray(items)) { return new Response(JSON.stringify({ success: false, message: "Invalid data" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } if (!env.AI || items.length === 0) { return new Response(JSON.stringify({ success: true, data: items }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const modelName = '@cf/baai/bge-m3'; // 1️⃣ المرحلة الأولى: الفرز النصي الأولي لجميع المنتجات const originalEmbedding = await env.AI.run(modelName, { text: [originalTitle] }); const baseVector = originalEmbedding.data[0]; const itemTitles = items.map(it => it.title); let itemsEmbeddingData = []; for (let i = 0; i < itemTitles.length; i += 50) { const chunk = itemTitles.slice(i, i + 50); const emb = await env.AI.run(modelName, { text: chunk }); itemsEmbeddingData = itemsEmbeddingData.concat(emb.data); } let scoredItems = items.map((it, index) => { const itemVector = itemsEmbeddingData[index]; const aiScore = cosineSimilarity(baseVector, itemVector); return { ...it, aiScore, finalScore: aiScore * 100 }; }); // ترتيب مبدئي لأخذ أفضل منتجين scoredItems.sort((a, b) => b.finalScore - a.finalScore); const top2Items = scoredItems.slice(0, 2); // 2️⃣ المرحلة الثانية: التحليل البصري (أفضل منتجين + المنتج الأصلي) let visualContext = ''; const prompt = "Describe the product in this image in 5 words or less."; // دالة مساعدة لوصف الصورة const describeImage = async (imgUrl) => { if (!imgUrl) return ''; try { const res = await fetch(imgUrl); if (!res.ok) return ''; const buffer = await res.arrayBuffer(); const imageData = [...new Uint8Array(buffer)]; const aiResponse = await env.AI.run('@cf/llava-hf/llava-1.5-7b-hf', { image: imageData, prompt: prompt, max_tokens: 20 }); return aiResponse.description || aiResponse.response || ''; } catch (e) { return ''; } }; // تحليل الصور الثلاث بالتوازي لتسريع العملية const [descOriginal, descTop1, descTop2] = await Promise.all([ describeImage(originalImage), describeImage(top2Items[0]?.img), describeImage(top2Items[1]?.img) ]); if (descOriginal) visualContext += ` Original product image shows: ${descOriginal}.`; if (descTop1) visualContext += ` Similar product 1 image shows: ${descTop1}.`; if (descTop2) visualContext += ` Similar product 2 image shows: ${descTop2}.`; // 3️⃣ المرحلة الثالثة: بناء النص المرجعي الشامل وإعادة الترتيب // نجمع العنوان الأصلي + عناوين أفضل منتجين + الأوصاف البصرية const referenceText = ` ${originalTitle}. ${top2Items[0]?.title || ''}. ${top2Items[1]?.title || ''}. ${visualContext} `.replace(/\s+/g, ' ').trim(); // إنشاء متجه للنص المرجعي const refEmbedding = await env.AI.run(modelName, { text: [referenceText] }); const refVector = refEmbedding.data[0]; // إعادة حساب التشابه لكل منتج مع النص المرجعي الجديد const cleanOriginalTitle = originalTitle.toLowerCase(); scoredItems = items.map((it, index) => { const itemVector = itemsEmbeddingData[index]; const finalScore = cosineSimilarity(refVector, itemVector) * 100; const itemTitleLower = it.title.toLowerCase(); const isAccessory = (!/case|coque|housse|غطاء|حافظة/i.test(cleanOriginalTitle) && /case|coque|housse|غطاء|حافظة/i.test(itemTitleLower)) || (!/cable|charger|كابل|شاحن/i.test(cleanOriginalTitle) && /cable|charger|كابل|شاحن/i.test(itemTitleLower)) || /protector|sticker|skin|sleeve|stand|holder|حماية|ملصق/i.test(itemTitleLower); // إذا كان المنتج من أفضل 2، نضمن بقاءه في القمة const isTop2 = top2Items.some(t => t.id === it.id); let adjustedScore = isTop2 ? finalScore + 100 : finalScore; const isExact = !isAccessory && (adjustedScore > 60 || isTop2); return { ...it, finalScore: adjustedScore, isExact }; }); scoredItems.sort((a, b) => b.finalScore - a.finalScore); return new Response(JSON.stringify({ success: true, data: scoredItems }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } // ========================================== // الكود القديم الخاص بك (جلب المنتجات) - لم يتغير // ========================================== let keyword = url.searchParams.get("q"); const originalTitle = url.searchParams.get("originalTitle") || ""; const page = parseInt(url.searchParams.get("page") || "1"); if (page > 3) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { if (originalTitle && env.AI) { keyword = await extractKeywordsWithAI(originalTitle, env); } if (!keyword) { return new Response(JSON.stringify({ success: false, message: "Missing keyword" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${env.SCRAPINGANT_API_KEY}&return_page_source=true&proxy_type=datacenter&browser=false`; const antResponse = await fetch(antUrl); if (!antResponse.ok) throw new Error(`ScrapingAnt Error: ${antResponse.status}`); const html = await antResponse.text(); if (html.includes('punish') || html.includes('captcha') || html.length < 5000) { return new Response(JSON.stringify({ success: false, message: "كابتشا أو حظر مؤقت." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } let products = parseItems(html); if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لا توجد منتجات." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } return new Response(JSON.stringify({ success: true, data: products, aiQuery: keyword }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // 🧠 دوال الذكاء الاصطناعي والاستخراج // ========================================== async function extractKeywordsWithAI(title, env) { try { const response = await env.AI.run('@cf/meta/llama-3-8b-instruct', { messages: [ { role: 'system', content: 'You are an e-commerce search expert. Extract the main product type, brand, and model from the title. Output ONLY 3 to 4 English keywords separated by spaces. Do NOT include colors, sizes, numbers, or promotional words like "free shipping". Do NOT write sentences.' }, { role: 'user', content: title } ], max_tokens: 20 }); if (response && response.response) { let cleanQuery = response.response.trim().replace(/[^a-zA-Z0-9\s]/g, ''); return cleanQuery.split(/\s+/).slice(0, 4).join(' '); } return ''; } catch (e) { return ''; } } function cosineSimilarity(vecA, vecB) { let dotProduct = 0, normA = 0, normB = 0; for (let i = 0; i < vecA.length; i++) { dotProduct += vecA[i] * vecB[i]; normA += vecA[i] * vecA[i]; normB += vecB[i] * vecB[i]; } return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB)); } function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const priceNum = extractPriceNumber(priceStr); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img: img, price: priceNum, rating: parseFloat(rating) || 0, soldStr: sold, url: `https://www.aliexpress.com/item/${id}.html` }); } catch (e) {} } return out; }
