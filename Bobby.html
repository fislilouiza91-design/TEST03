const MAX_PAGES_TO_FETCH = 3; // قائمة الكلمات الدعائية والحشو التي يجب تجاهلها const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'best', 'top', 'high', 'good', 'perfect', '2023', '2024', '2025', '2026', 'pcs', 'set', 'pack', 'piece', 'pieces', 'lot', 'items', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'no', 'yes', 'you', 'your', 'our', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'et', 'est', 'un', 'le', 'la', 'для', 'и', 'в', 'на', 'с', 'по', 'из', 'double', 'door', 'enclosed', 'large', 'angle', 'flip', 'cover', 'removable', 'easy', 'clean', 'waterproof', 'portable', 'multifunctional', 'design', 'filter' ]); // قاموس العلامات التجارية الشهيرة const KNOWN_BRANDS = [ 'apple', 'samsung', 'xiaomi', 'redmi', 'poco', 'huawei', 'honor', 'oppo', 'vivo', 'oneplus', 'realme', 'lenovo', 'hp', 'dell', 'asus', 'acer', 'msi', 'sony', 'lg', 'panasonic', 'philips', 'anker', 'baseus', 'ugreen', 'romoss', 'logitech', 'razer', 'corsair', 'nike', 'adidas', 'puma', 'dyson', 'ecovacs', 'roborock' ]; // قاموس الفئات الأساسية (للبحث عن المنتج الأساسي وليس الصفة) const CATEGORIES = { 'earbuds': ['earbuds', 'headphone', 'headset', 'airpods', 'casque', 'навушники'], 'watch': ['watch', 'smartwatch', 'montre', 'часы', 'strap', 'watchband'], 'case': ['case', 'cover', 'coque', 'housse', 'чехол', 'حافظة'], 'charger': ['charger', 'charging', 'adapter', 'chargeur', 'зарядка'], 'cable': ['cable', 'cord', 'câble', 'кабель'], 'powerbank': ['powerbank', 'power bank', 'battery', 'batterie', 'аккумулятор'], 'mouse': ['mouse', 'souris', 'мышь'], 'keyboard': ['keyboard', 'clavier', 'клавиатура'], 'lamp': ['lamp', 'light', 'led', 'lampe', 'лампа', 'светильник'], 'vacuum': ['vacuum', 'sweeper', 'aspirateur', 'пилосос'], 'drone': ['drone', 'дрон'], 'camera': ['camera', 'cam', 'caméra', 'камера'], 'speaker': ['speaker', 'soundbar', 'haut-parleur', 'колонка'], 'tool': ['tool', 'screwdriver', 'wrench', 'plier', 'drill', 'multimeter', 'voltmeter'] }; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } console.log(`🎯 الكلمات المفتاحية المستخرجة للبحث: [${searchQuery}]`); const q = encodeURIComponent(searchQuery); const cookieParam = msg.cookie ? `&cookie=${encodeURIComponent(msg.cookie)}` : ''; fetch(`${WORKER_URL}?q=${q}&page=${page}${cookieParam}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: 100 })); let hasMore = items.length >= 20; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } }); // ========================================== // 🔥 محرك استخراج الكلمات المفتاحية الذكي // ========================================== function buildSmartSearchQuery(title) { if (!title) return ''; // 1. تنظيف العنوان من الرموز والإيموجي const cleanTitle = title.replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ').replace(/\s+/g, ' ').trim(); const lowerTitle = cleanTitle.toLowerCase(); const words = cleanTitle.split(' '); // 2. استخراج البراند (Brand) let brand = ''; for (let w of words) { const cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 10) { if (KNOWN_BRANDS.includes(cleanW.toLowerCase())) { brand = cleanW.toLowerCase(); break; } // إذا كانت الكلمة كلها أحرف كبيرة (وكانت ضمن أول 3 كلمات) if (w === w.toUpperCase() && /^[A-Z]+$/.test(cleanW) && words.indexOf(w) < 3) { brand = cleanW.toLowerCase(); break; } } } // 3. استخراج الموديل (Model) - كلمة تحتوي على حروف وأرقام let model = ''; for (let w of words) { const cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 12) { const hasLetters = /[a-zA-Z]/.test(cleanW); const hasNumbers = /[0-9]/.test(cleanW); if (hasLetters && hasNumbers && !KNOWN_BRANDS.includes(cleanW.toLowerCase()) && cleanW !== brand) { model = cleanW.toLowerCase(); break; } } } // 4. استخراج الفئة الأساسية (Category) let category = ''; for (const [catKey, keywords] of Object.entries(CATEGORIES)) { if (keywords.some(k => lowerTitle.includes(k))) { category = catKey; break; } } // 5. تجميع الكلمات المفتاحية النهائية let parts = []; if (brand) parts.push(brand); if (model && model !== brand) parts.push(model); if (category) parts.push(category); // إذا لم نجد فئة معروفة، نأخذ أطول كلمتين متبقيتين if (!category) { const remainWords = words .map(w => w.toLowerCase().replace(/[^a-z0-9]/g, '')) .filter(w => w.length > 2 && !STOP_WORDS.has(w) && w !== brand && w !== model) .sort((a, b) => b.length - a.length); parts.push(...remainWords.slice(0, 2)); } // إذا كان الناتج فارغاً، نأخذ أول 3 كلمات من العنوان الأصلي if (parts.length === 0) { return words.slice(0, 3).join(' '); } // إزالة التكرار وإرجاع النتيجة النهائية (مثلاً: "xiaomi earbuds" أو "iphone 15 case") return [...new Set(parts)].join(' '); }
