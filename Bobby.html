export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } const url = new URL(request.url); if (request.method === "POST" && url.pathname === "/sort") { try { const body = await request.json(); const { originalTitle, items } = body; if (!originalTitle || !items || !Array.isArray(items)) { return new Response(JSON.stringify({ success: false, message: "Invalid data" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } if (!env.AI || items.length === 0) { return new Response(JSON.stringify({ success: true, data: items }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const modelName = '@cf/baai/bge-m3'; // 🔥 دالة تجزئة بسيطة const STOP_WORDS = new Set([ 'the', 'and', 'for', 'with', 'new', 'hot', 'sale', 'free', 'shipping', 'original', 'high', 'quality', 'best', 'pcs', 'set', 'pack', 'piece', 'pieces', 'items', 'item', 'wholesale', 'dropshipping', 'compatible', 'fits', 'fit', 'suitable', 'accessories', 'accessory', 'universal' ]); const tokenize = (text) => { return text.toLowerCase() .replace(/[^a-z0-9\u0600-\u06FF\u0400-\u04FF\s]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !STOP_WORDS.has(w)); }; const originalTokensList = tokenize(originalTitle); const originalTokens = new Set(originalTokensList); // ========================================== // المرحلة 1: فرز محلي لاختيار أفضل 5 مرشحين // ========================================== const preScoredItems = items.map(it => { const itemTokens = new Set(tokenize(it.title)); if (itemTokens.size === 0) return { ...it, preScore: 0 }; const intersection = [...itemTokens].filter(w => originalTokens.has(w)).length; const union = new Set([...itemTokens, ...originalTokens]).size; const jaccard = union > 0 ? const baseVector = originalEmbedding.data[0]; const aiTitles = aiItems.map(it => it.title); const aiEmbeddings = await env.AI.run(modelName, { text: aiTitles }); let scoredAIItems = aiItems.map((it, index) => { const aiScore = cosineSimilarity(baseVector, aiEmbeddings.data[index]); return { ...it, finalScore, isExact }; }); scoredRemaining.sort((a, b) => b.finalScore - a.finalScore); // ========================================== // المرحلة 5: الدمج النهائي // ========================================== const finalItems = [...scoredAIItems, ...scoredRemaining].map(it => { // إزالة الحقول المؤقتة const { _type, _acc, ...clean } = it; return clean; }); return new Response(JSON.stringify({ success: true, data: finalItems }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } // ========================================== // الكود القديم (جلب المنتجات) // ========================================== let keyword = url.searchParams.get("q"); const originalTitle = url.searchParams.get("originalTitle") || ""; const page = parseInt(url.searchParams.get("page") || "1"); if (page > 3) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { if (originalTitle && env.AI) { keyword = await extractKeywordsWithAI(originalTitle, env); } if (!keyword) { return new Response(JSON.stringify({ success: false, message: "Missing keyword" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${env.SCRAPINGANT_API_KEY}&return_page_source=true&proxy_type=datacenter&browser=false`; } let products = parseItems(html); if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لا توجد منتجات." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } return new Response(JSON.stringify({ success: true, data: products, aiQuery: keyword }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // 🧠 دوال مساعدة (لم تتغير) // ========================================== async function extractKeywordsWithAI(title, env) { try { const response = await env.AI.run('@cf/meta/llama-3-8b-instruct', { messages: [ { role: 'system', content: 'You are an e-commerce search expert. Extract the main product type, brand, and model from the title. Output ONLY 3 to 4 English keywords separated by spaces. Do NOT include colors, sizes, numbers, or promotional words like "free shipping". Do NOT write sentences.' }, { role: 'user', content: title } ], max_tokens: 20 }); if (response && response.response) { let cleanQuery = response.response.trim().replace(/[^a-zA-Z0-9\s]/g, ''); return cleanQuery.split(/\s+/).slice(0, 4).join(' '); } return ''; } catch (e) { return ''; } } function cosineSimilarity(vecA, vecB) { let dotProduct = 0, normA = 0, normB = 0; for (let i = 0; i < vecA.length; i++) { dotProduct += vecA[i] * vecB[i]; normA += vecA[i] * vecA[i]; normB += vecB[i] * vecB[i]; } return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB)); } function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ?
