/* ============================================ Smart Shopper - Background Service Worker (Smart Category & Brand Search) ============================================ */ const WORKER_URL = 'https://competitors-api.smailsuit11.workers.dev'; const MAX_PAGES_TO_FETCH = 3; // الحد الأقصى 3 طلبات const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'high', 'good', '2023', '2024', '2025', '2026', 'best', 'top', 'pcs', 'set', 'pack', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par' ]); chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { fetch(`${WORKER_URL}/api/competitors?productId=${msg.productId}`) .then(res => { if(res.ok) return res.json(); throw new Error('Not found'); }) .then(data => sendResponse({ ok: true, data: data.data })) .catch(() => sendResponse({ ok: false })); return true; } if (msg.type === 'SMART_SEARCH') { // 🔥 بناء استعلام البحث الذكي والشامل const searchQuery = buildSmartSearchQuery(msg.title); console.log('🔍 Comprehensive Search Query:', searchQuery); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=1&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { sendResponse({ ok: true, items: data.items, hasMore: data.hasMore }); if (msg.productId && data.items.length > 0) saveToCloudflare(msg.productId, data.items); } else { sendResponse({ ok: false, error: data.error }); } }).catch(err => sendResponse({ ok: false, error: err.message })); return true; } if (msg.type === 'SMART_LOAD_MORE') { const nextPage = Math.max(2, parseInt(msg.startPage) || 2); if (nextPage > MAX_PAGES_TO_FETCH) return sendResponse({ ok: true, items: [], hasMore: false }); const searchQuery = buildSmartSearchQuery(msg.title); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=${nextPage}&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { let existingIds = new Set(msg.existingIds); let unique = data.items.filter(i => !existingIds.has(i.id)); sendResponse({ ok: true, items: unique, hasMore: data.hasMore }); if (msg.productId && unique.length > 0) saveToCloudflare(msg.productId, unique); } else { sendResponse({ ok: false, error: data.error, hasMore: false }); } }).catch(err => sendResponse({ ok: false, error: err.message, hasMore: false })); return true; } }); // 🔥 استراتيجية البحث الشاملة (الماركة + الموديل + التصنيف) function buildSmartSearchQuery(title) { if (!title) return ''; const cleanTitle = title.toLowerCase().replace(/[^a-z0-9\s]/g, ' '); const words = cleanTitle.split(/\s+/).filter(w => w.length > 1 && !STOP_WORDS.has(w)); // 1. استخراج الماركة (أول كلمة مكتوبة بحروف كبيرة) const originalWords = title.split(/\s+/); let brand = ''; for (let w of originalWords) { let cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 6 && cleanW === cleanW.toUpperCase() && /^[A-Z]+$/.test(cleanW)) { brand = cleanW.toLowerCase(); break; } } if (!brand && words.length > 0) brand = words[0]; // 2. استخراج الموديل const models = words.filter(w => { const hasLetters = /[a-z]/.test(w); const hasNumbers = /[0-9]/.test(w); const isVersion = w.includes('.'); const isCapacity = /^\d+[h|m|mah|w]$/i.test(w); return hasLetters && hasNumbers && !isVersion && !isCapacity; }); // 3. تحديد التصنيف العام (لضمان جلب المنتجات المشابهة حتى لو لم تحمل نفس الاسم) let category = ''; if (cleanTitle.includes('airtag') || cleanTitle.includes('tracker') || cleanTitle.includes('locator') || cleanTitle.includes('find my')) category = 'tracker'; else if (cleanTitle.includes('casque') || cleanTitle.includes('headphone') || cleanTitle.includes('écouteur') || cleanTitle.includes('earbuds')) category = 'earbuds'; else if (cleanTitle.includes('montre') || cleanTitle.includes('watch')) category = 'watch'; else if (cleanTitle.includes('chargeur') || cleanTitle.includes('charger')) category = 'charger'; // 4. بناء الاستعلام الذكي if (models.length > 0 && category) { // إذا كان لدينا موديل وتصنيف (مثل: e96a tracker) -> هذا سيجلب المطابق والمشابه return `${models[0]} ${category}`; } else if (brand && category) { // إذا لم نجد موديل، نبحث بالماركة والتصنيف (مثل: hoco tracker) return `${brand} ${category}`; } else if (models.length > 0) { return models[0]; } else { return words.slice(0, 3).join(' '); } } function saveToCloudflare(productId, items) { const payload = { base_product_id: productId, all_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })), relevant_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })) }; fetch(`${WORKER_URL}/api/competitors`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).then(res => res.json()).then(data => console.log('💾 Saved:', data)).catch(console.error); }
