/* ============================================ Proxy Manager - v3.0 (Text Sources) ============================================ */ // مصادر نصية موثوقة (تعيد IP:PORT مباشرة) const PROXY_SOURCES = [ 'https://raw.githubusercontent.com/TheSpeedX/PROXY-List/master/http.txt', 'https://raw.githubusercontent.com/TheSpeedX/PROXY-List/master/socks5.txt', 'https://raw.githubusercontent.com/monosans/proxy-list/main/proxies/http.txt' ]; let currentProxy = null; let proxyPool = []; let proxyIndex = 0; let authCredentials = { username: '', password: '' }; /* ============================================ 1. جلب بروكسيات جديدة ============================================ */ export async function refreshProxyPool() { const all = []; for (const url of PROXY_SOURCES) { try { const res = await fetch(url, { signal: AbortSignal.timeout(10000) }); if (!res.ok) continue; const text = await res.text(); const parsed = parseProxyText(text); all.push(...parsed); console.log(`[Proxy] ✅ ${parsed.length} بروكسي من ${url.split('/')[4]}`); } catch (e) { console.warn(`[Proxy] ❌ فشل جلب من ${url}:`, e.message); } } // إزالة التكرارات وخلط القائمة const unique = Array.from( new Map(all.map(p => [`${p.host}:${p.port}`, p])).values() ); unique.sort(() => Math.random() - 0.5); proxyPool = unique; proxyIndex = 0; console.log(`[Proxy] 📦 جاهز ${proxyPool.length} بروكسي`); await chrome.storage.local.set({ proxyPool: proxyPool.slice(0, 1000), poolUpdatedAt: Date.now() }); return proxyPool.length; } // دالة تحليل النص (ip:port) function parseProxyText(text) { const out = []; const lines = text.split('\n'); for (let line of lines) { line = line.trim(); // تجاهل التعليقات والأسطر الفارغة if (!line || line.startsWith('#')) continue; const parts = line.split(':'); if (parts.length === 2) { const host = parts[0]; const port = parseInt(parts[1], 10); if (host && !isNaN(port) && port > 0) { out.push({ scheme: 'http', host, port, country: '' }); } } } return out; } /* ============================================ 2. اختبار البروكسي ============================================ */ async function testProxy(proxy, timeoutMs = 5000) { try { await applyProxyInternal(proxy); // نستخدم httpbin لاختبار الاتصال const res = await fetch(`https://httpbin.org/ip?t=${Date.now()}`, { signal: AbortSignal.timeout(timeoutMs) }); if (!res.ok) return false; const data = await res.json(); return !!data.origin; } catch (e) { return false; } } /* ============================================ 3. تطبيق البروكسي (مؤقت للاختبار) ============================================ */ async function applyProxyInternal(proxy) { authCredentials = { username: proxy.username || '', password: proxy.password || '' }; await chrome.proxy.settings.set({ scope: 'regular', value: { mode: 'fixed_servers', rules: { singleProxy: { scheme: proxy.scheme || 'http', host: proxy.host, port: proxy.port }, bypassList: ['localhost', '127.0.0.1', ''] } } }); } /* ============================================ 4. تطبيق PAC لتوجيه AliExpress فقط ============================================ */ export async function applyProxyForAliExpress(proxy) { if (!proxy) { await chrome.proxy.settings.clear({ scope: 'regular' }); currentProxy = null; return; } currentProxy = proxy; authCredentials = { username: proxy.username || '', password: proxy.password || '' }; // ⚠️ سكربت PAC يجب أن يكون ASCII فقط (بدون تعليقات عربية) const pacScript = `function FindProxyForURL(url, host) { if (host.indexOf("aliexpress.com") !== -1 || host.indexOf("aliexpress.us") !== -1) { return "PROXY ${proxy.host}:${proxy.port}"; } return "DIRECT"; }`; await chrome.proxy.settings.set({ scope: 'regular', value: { mode: 'pac_script', pacScript: { data: pacScript } } }); console.log(`[Proxy] 🎯 AliExpress → ${proxy.host}:${proxy.port}`); } /* ============================================ 5. الحصول على بروكسي جديد ============================================ */ export async function getNextProxy(maxTries = 5) { // إذا كانت القائمة فارغة، جلب من جديد if (proxyPool.length === 0) { const stored = await chrome.storage.local.get('proxyPool'); if (stored.proxyPool && stored.proxyPool.length > 0) { proxyPool = stored.proxyPool; } else { await refreshProxyPool(); } } if (proxyPool.length === 0) { console.warn('[Proxy] ⚠️ لا توجد بروكسيات متاحة، سيتم التصفح مباشرة'); return null; } // محاولة اختبار 5 بروكسيات for (let i = 0; i < maxTries && proxyIndex < proxyPool.length; i++) { const candidate = proxyPool[proxyIndex++]; console.log(`[Proxy] 🧪 اختبار ${candidate.host}:${candidate.port} (${i + 1}/${maxTries})`); const works = await testProxy(candidate, 4000); if (works) { console.log(`[Proxy] ✅ بروكسي صالح: ${candidate.host}:${candidate.port}`); return candidate; } } console.warn('[Proxy] ❌ فشلت المحاولات الحالية، جلب قائمة جديدة...'); await refreshProxyPool(); if (proxyPool.length > 0) { return proxyPool[0]; } return null; } /* ============================================ 6. مصادقة البروكسي ============================================ */ export function setupProxyAuth() { chrome.webRequest.onAuthRequired.addListener( (details, callback) => { if (authCredentials.username || authCredentials.password) { callback({ authCredentials }); } else { callback({}); } }, { urls: [''] }, ['asyncBlocking'] ); } /* ============================================ 7. حالة البروكسي ============================================ */ export function getStatus() { return { currentProxy, poolSize: proxyPool.length, remaining: proxyPool.length - proxyIndex };
