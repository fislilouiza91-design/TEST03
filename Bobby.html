export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } const url = new URL(request.url); const keyword = url.searchParams.get("q"); const page = parseInt(url.searchParams.get("page") || "1"); if (!keyword) { return new Response(JSON.stringify({ success: false, message: "Missing keyword" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } if (page > 3) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { // 🧠 السر الأول: التخزين المؤقت (KV Cache) - يقلل الاستهلاك بنسبة 80% const cacheKey = `search:${keyword.toLowerCase()}:${page}`; if (env.COMPETITOR_CACHE) { const cached = await env.COMPETITOR_CACHE.get(cacheKey, 'json'); if (cached) { return new Response(JSON.stringify({ success: true, data: cached }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } // 🧠 السر الثاني: الرابط الذكي (يجلب نتائج أكثر بكثير) const searchUrl = `https://www.aliexpress.com/w/wholesale-${keyword.replace(/\s+/g, '-').toLowerCase()}.html?page=${page}`; // 🧠 السر الثالث: إعدادات ScrapingAnt (بدون متصفح + تحديد الدولة لتجنب الحظر) const antUrl = new URL('https://api.scrapingant.com/v2/general'); antUrl.searchParams.set('url', searchUrl); antUrl.searchParams.set('x-api-key', env.SCRAPINGANT_API_KEY); antUrl.searchParams.set('return_page_source', 'true'); antUrl.searchParams.set('proxy_country', 'us'); // ضروري جداً antUrl.searchParams.set('browser', 'false'); // 🔥 يوفر نصف الرصيد antUrl.searchParams.set('headers', JSON.stringify({'Accept-Language': 'en-US,en;q=0.9'})); const antResponse = await fetch(antUrl.toString()); if (!antResponse.ok) throw new Error(`ScrapingAnt Error: ${antResponse.status}`); const html = await antResponse.text(); // التحقق من وجود كابتشا if (html.includes('punish') || html.includes('captcha') || html.length < 5000) { return new Response(JSON.stringify({ success: false, message: "تم حظر الطلب من AliExpress (كابتشا). يرجى المحاولة لاحقاً." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // استخدام دالة الاستخراج الخاصة بك (التي تعمل 100%) const products = parseItems(html); if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لم يتم العثور على منتجات في هذه الصفحة." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } // 🧠 حفظ النتائج في الكاش لمدة 5 أيام لتوفير الرصيد للزيارات القادمة if (env.COMPETITOR_CACHE) { await env.COMPETITOR_CACHE.put(cacheKey, JSON.stringify(products), { expirationTtl: 432000 }); } return new Response(JSON.stringify({ success: true, data: products }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Cache-Control": "s-maxage=3600" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // دوال الاستخراج (الخاصة بك - بدون أي تعديل) // ========================================== function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const priceNum = extractPriceNumber(priceStr); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img: img, price: priceNum, rating: parseFloat(rating) || 0, soldStr: sold, url: `https://www.aliexpress.com/item/${id}.html` }); } catch (e) {} } return out; }
