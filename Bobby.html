import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); async function main() { console.log('🔧 ترحيل قاعدة البيانات...\n'); // 1. احذف الجدول القديم console.log('1️⃣ حذف الجدول القديم...'); await db.execute('DROP TABLE IF EXISTS products_current'); console.log(' ✅ تم'); // 2. أنشئ الجدول الجديد console.log('\n2️⃣ إنشاء الجدول الجديد...'); await db.execute(` CREATE TABLE products_current ( product_id TEXT, keyword TEXT, language TEXT DEFAULT 'en', title TEXT, current_price REAL, original_price REAL, orders_count INTEGER DEFAULT 0, rating REAL DEFAULT 0, image_url TEXT, seller_name TEXT, seller_rating REAL DEFAULT 0, shipping_days INTEGER DEFAULT 0, is_free_shipping INTEGER DEFAULT 0, is_choice INTEGER DEFAULT 0, first_seen_at INTEGER, last_updated_at INTEGER, update_count INTEGER DEFAULT 1, PRIMARY KEY (product_id, keyword, language) ) `); console.log(' ✅ تم'); // 3. الفهارس console.log('\n3️⃣ إنشاء الفهارس...'); await db.execute('CREATE INDEX idx_keyword ON products_current(keyword)'); await db.execute('CREATE INDEX idx_orders ON products_current(keyword, orders_count DESC)'); await db.execute('CREATE INDEX idx_lang ON products_current(language)'); console.log(' ✅ تم'); // 4. تحقق console.log('\n4️⃣ التحقق:'); const schema = await db.execute("PRAGMA table_info(products_current)"); console.log(` عدد الأعمدة: ${schema.rows.length}`); schema.rows.forEach(r => console.log(` - ${r.name}`)); const count = await db.execute('SELECT COUNT(*) AS c FROM products_current'); console.log(`\n📦 عدد الصفوف: ${count.rows[0].c} (يجب 0)`); console.log('\n🎉 اكتمل الترحيل بنجاح!'); } main().catch(error => { console.error('❌ خطأ:', error); process.exit(1); });
