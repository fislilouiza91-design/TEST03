const grid = document.getElementById('grid'); let hasHistory = false; function renderCards(products) { grid.innerHTML = ''; products.forEach(item => { const card = document.createElement('a'); card.href = item.url; card.target = '_blank'; card.className = 'card'; let img = item.img; if (img && img.startsWith('//')) img = 'https:' + img; if (!img) img = 'https://via.placeholder.com/100?text=No+Image'; card.innerHTML = `
${item.title}
${item.price}
`; grid.appendChild(card); }); } // 1. نحاول جلب السجل أولاً chrome.storage.local.get(['test_history'], (res) => { const history = res.test_history || []; if (history.length > 0) { hasHistory = true; renderCards(history); } else { // إذا لم يكن هناك سجل، نضع رسالة مؤقتة وننتظر الرسالة من الصفحة grid.innerHTML = '
Loading recommended products...
'; } }); // 2. نستمع للرسالة القادمة من content.js (إذا كان السجل فارغاً) window.addEventListener('message', (event) => { if (event.data && event.data.action === 'HOMEPAGE_PRODUCTS' && !hasHistory) { if (event.data.products && event.data.products.length > 0) { renderCards(event.data.products); } else { grid.innerHTML = '
No products found on this page.
Browse some items to build your history!
'; } } });
