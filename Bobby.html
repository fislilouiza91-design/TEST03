/* ============================================ Smart Shopper - Background Service Worker (Aggressive Keyword Extraction) ============================================ */ const WORKER_URL = 'https://competitors-api.smailsuit11.workers.dev'; const MAX_PAGES_TO_FETCH = 3; // قائمة كلمات الحشو والصفات التي تدمر البحث (يجب حذفها) const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'high', 'good', '2023', '2024', '2025', '2026', 'best', 'top', 'pcs', 'set', 'pack', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'double', 'door', 'enclosed', 'large', 'angle', 'flip', 'cover', 'removable', 'easy', 'clean', 'waterproof', 'portable', 'multifunctional', 'design', 'filter' ]); chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { fetch(`${WORKER_URL}/api/competitors?productId=${msg.productId}`) .then(res => { if(res.ok) return res.json(); throw new Error('Not found'); }) .then(data => sendResponse({ ok: true, data: data.data })) .catch(() => sendResponse({ ok: false })); return true; } if (msg.type === 'SMART_SEARCH') { const searchQuery = buildSmartSearchQuery(msg.title); console.log('✂️ Cleaned Search Query:', searchQuery); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=1&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { sendResponse({ ok: true, items: data.items, hasMore: data.hasMore }); if (msg.productId && data.items.length > 0) saveToCloudflare(msg.productId, data.items); } else { sendResponse({ ok: false, error: data.error }); } }).catch(err => sendResponse({ ok: false, error: err.message })); return true; } if (msg.type === 'SMART_LOAD_MORE') { const nextPage = Math.max(2, parseInt(msg.startPage) || 2); if (nextPage > MAX_PAGES_TO_FETCH) return sendResponse({ ok: true, items: [], hasMore: false }); const searchQuery = buildSmartSearchQuery(msg.title); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=${nextPage}&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { let existingIds = new Set(msg.existingIds); let unique = data.items.filter(i => !existingIds.has(i.id)); sendResponse({ ok: true, items: unique, hasMore: data.hasMore }); if (msg.productId && unique.length > 0) saveToCloudflare(msg.productId, unique); } else { sendResponse({ ok: false, error: data.error, hasMore: false }); } }).catch(err => sendResponse({ ok: false, error: err.message, hasMore: false })); return true; } }); // 🔥 خوارزمية "المقص" لاستخراج الكلمات الجوهرية فقط function buildSmartSearchQuery(title) { if (!title) return ''; const cleanTitle = title.toLowerCase().replace(/[^a-z0-9\s]/g, ' '); const words = cleanTitle.split(/\s+/).filter(w => w.length > 1 && !STOP_WORDS.has(w)); // محاولة إيجاد ماركة (حروف كبيرة) const originalWords = title.split(/\s+/); let brand = ''; for (let w of originalWords) { let cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 6 && cleanW === cleanW.toUpperCase() && /^[A-Z]+$/.test(cleanW)) { brand = cleanW.toLowerCase(); break; } } // إزالة الكلمات التي تحتوي على شرطات (double-door) أو أرقام إصدارات const coreWords = words.filter(w => !w.includes('-') && !/^\d+[a-z]?$/i.test(w)); // إذا وجدنا ماركة، نأخذ الماركة + أول 3 كلمات أساسية if (brand) { return `${brand} ${coreWords.slice(0, 3).join(' ')}`.trim(); } // إذا لم نجد ماركة، نأخذ أول 3 كلمات أساسية فقط (والتي يجب أن تكون اسم المنتج الحقيقي) // مثلاً: caja de arena para gatos -> caja arena gatos return coreWords.slice(0, 3).join(' '); } function saveToCloudflare(productId, items) { const payload = { base_product_id: productId, all_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })), relevant_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })) }; fetch(`${WORKER_URL}/api/competitors`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).then(res => res.json()).then(data => console.log('💾 Saved:', data)).catch(console.error); }
