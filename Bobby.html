export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } const url = new URL(request.url); const keyword = url.searchParams.get("q"); const page = parseInt(url.searchParams.get("page") || "1"); const originalTitle = url.searchParams.get("originalTitle") || ""; const originalPrice = parseFloat(url.searchParams.get("originalPrice") || "0"); if (!keyword) { return new Response(JSON.stringify({ success: false, message: "Missing keyword" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } if (page > 2) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${env.SCRAPINGANT_API_KEY}&return_page_source=true&proxy_type=datacenter&browser=false&proxy_country=us`; const antResponse = await fetch(antUrl); if (!antResponse.ok) throw new Error(`ScrapingAnt Error: ${antResponse.status}`); const html = await antResponse.text(); if (html.includes('punish') || html.includes('captcha') || html.length < 5000) { return new Response(JSON.stringify({ success: false, message: "كابتشا أو حظر مؤقت." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } let products = parseItems(html); // 🔥 تطبيق البحث الذكي مع تقارب الأسعار if (originalTitle && products.length > 0) { products = smartFilterAndSort(originalTitle, originalPrice, products); } if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لا توجد منتجات مطابقة." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } return new Response(JSON.stringify({ success: true, data: products }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // 🧠 دالة البحث الذكي وتقارب الأسعار // ========================================== function smartFilterAndSort(originalTitle, originalPrice, items) { if (!items || items.length === 0) return items; const cleanTitle = originalTitle.toLowerCase().replace(/[^\w\s]/g, ' '); const words = cleanTitle.split(/\s+/).filter(w => w.length > 1); // استخراج البراند let brand = ''; for (let w of originalTitle.split(/\s+/)) { const cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW === cleanW.toUpperCase() && /^[A-Z]+$/.test(cleanW)) { brand = cleanW.toLowerCase(); break; } } // استخراج الموديل const model = words.find(w => /[a-z]/.test(w) && /[0-9]/.test(w)); // هل المنتج الأصلي إكسسوار؟ const isAccessory = /case|cover|coque|housse|protector|film|screen|cable|charger|adapter|strap|band|bag|pouch|skin|sleeve|stand|holder/i.test(cleanTitle); const stopWords = new Set(['the','and','for','with','new','hot','sale','free','shipping','original','high','quality','best','pcs','set','pack','piece','items','item','wholesale','dropshipping','compatible','fits','fit','suitable','accessories']); const essentialWords = words.filter(w => !stopWords.has(w) && w.length > 2); return items.map(item => { const itemTitle = item.title.toLowerCase(); let score = 0; let reject = false; // 1. فلترة البراند if (brand && !itemTitle.includes(brand)) reject = true; else if (brand) score += 30; // 2. فلترة الموديل if (model && !itemTitle.includes(model)) reject = true; else if (model) score += 40; // 3. فلترة الإكسسوارات if (!isAccessory) { const itemIsAccessory = /case|cover|coque|housse|protector|film|screen|cable|charger|adapter|strap|band|bag|pouch|skin|sleeve|stand|holder/i.test(itemTitle); if (itemIsAccessory) reject = true; } // 4. 🔥 تقارب الأسعار (Price Proximity) 🔥 if (originalPrice > 0 && item.price > 0) { const ratio = item.price / originalPrice; // رفض المنتجات البعيدة جداً في السعر if (ratio < 0.25 || ratio > 4) reject = true; // مكافأة السعر القريب جداً if (ratio >= 0.7 && ratio <= 1.3) score += 20; // مكافأة السعر الأرخص if (ratio < 1 && ratio >= 0.5) score += 15; // عقوبة السعر الأغلى بكثير if (ratio > 1.5) score -= 10; } // 5. تطابق الكلمات الجوهرية const matchedWords = essentialWords.filter(w => itemTitle.includes(w)).length; score += matchedWords * 5; if (essentialWords.length >= 3 && matchedWords < 2) reject = true; return reject ? null : { ...item, score }; }) .filter(item => item !== null) .sort((a, b) => b.score - a.score); // ترتيب حسب الأكثر دقة } // ========================================== // دوال الاستخراج // ========================================== function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const priceNum = extractPriceNumber(priceStr); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img: img, price: priceNum, rating: parseFloat(rating) || 0, soldStr: sold, url: `https://www.aliexpress.com/item/${id}.html` }); } catch (e) {} } return out; }
