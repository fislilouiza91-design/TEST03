// popup/popup.js const state = { products: [], streamId: null, host: null, page: 1, sortMode: 'popular', // cheapest | popular | rating loading: false, hasMore: true, }; const $ = (s) => document.querySelector(s); const resultsEl = $('#results'); const loadMoreBtn = $('#loadMore'); const statusEl = $('#status'); // ============================================================ // استدعاء الـ background // ============================================================ async function fetchFromBackground({ page = 1, streamId = null, sortType = 'default', sortOrder = 'default' } = {}) { return new Promise((resolve) => { chrome.runtime.sendMessage( { action: 'FETCH_SIMILAR', page, streamId, sortType, sortOrder }, resolve ); }); } // ============================================================ // تحويل sortMode إلى قيم API // ============================================================ function sortModeToAPI(mode) { switch (mode) { case 'cheapest': return { sortType: 'price_asc', sortOrder: 'asc' }; case 'rating': return { sortType: 'rating_desc', sortOrder: 'desc' }; case 'popular': default: return { sortType: 'total_tranpro_desc', sortOrder: 'desc' }; } } // ============================================================ // ترتيب client-side (احتياطي إذا تجاهل API الترتيب) // ============================================================ function sortProductsClientSide(products, mode) { const arr = [...products]; if (mode === 'cheapest') { arr.sort((a, b) => getPrice(a) - getPrice(b)); } else if (mode === 'rating') { arr.sort((a, b) => getRating(b) - getRating(a)); } else { arr.sort((a, b) => getTradeCount(b) - getTradeCount(a)); } return arr; } function getPrice(p) { return Number(p?.prices?.salePrice?.minPrice) || Number(p?.prices?.salePrice?.formattedPrice?.replace(/[^\d.]/g, '')) || 999999; } function getRating(p) { return Number(p?.evaluation?.starRating) || Number(p?.evaluation?.ratingAvg) || Number(p?.evaluation?.rating) || 0; } function getTradeCount(p) { return Number(p?.trade?.realTradeCount) || Number(p?.trade?.tradeCount) || 0; } // ============================================================ // عرض المنتجات // ============================================================ function render() { if (!state.products.length) { resultsEl.innerHTML = `

لا توجد منتجات مشابهة حالياً

`; return; } const sorted = sortProductsClientSide(state.products, state.sortMode); resultsEl.innerHTML = sorted.map((p) => { const rawImg = p?.image?.imgUrl || ''; const img = rawImg.startsWith('//') ? 'https:' + rawImg : rawImg; const price = p?.prices?.salePrice?.formattedPrice || '—'; const title = p?.title?.displayTitle || p?.title?.seoTitle || 'منتج'; const rating = getRating(p); const trades = getTradeCount(p); const url = `https://${state.host}/item/${p.productId}.html`; // ⚠️ استبدل هذا برابط العمولة الخاص بك const aff = `https://YOUR-AFFILIATE-LINK.com/redirect?to=${encodeURIComponent(url)}`; return `
${escapeHtml(title)}
${price}
⭐ ${rating ? rating.toFixed(1) : '—'} 🔥 ${formatNumber(trades)}
`; }).join(''); } function escapeHtml(s) { const d = document.createElement('div'); d.textContent = s; return d.innerHTML; } function formatNumber(n) { if (n >= 1000000) return (n / 1000000).toFixed(1) + 'M'; if (n >= 1000) return (n / 1000).toFixed(1) + 'K'; return String(n); } // ============================================================ // تحميل الصفحة الأولى (عند فتح الـ popup) // ============================================================ async function loadInitial() { state.loading = true; state.page = 1; state.products = []; state.streamId = null; state.hasMore = true; resultsEl.innerHTML = `

جاري البحث...

`; loadMoreBtn.disabled = true; statusEl.textContent = ''; const apiSort = sortModeToAPI(state.sortMode); const res = await fetchFromBackground({ page: 1, streamId: null, ...apiSort, }); state.loading = false; if (res?.error) { showError(res.error); return; } state.products = res.products || []; state.streamId = res.streamId || null; state.host = res.host || null; state.hasMore = state.products.length >= 20; if (!state.products.length) { resultsEl.innerHTML = `

لم نجد منتجات مشابهة لهذا المنتج

`; loadMoreBtn.disabled = true; return; } render(); loadMoreBtn.disabled = !state.hasMore; statusEl.textContent = `${state.products.length} منتج`; } // ============================================================ // تحميل المزيد // ============================================================ async function loadMore() { if (state.loading || !state.hasMore) return; state.loading = true; loadMoreBtn.disabled = true; loadMoreBtn.textContent = 'جاري التحميل...'; const apiSort = sortModeToAPI(state.sortMode); const nextPage = state.page + 1; const res = await fetchFromBackground({ page: nextPage, streamId: state.streamId, ...apiSort, }); state.loading = false; loadMoreBtn.textContent = 'تحميل المزيد'; if (res?.error) { showError(res.error); return; } const more = res.products || []; if (more.length) { state.products = [...state.products, ...more]; state.streamId = res.streamId || state.streamId; state.page = nextPage; state.hasMore = more.length >= 20; render(); statusEl.textContent = `${state.products.length} منتج`; } else { state.hasMore = false; } loadMoreBtn.disabled = !state.hasMore; } // ============================================================ // إظهار خطأ // ============================================================ function showError(msg) { const map = { 'NOT_PRODUCT_PAGE': 'افتح صفحة منتج في AliExpress أولاً', 'NO_TOKEN': 'لم يتم العثور على توكن. أعد تحميل صفحة AliExpress.', 'EMPTY_RESULT': 'الاستجابة فارغة. حاول مرة أخرى.', }; const friendly = map[msg] || (msg?.startsWith('API_') ? 'تم رفض الطلب من AliExpress. حاول لاحقاً.' : msg); resultsEl.innerHTML = `

⚠️ ${escapeHtml(friendly || 'حدث خطأ غير متوقع')}

`; loadMoreBtn.disabled = true; } // ============================================================ // أحداث الأزرار // ============================================================ document.querySelectorAll('.sort-btn').forEach(btn => { btn.addEventListener('click', () => { document.querySelectorAll('.sort-btn').forEach(b => b.classList.remove('active')); btn.classList.add('active'); state.sortMode = btn.dataset.sort; loadInitial(); }); }); loadMoreBtn.addEventListener('click', loadMore); // الوضع الليلي const themeToggle = $('#themeToggle'); chrome.storage.local.get(['darkMode'], (r) => { if (r.darkMode) { document.body.classList.add('dark'); themeToggle.textContent = '☀️'; } }); themeToggle.addEventListener('click', () => { document.body.classList.toggle('dark'); const isDark = document.body.classList.contains('dark'); themeToggle.textContent = isDark ? '☀️' : '🌙'; chrome.storage.local.set({ darkMode: isDark }); }); // ============================================================ // البداية // ============================================================ loadInitial();
