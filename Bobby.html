/* ============================================ Smart Shopper - Background Service Worker (Precise Semantic Search) ============================================ */ const WORKER_URL = 'https://ali-scraper.your-username.workers.dev'; const MAX_PAGES_TO_FETCH = 2; // أنت تبحث في صفحتين فقط // كلمات الحشو التي يجب تجاهلها const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'high', 'good', '2023', '2024', '2025', '2026', 'best', 'top', 'pcs', 'set', 'pack', 'piece', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'double', 'door', 'enclosed', 'large', 'angle', 'flip', 'cover', 'removable', 'easy', 'clean', 'waterproof', 'portable', 'multifunctional', 'design', 'filter', 'compatible', 'fits', 'fit', 'suitable', 'accessories', 'accessory' ]); // العلامات التجارية الشهيرة const KNOWN_BRANDS = [ 'apple', 'samsung', 'xiaomi', 'redmi', 'poco', 'huawei', 'honor', 'oppo', 'vivo', 'oneplus', 'realme', 'lenovo', 'hp', 'dell', 'asus', 'acer', 'msi', 'sony', 'lg', 'panasonic', 'philips', 'anker', 'baseus', 'ugreen', 'romoss', 'logitech', 'razer', 'corsair', 'nike', 'adidas', 'puma', 'dyson', 'ecovacs', 'roborock' ]; // الفئات الأساسية والكلمات المرادفة const CATEGORIES = { 'case': ['case', 'cover', 'coque', 'housse', 'чехол', 'حافظة', 'shell'], 'earbuds': ['earbuds', 'headphone', 'headset', 'airpods', 'casque', 'навушники'], 'watch': ['watch', 'smartwatch', 'montre', 'часы', 'strap', 'watchband'], 'charger': ['charger', 'charging', 'adapter', 'chargeur', 'зарядка'], 'cable': ['cable', 'cord', 'câble', 'кабель'], 'powerbank': ['powerbank', 'power bank', 'battery', 'batterie', 'аккумулятор'], 'mouse': ['mouse', 'souris', 'мышь'], 'keyboard': ['keyboard', 'clavier', 'клавиатура'], 'lamp': ['lamp', 'light', 'led', 'lampe', 'лампа', 'светильник'], 'vacuum': ['vacuum', 'sweeper', 'aspirateur', 'пилосос'], 'drone': ['drone', 'дрон'], 'camera': ['camera', 'cam', 'caméra', 'камера'], 'speaker': ['speaker', 'soundbar', 'haut-parleur', 'колонка'], 'tool': ['tool', 'screwdriver', 'wrench', 'plier', 'drill', 'multimeter', 'voltmeter'] }; // الكلمات الوصفية المهمة (Descriptors) const DESCRIPTORS = ['silicone', 'leather', 'clear', 'magnetic', 'wireless', 'bluetooth', 'waterproof', 'shockproof', 'metal', 'glass', 'nylon', 'cotton', 'plastic', 'stainless', 'steel']; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } console.log(`🎯 كلمة البحث الدقيقة: [${searchQuery}]`); const q = encodeURIComponent(searchQuery); fetch(`${WORKER_URL}?q=${q}&page=${page}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: 100 })); let hasMore = items.length >= 20; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } }); // ========================================== // 🔥 محرك البحث الدقيق (النسخة المطورة) // ========================================== function buildSmartSearchQuery(title) { if (!title) return ''; // 1. تنظيف العنوان const cleanTitle = title.replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ').replace(/\s+/g, ' ').trim(); const lowerTitle = cleanTitle.toLowerCase(); const rawWords = cleanTitle.split(' '); // 2. استخراج البراند (Brand) let brand = ''; for (let w of rawWords) { const cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && cleanW.length <= 10) { if (KNOWN_BRANDS.includes(cleanW.toLowerCase())) { brand = cleanW.toLowerCase(); break; } if (w === w.toUpperCase() && /^[A-Z]+$/.test(cleanW) && rawWords.indexOf(w) < 3) { brand = cleanW.toLowerCase(); break; } } } // 3. استخراج الموديل الكامل (Full Model) - تسلسل الكلمات التي تحتوي على أرقام let model = ''; let modelStartIndex = -1; for (let i = 0; i < rawWords.length; i++) { const w = rawWords[i]; const cleanW = w.replace(/[^a-zA-Z0-9]/g, ''); if (cleanW.length >= 2 && /[0-9]/.test(cleanW) && /[a-zA-Z]/.test(cleanW) || /^\d+$/.test(cleanW)) { modelStartIndex = i; break; } } if (modelStartIndex !== -1) { // نأخذ الكلمة التي تحتوي على الرقم وما يليها من كلمات وصفية (مثل "15 Pro Max") let modelParts = [rawWords[modelStartIndex].replace(/[^a-zA-Z0-9]/g, '')]; // إذا كان الرقم في كلمة منفصلة (مثل iPhone 15)، نأخذ الكلمات التالية if (/^\d+$/.test(rawWords[modelStartIndex].replace(/[^a-zA-Z0-9]/g, ''))) { let j = modelStartIndex + 1; while (j < rawWords.length && j <= modelStartIndex + 3) { const nextW = rawWords[j].replace(/[^a-zA-Z0-9]/g, ''); if (['pro', 'max', 'plus', 'ultra', 'mini', 'lite', 'air', 'note', 's24', 's23', 's22', 's21', 's20'].includes(nextW.toLowerCase())) { modelParts.push(nextW.toLowerCase()); j++; } else { break; } } } model = modelParts.join(' '); } // 4. استخراج الفئة (Category) let category = ''; for (const [catKey, keywords] of Object.entries(CATEGORIES)) { if (keywords.some(k => lowerTitle.includes(k))) { category = catKey; break; } } // 5. استخراج الوصف المميز (Descriptor) - كلمة واحدة فقط let descriptor = ''; for (let d of DESCRIPTORS) { if (lowerTitle.includes(d)) { descriptor = d; break; } } // 6. تجميع الكلمات المفتاحية let parts = []; if (brand) parts.push(brand); if (model && model !== brand) parts.push(model); if (descriptor) parts.push(descriptor); if (category) parts.push(category); // إذا لم نجد فئة معروفة، نأخذ أطول كلمتين متبقيتين if (!category) { const remainWords = rawWords .map(w => w.toLowerCase().replace(/[^a-z0-9]/g, '')) .filter(w => w.length > 2 && !STOP_WORDS.has(w) && w !== brand && !model.includes(w)) .sort((a, b) => b.length - a.length); parts.push(...remainWords.slice(0, 2)); } if (parts.length === 0) { return rawWords.slice(0, 3).join(' '); } // إزالة التكرار وإرجاع النتيجة النهائية return [...new Set(parts)].join(' '); }
