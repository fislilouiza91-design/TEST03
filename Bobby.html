const MAX_PAGES = 10; const DELAY_BETWEEN_PAGES = 1000; const CACHE_TTL = 12 * 60 * 60 * 1000; // 12 ساعة chrome.runtime.onInstalled.addListener(async () => { await clearProxy(); setupProxyAuth(); await refreshProxyPool(); }); chrome.runtime.onStartup.addListener(clearProxy); chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CLEAR_PROXY') { clearProxy().then(() => sendResponse({ ok: true })); return true; } if (msg.type !== 'SMART_SEARCH') return; const tabId = sender.tab && sender.tab.id; (async () => { try { const searchTitle = msg.title || ''; // 1. التحقق من الذاكرة const cached = await getCachedResults(searchTitle); // ⭐ سحب بروكسي واحد من الطابور لهذا المنتج let productProxy = await getNextProxyInQueue(); if (!productProxy) return sendResponse({ ok: true, items: [], done: true }); while (page <= MAX_PAGES) { try { // نطبق البروكسي قبل كل صفحة (لحماية تصفحك) await applyProxy(productProxy); newItems.forEach(it => seenIds.add(it.id)); allFetchedItems.push(...newItems); if (newItems.length > 0) { totalFetched += newItems.length; if (tabId) { if (!firstBatchSent) { sendResponse({ ok: true, items: newItems, searchUrl: baseUrl.replace('&_ext_fetch=1', ''), done: false }); firstBatchSent = true; } else { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_UPDATE', items: newItems }).catch(() => {}); } } } } page++; // الانتقال للصفحة التالية بنفس البروكسي await new Promise(r => setTimeout(r, DELAY_BETWEEN_PAGES)); } catch (err) { if (err.message === 'PROXY_FAILED') { console.warn(`[Search] ⚠️ البروكسي فشل في الصفحة ${page}، جاري سحب بروكسي جديد...`); // ⭐ البروكسي تالف: نحذفه من الطابور ونجلب الذي يليه markProxyAsDead(productProxy); productProxy = await getNextProxyInQueue(); if (!productProxy) break; // إذا نفدت البروكسيات // لا نزيد رقم الصفحة (page++) لكي نعيد المحاولة لنفس الصفحة بالبروكسي الجديد } else break; } finally { await clearProxy(); // تنظيف فوري } } if (allFetchedItems.length > 0) { await saveToCache(searchTitle, allFetchedItems); } if (tabId) chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_DONE', total: totalFetched }).catch(() => {}); if (!firstBatchSent) sendResponse({ ok: true, items: [], done: true }); } catch (e) { sendResponse({ ok: false, error: String(e) }); } })(); return true; }); async function fetchPage(baseUrl, page, timeoutMs) { try { const url = baseUrl + '&page=' + page; const res = await fetch(url, { credentials: 'omit', headers: { 'Accept-Language': 'en-US,en;q=0.9', 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8' }, signal: AbortSignal.timeout(timeoutMs) }); let items = parseItems(html); if (!items.length) items = fallbackIds(html); return items; } catch (e) { throw new Error('PROXY_FAILED'); } } /* دوال التخزين المؤقت (Cache) */ async function getCachedResults(title) { try { const cacheKey = 'cache_' + title.slice(0, 40).toLowerCase().replace(/\s+/g, ''); const result = await chrome.storage.local.get(cacheKey); const data = result[cacheKey]; if (data && data.timestamp && (Date.now() - data.timestamp < CACHE_TTL)) { return data.items; } return null; } catch (e) { return null; } } async function saveToCache(title, items) { try { const cacheKey = 'cache_' + title.slice(0, 40).toLowerCase().replace(/\s+/g, ''); await chrome.storage.local.set({ [cacheKey]: { timestamp: Date.now(), items: items } }); } catch (e) {} } /* دوال الاستخراج */ function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } return out; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function clean(s) { return String(s).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim();
