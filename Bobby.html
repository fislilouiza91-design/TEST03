export default { async fetch(request, env, ctx) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders }); const url = new URL(request.url); if (url.pathname === '/search-db') { let title = url.searchParams.get('q') || ''; const pid = url.searchParams.get('pid') || ''; const page = parseInt(url.searchParams.get('page')) || 1; const existingIds = url.searchParams.get('existingIds') || ''; try { if (!env.TURSO_URL || !env.TURSO_TOKEN) throw new Error("Missing Turso config"); let englishTitle = ''; const isNonEnglish = /[^\x00-\x7F]/.test(title); // 1. جلب العنوان الإنجليزي الأصلي if (pid) { try { const dbTitle = await getDbTitle(pid, env); if (dbTitle) englishTitle = dbTitle; } catch (e) {} } if (!englishTitle && isNonEnglish) { englishTitle = await translateToEnglish(title); } if (!englishTitle) englishTitle = title; // 2. البحث في قاعدة البيانات const dbResult = await searchTursoDB(englishTitle, page, existingIds, env); // 🔥 3. الحل الجذري: إذا كانت قاعدة البيانات فارغة من هذا المنتج، نجلبه من علي إكسبرس مباشرة! String(obj.oldPrice) : '', discount: discount > 0 ? String(discount) : '', rating: obj.rating ? String(obj.rating) : '', sold: obj.sold ? const aliexpressUrl = baseUrl + '&page=' + page; // رابط الـ Worker القديم الخاص بسحب الصفحات const workerEndpoint = `https://smart-shopper.smailsuit11.workers.dev/fetch?url=${encodeURIComponent(aliexpressUrl)}`; // نفس دالة الاستخراج الخاصة بالإضافة const items = parseAliExpressHtml(html); // فلترة المنتجات المكررة const filteredItems = items.filter(it => !existingIds.includes(it.id)); return { ok: true, items: filteredItems, hasMore: filteredItems.length === 60 }; } catch (e) { return { ok: true, items: [], hasMore: false }; } } // دالة استخراج البيانات من HTML (نسخة Cloudflare) function parseAliExpressHtml(html) { const extractJsonArray = (html, marker) => { const i = html.indexOf(marker);
