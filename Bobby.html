/* ============================================ Smart Shopper - Background v3.1 (Sticky Proxy) ============================================ */ import { refreshProxyPool, getNextProxy, applyProxyForAliExpress, setupProxyAuth, getStatus } from './proxy-manager.js'; const MAX_PAGES = 10; const DELAY_BETWEEN_PAGES = 2500; // زيادة التأخير لتقليل الحظر /* ============================================ التهيئة ============================================ */ chrome.runtime.onInstalled.addListener(async () => { setupProxyAuth(); await refreshProxyPool(); console.log('[Init] الإضافة جاهزة'); }); chrome.runtime.onStartup.addListener(() => { setupProxyAuth(); }); /* ============================================ مستمع الرسائل الرئيسي ============================================ */ chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'PROXY_STATUS') { sendResponse(getStatus()); return true; } if (msg.type !== 'SMART_SEARCH') return; const tabId = sender.tab && sender.tab.id; (async () => { try { // 1. الحصول على بروكسي وتطبيقه (مرة واحدة لكل منتج) console.log('[Search] 🔄 جلب بروكسي جديد...'); let proxy = await getNextProxy(); await applyProxyForAliExpress(proxy); await new Promise(r => setTimeout(r, 2000)); const q = encodeURIComponent((msg.title || '').slice(0, 90)); const baseUrl = 'https://www.aliexpress.com/wholesale?SearchText=' + q + '&catId=0'; const seenIds = new Set(); let firstBatchSent = false; let totalFetched = 0; let page = 1; let retries = 0; const MAX_RETRIES = 5; // 2. حلقة جلب الصفحات while (page <= MAX_PAGES && retries < MAX_RETRIES) { try { console.log(`[Fetch] جلب الصفحة ${page} باستخدام ${proxy.host}:${proxy.port}...`); const items = await fetchPage(baseUrl, page); if (items && items.length > 0) { const newItems = []; for (const it of items) { if (!it || !it.id || seenIds.has(it.id)) continue; seenIds.add(it.id); newItems.push(it); } if (newItems.length > 0) { totalFetched += newItems.length; if (tabId) { if (!firstBatchSent) { sendResponse({ ok: true, items: newItems, searchUrl: baseUrl, done: false, count: newItems.length }); firstBatchSent = true; } else { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_UPDATE', items: newItems, searchUrl: baseUrl }).catch(() => {}); } } } } else { // صفحة فارغة (نهاية النتائج) if (page === 1) break; // إذا كانت الأولى فارغة، ننهي البحث } page++; retries = 0; // تصفير المحاولات لأن البروكسي أثبت نجاحه await new Promise(r => setTimeout(r, DELAY_BETWEEN_PAGES)); } catch (err) { if (err.message === 'PROXY_FAILED') { console.warn(`[Search] ⚠️ البروكسي توقف أو تم حظره، جاري التبديل (محاولة ${retries + 1}/${MAX_RETRIES})...`); retries++; proxy = await getNextProxy(); await applyProxyForAliExpress(proxy); await new Promise(r => setTimeout(r, 2000)); } else { console.error('[Search] خطأ غير متوقع:', err); break; } } } // 3. إنهاء البحث if (tabId) { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_DONE', total: totalFetched }).catch(() => {}); } if (!firstBatchSent) { sendResponse({ ok: true, items: [], searchUrl: baseUrl, done: true }); } } catch (e) { console.error('[Search] فشل كلي:', e); sendResponse({ ok: false, error: String(e) }); } })(); return true; }); /* ============================================ دالة جلب صفحة (ترمي خطأ عند الحظر لتبديل البروكسي) ============================================ */ async function fetchPage(baseUrl, page) { try { const url = baseUrl + '&page=' + page; const res = await fetch(url, { credentials: 'include', headers: { 'Accept-Language': 'en-US,en;q=0.9', 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8' }, signal: AbortSignal.timeout(15000) }); if (!res.ok) { if (res.status === 403 || res.status === 429) { console.warn(`[Fetch] صفحة ${page} → محظور (HTTP ${res.status})`); throw new Error('PROXY_FAILED'); // ⭐ يتبدل البروكسي فوراً } return []; } const html = await res.text(); if (html.includes('captcha') || html.includes('punish') || html.includes('slide to verify')) { console.warn(`[Fetch] صفحة ${page} → كابتشا`); throw new Error('PROXY_FAILED'); // ⭐ يتبدل البروكسي فوراً } let items = parseItems(html); if (!items.length) items = fallbackIds(html); return items; } catch (e) { if (e.message === 'PROXY_FAILED') throw e; console.warn(`[Fetch] صفحة ${page} فشلت:`, e.message); throw new Error('PROXY_FAILED'); } } /* ============================================ دوال استخراج البيانات ============================================ */ function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const title = clean(o.title?.displayTitle || o.title || ''); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const price = sale.formatedActivityPrice || sale.formatedPrice || p.formatedPrice || ''; const discount = o.discount || sale.discount || ''; const rating = (o.evaluation && o.evaluation.starRating) || o.averageStarRate || ''; const sold = o.tradeDesc || o.sold || ''; out.push({ id: String(id), title: title || 'Produit similaire', img, price, discount, rating, sold, link: 'https://www.aliexpress.com/item/' + id + '.html' }); } catch {} } return out; } function fallbackIds(html) { const re = /"productId"\s*:\s*"(\d{6,})"/g; const seen = new Set(); const out = []; let m; while ((m = re.exec(html))) { if (seen.has(m[1])) continue; seen.add(m[1]); out.push({ id: m[1], title: 'Produit', img: '', price: '', discount: '', rating: '', sold: '', link: 'https://www.aliexpress.com/item/' + m[1] + '.html' }); } return out; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function clean(s) { return String(s).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); }
