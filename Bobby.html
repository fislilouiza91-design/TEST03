export default { async fetch(request, env, ctx) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } const url = new URL(request.url); if (url.pathname === '/search-db') { let title = url.searchParams.get('q') || ''; const pid = url.searchParams.get('pid') || ''; const page = parseInt(url.searchParams.get('page')) || 1; const existingIds = url.searchParams.get('existingIds') || ''; try { if (!env.TURSO_URL || !env.TURSO_TOKEN) { throw new Error("Missing Turso configuration in Cloudflare Variables"); } let englishTitle = ''; // التحقق مما إذا كان العنوان يحتوي على حروف غير إنجليزية (عربية، فرنسية، إلخ) const isNonEnglish = /[^\x00-\x7F]/.test(title); // ⚡ 1. البحث في قاعدة البيانات فوراً (إذا كان المنتج موجوداً نأخذ عنوانه الإنجليزي) if (isNonEnglish && pid) { try { const dbTitle = await getDbTitle(pid, env); if (dbTitle) { englishTitle = dbTitle; console.log("✅ Found in DB:", englishTitle); } } catch (e) { console.warn("DB Lookup failed:", e.message); } } // ⚡ 2. إذا لم نجده في القاعدة، نترجمه فوراً باستخدام جوجل if (!englishTitle && isNonEnglish) { englishTitle = await translateToEnglish(title); console.log("🌐 Translated:", englishTitle); } // إذا كان العنوان إنجليزياً بالفعل، نستخدمه كما هو if (!englishTitle) { englishTitle = title; } // ⚡ 3. البحث السريع في القاعدة const result = await searchTursoDB(englishTitle, page, existingIds, env); return new Response(JSON.stringify(result), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } catch (e) { return new Response(JSON.stringify({ ok: false, error: e.message }), { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } } return new Response('Smart-Turso API is running.', { status: 200, headers: corsHeaders }); } } // ═══════════════════════════════════════════ // دالة جلب العنوان من قاعدة البيانات (سريعة جداً بفضل الفهرس) // ═══════════════════════════════════════════ async function getDbTitle(pid, env) { const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const res = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [ { type: 'execute', stmt: { sql: 'SELECT title FROM products WHERE product_id = ? LIMIT 1', args: [{ type: 'text', value: pid }] } }, { type: 'close' } ] }) }); const data = await res.json(); if (data.results[0].type === 'error') return null; const rows = data.results[0].response.result.rows; return rows.length > 0 ? rows[0][0].value : null; } // ═══════════════════════════════════════════ // دالة الترجمة السريعة (خطة بديلة) // ═══════════════════════════════════════════ async function translateToEnglish(text) { if (!text) return text; try { const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=en&dt=t&q=${encodeURIComponent(text)}`; const res = await fetch(url); if (res.ok) { const data = await res.json(); if (data && data[0] && data[0][0]) { return data[0].map(item => item[0]).join(''); } } } catch (e) {} return text; } // ═══════════════════════════════════════════ // دالة البحث (محسنة للسرعة القصوى) // ═══════════════════════════════════════════ async function searchTursoDB(title, page, existingIds, env) { const stopWords = new Set([ 'for', 'the', 'and', 'with', 'from', 'new', 'hot', 'best', 'free', 'high', 'pro', 'sale', 'sales', 'shipping', 'pcs', 'piece', 'set', 'plus', 'original', 'official', 'good', 'quality', 'smart', 'portable', 'wireless', 'digital', 'type', 'model', 'color', 'size', 'men', 'women', 'kids', 'baby', 'home', 'car', 'phone', 'case', 'made', 'material', 'style', 'fashion', 'design', 'item', 'top', 'super', 'ultra', 'mini', 'max', 'premium', 'fast', 'pack', 'box', 'quick', 'release', 'replacement', 'gift', 'drop', 'ship', 'factory', 'direct', 'wholesale', 'dropshipping', 'performance', 'sports', 'racing', 'gas', 'cc', '4x', '10x', '100', 'mah', 'large', 'capacity', 'charging', 'charger', 'external', 'device', 'storage', 'capsule', 'cable', 'adapter', 'cover', 'protector', 'glass', 'iphone', 'samsung', 'xiaomi', 'huawei', 'redmi', 'poco', 'oppo', 'vivo', 'realme' ]); const allWords = title.toLowerCase().replace(/[^\w\s]/g, ' ').split(/\s+/); let meaningfulWords = allWords.filter(w => w.length > 2 && !stopWords.has(w) && !/^\d+$/.test(w)); if (meaningfulWords.length === 0) { meaningfulWords = allWords.filter(w => w.length > 1).slice(0, 2); } if (meaningfulWords.length === 0) return { ok: true, items: [], hasMore: false }; // نأخذ أطول كلمة (لأنها عادة اسم المنتج الأساسي) لتسريع البحث meaningfulWords.sort((a, b) => b.length - a.length); let keyword = meaningfulWords[0]; if (keyword.endsWith('s') && keyword.length > 4) { keyword = keyword.slice(0, -1); // إزالة الجمع (s) } // ⚡ استخدام LIKE 'keyword%' (البحث بالبادئة) لتسريع البحث بفضل الفهرس const conditions = `p.title LIKE ?`; const queryArgs = [{ type: 'text', value: `${keyword}%` }]; let excludeCondition = ''; if (existingIds && existingIds.trim() !== '') { const ids = existingIds.split(','); excludeCondition = `AND p.product_id NOT IN (${ids.map(() => '?').join(',')})`; ids.forEach(id => queryArgs.push({ type: 'text', value: id })); } const LIMIT = 60; const OFFSET = (page - 1) * LIMIT; // ⚡ الترتيب باستخدام rowid (أسرع بكثير من last_updated_at) const sql = ` SELECT p.product_id as id, p.title, COALESCE(NULLIF(p.image_url, ''), pi.image_url) as img, p.current_price_usd as price, p.original_price_usd as oldPrice, p.rating, p.orders_count as sold FROM products p LEFT JOIN product_images pi ON p.product_id = pi.product_id AND pi.is_main = 1 WHERE (${conditions}) ${excludeCondition} ORDER BY p.rowid DESC LIMIT ${LIMIT} OFFSET ${OFFSET} `; const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const tursoRes = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: sql, args: queryArgs } }, { type: 'close' }] }) }); if (!tursoRes.ok) throw new Error(`Turso HTTP Error: ${tursoRes.status}`); const data = await tursoRes.json(); const result = data.results[0]; if (result.type === 'error') throw new Error(result.error.message); const cols = result.response.result.cols.map(c => c.name); const rows = result.response.result.rows; const items = rows.map(row => { const obj = {}; cols.forEach((col, i) => { obj[col] = row[i].value; }); const discount = obj.oldPrice && obj.price ? Math.round(((obj.oldPrice - obj.price) / obj.oldPrice) * 100) : 0; return { id: obj.id, title: obj.title, img: obj.img || '', price: obj.price ? String(obj.price) : '', oldPrice: obj.oldPrice ? String(obj.oldPrice) : '', discount: discount > 0 ? String(discount) : '', rating: obj.rating ? String(obj.rating) : '', sold: obj.sold ? String(obj.sold) : '', link: `https://www.aliexpress.com/item/${obj.id}.html` }; }); return { ok: true, items: items, hasMore: items.length === LIMIT }; }
