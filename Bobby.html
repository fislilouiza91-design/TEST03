// src/category-fetcher.js import fetch from 'node-fetch'; import crypto from 'crypto'; import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const SCRAPINGANT_KEY = process.env.SCRAPINGANT_API_KEY; const SCRAPERAPI_KEY = process.env.SCRAPERAPI_API_KEY; const KEYWORDS_PER_RUN = 8; // استخراج hash من رابط الصورة function generateImageHash(imageUrl) { if (!imageUrl) return null; const m = imageUrl.match(/\/([^/]+\.(jpg|jpeg|png|webp))/i); if (m) { return crypto.createHash('md5').update(m[1]).digest('hex').substring(0, 16); } return null; } // استخراج كلمات مفتاحية من العنوان function extractTokens(title) { if (!title) return ''; const stop = new Set(['for','the','and','with','from','new','hot','best','free','high','pro','sale','shipping','pcs','piece','set']); return title.toLowerCase() .replace(/[^\w\s-]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !stop.has(w) && !/^\d+$/.test(w)) .slice(0, 10) .join(' '); } // تصنيف المنتج function detectCategory(title) { if (!title) return 'general'; const t = title.toLowerCase(); const cats = { 'electronics': ['earphone','headphone','speaker','charger','cable','usb','camera','drone','smart','watch'], 'fashion-women': ['women','dress','skirt','blouse','heels','handbag'], 'fashion-men': ['men','shirt','jeans','sneakers','wallet','tie'], 'home-kitchen': ['kitchen','mug','knife','cookware','storage','lamp'], 'beauty': ['makeup','skin','hair','perfume','lipstick','nail'], 'sports': ['sport','yoga','gym','bike','fitness','running'], 'toys-kids': ['kids','toy','baby','lego','puzzle','doll'], 'automotive': ['car','motorcycle','auto','tire','engine'], 'pets': ['dog','cat','pet','fish','bird'], 'tools': ['tool','drill','wrench','hammer','screw'], }; for (const [cat, keywords] of Object.entries(cats)) { if (keywords.some(k => t.includes(k))) return cat; } return 'general'; } // جلب من ScrapingAnt (المصدر الأساسي) async function fetchFromScrapingAnt(keyword) { const url = `https://www.aliexpress.com/w/wholesale-${encodeURIComponent(keyword)}.html?SortType=total_tranpro_desc`; const params = new URLSearchParams({ 'x-api-key': SCRAPINGANT_KEY, 'url': url, 'browser': 'false', 'proxy_country': 'US', }); const res = await fetch(`https://api.scrapingant.com/v2/general?${params}`, { timeout: 30000 }); if (!res.ok) { const err = await res.text(); throw new Error(`ScrapingAnt ${res.status}: ${err.substring(0, 100)}`); } return await res.text(); } // جلب من ScraperAPI (الاحتياطي) async function fetchFromScraperAPI(keyword) { const url = `https://www.aliexpress.com/w/wholesale-${encodeURIComponent(keyword)}.html?SortType=total_tranpro_desc`; const params = new URLSearchParams({ 'api_key': SCRAPERAPI_KEY, 'url': url, 'render': 'false', }); const res = await fetch(`https://api.scraperapi.com?${params}`, { timeout: 30000 }); if (!res.ok) { const err = await res.text(); throw new Error(`ScraperAPI ${res.status}: ${err.substring(0, 100)}`); } return await res.text(); } // جلب مع Fallback async function fetchWithFallback(keyword) { try { console.log(` 🔵 ScrapingAnt...`); const html = await fetchFromScrapingAnt(keyword); return { html, source: 'scrapingant' }; } catch (e) { console.log(` ⚠️ فشل: ${e.message}`); console.log(` 🟠 ScraperAPI...`); const html = await fetchFromScraperAPI(keyword); return { html, source: 'scraperapi' }; } } // استخراج المنتجات من HTML function parseProducts(html) { const products = []; const seen = new Set(); // استخراج product_id من روابط المنتجات const itemRegex = /\/item\/(\d+)\.html/g; const matches = [...html.matchAll(itemRegex)]; // استخراج العناوين const titleRegex = /"title"\s*:\s*"([^"]{10,200})"/g; const titles = [...html.matchAll(titleRegex)].map(m => m[1]); // استخراج الصور const imgRegex = /"(https:\/\/ae[^"]+\.(jpg|jpeg|png|webp))"/g; const images = [...html.matchAll(imgRegex)].map(m => m[1]); // استخراج الأسعار const priceRegex = /"salePrice"\s*:\s*\{[^}]*"formattedPrice"\s*:\s*"([^"]+)"/g; const prices = [...html.matchAll(priceRegex)].map(m => parseFloat(m[1].replace(/[^\d.]/g, '')) || 0); let idx = 0; for (const match of matches) { const pid = match[1]; if (seen.has(pid)) continue; seen.add(pid); products.push({ product_id: pid, title: titles[idx] || '', current_price: prices[idx] || 0, original_price: 0, image_url: images[idx] || '', orders_count: 0, rating: 0, seller_name: '', }); idx++; if (products.length >= 60) break; } return products; } // حفظ في Turso async function saveToTurso(keyword, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { if (!p.product_id || !p.title) continue; try { const imageHash = generateImageHash(p.image_url); const tokens = extractTokens(p.title); const category = detectCategory(p.title); await db.execute({ sql: `INSERT INTO products_current (product_id, keyword, language, title, current_price, original_price, orders_count, rating, image_url, seller_name, image_hash, title_tokens, category, first_seen_at, last_updated_at, update_count) VALUES (?, ?, 'en', ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1) ON CONFLICT (product_id, keyword, language) DO UPDATE SET current_price = excluded.current_price, last_updated_at = excluded.last_updated_at, update_count = update_count + 1`, args: [ p.product_id, keyword, p.title, p.current_price, p.original_price, p.orders_count, p.rating, p.image_url, p.seller_name, imageHash, tokens, category, now, now ], }); saved++; } catch (e) { // تجاهل الأخطاء الفردية } } return saved; } // اختيار الكلمات التالية async function pickNextKeywords(count) { const res = await db.execute({ sql: `SELECT keyword, category FROM keyword_rotation WHERE enabled = 1 ORDER BY CASE WHEN last_run_at = 0 THEN 0 ELSE 1 END, priority ASC, run_count ASC, last_run_at ASC LIMIT ?`, args: [count], }); return res.rows; } async function markProcessed(keyword, success) { await db.execute({ sql: `UPDATE keyword_rotation SET last_run_at = ?, run_count = run_count + 1, success_count = success_count + ?, fail_count = fail_count + ? WHERE keyword = ?`, args: [Date.now(), success ? 1 : 0, success ? 0 : 1, keyword], }); } async function main() { const start = Date.now(); console.log('🚀 بدء الجلب (ScrapingAnt + ScraperAPI)...\n'); const keywords = await pickNextKeywords(KEYWORDS_PER_RUN); console.log(`📋 سيتم معالجة ${keywords.length} كلمة\n`); let totalSaved = 0; let successCount = 0; let failCount = 0; let sa = 0, spa = 0; for (const row of keywords) { console.log(`🔍 "${row.keyword}" [${row.category}]`); try { const { html, source } = await fetchWithFallback(row.keyword); if (source === 'scrapingant') sa++; else spa++; const products = parseProducts(html); const saved = await saveToTurso(row.keyword, products); console.log(` ✅ ${saved} منتج (${source})`); totalSaved += saved; successCount++; await markProcessed(row.keyword, true); await new Promise(r => setTimeout(r, 3000)); } catch (error) { console.log(` ❌ فشل: ${error.message}`); failCount++; await markProcessed(row.keyword, false); } } const mins = ((Date.now() - start) / 60000).toFixed(1); console.log(`\n📊 تقرير الدورة (${mins} دقيقة):`); console.log(` ✅ نجح: ${successCount}/${keywords.length}`); console.log(` ❌ فشل: ${failCount}`); console.log(` 📦 مخزّن: ${totalSaved}`); console.log(` 🔵 ScrapingAnt: ${sa} طلب`); console.log(` 🟠 ScraperAPI: ${spa} طلب`); if (failCount > successCount) process.exit(1); } main().catch(err => { console.error('💥 خطأ قاتل:', err); process.exit(1); });
