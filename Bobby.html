export default { async fetch(request, env, ctx) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Cache-Control": "no-store" }; } } async function getDbTitle(pid, env) { const httpUrl = env.TURSO_URL.replace(/^libsql:\/\//, 'https://'); const res = await fetch(`${httpUrl}/v2/pipeline`, { method: 'POST', headers: { 'Authorization': `Bearer ${env.TURSO_TOKEN}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ requests: [{ type: 'execute', stmt: { sql: 'SELECT title FROM products WHERE product_id = ? return rows.length > 0 ? rows[0][0].value : null; } async function searchTursoDB(title, page, existingIds, env) { const stopWords = new Set([ 'for', 'the', 'and', 'with', 'from', 'new', 'hot', 'best', 'free', 'high', 'pro', 'sale', 'sales', 'shipping', 'pcs', 'piece', 'set', 'plus', 'original', 'official', 'good', 'quality', 'smart', 'portable', 'wireless', 'digital', 'type', 'model', 'color', 'size', 'men', 'women', 'kids', 'baby', 'home', 'car', 'phone', 'case', 'made', 'material', 'style', 'fashion', 'design', 'item', 'top', 'super', 'ultra', 'mini', 'max', 'premium', 'fast', 'pack', 'box', 'quick', 'release', 'replacement', 'gift', 'drop', 'ship', 'factory', 'direct', 'wholesale', 'dropshipping', 'performance', 'sports', 'racing', 'gas', 'cc', '4x', '10x', '100', 'mah', 'large', 'capacity', 'charging', 'charger', 'external', 'device', 'storage', 'capsule', 'cable', 'adapter', 'cover', 'protector', 'glass', 'iphone', 'samsung', 'xiaomi', 'huawei', 'redmi', 'poco', 'oppo', 'vivo', 'realme' ]); String(obj.oldPrice) : '', discount: discount > 0 ? String(discount) : '', rating: obj.rating ? String(obj.rating) : '', sold: obj.sold ? String(obj.sold) : '', link: `https://www.aliexpress.com/item/${obj.id}.html` }; }); return { ok: true, items: items, hasMore: items.length === LIMIT }; } async function fetchLiveFromAliExpress(title) { try { const baseUrl = 'https://www.aliexpress.com/wholesale?SearchText=' + encodeURIComponent(title.slice(0, 90)) + '&catId=0'; const workerEndpoint = `https://smart-shopper.smailsuit11.workers.dev/fetch?url=${encodeURIComponent(baseUrl)}`; const res = await fetch(workerEndpoint); if (!res.ok) return []; const html = await res.text(); if (html.length < 5000) return []; return parseAliExpressHtml(html); } catch (e) { return []; } } function parseAliExpressHtml(html) { const extractJsonArray = (html, marker) => { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; }; const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { if (!o || typeof o !== 'object') continue; const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(rawTitle).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ?
