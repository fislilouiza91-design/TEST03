const express = require('express'); const cors = require('cors'); const { exec } = require('child_process'); const app = express(); app.use(cors()); const COUNTRIES = ['US', 'DE', 'GB', 'FR', 'NL', 'CA', 'JP', 'SG']; let currentCountryIndex = 0; // تأكد من مسار Psiphon على جهازك const PSIPHON_PATH = `"C:\\Program Files\\Psiphon3\\psiphon3.exe"`; app.get('/rotate', (req, res) => { currentCountryIndex = (currentCountryIndex + 1) % COUNTRIES.length; const newCountry = COUNTRIES[currentCountryIndex]; console.log(`\n🔄 جاري تغيير Psiphon إلى دولة: ${newCountry}`); exec('taskkill /F /IM psiphon3.exe', () => { const regCmd = `reg add "HKCU\\Software\\Psiphon3" /v EgressRegion /t REG_SZ /d ${newCountry} /f`; exec(regCmd, (err) => { if (err) { console.error('خطأ في تعديل السجل:', err); return res.status(500).send('Failed to change region'); } exec(`start "" ${PSIPHON_PATH}`); console.log(`✅ تم تشغيل Psiphon. الانتظار 12 ثانية للاتصال...`); setTimeout(() => { console.log(`🎉 Psiphon جاهز للعمل على ${newCountry}`); res.json({ ok: true, country: newCountry }); }, 12000); }); }); }); app.listen(3000, () => { console.log('🚀 Bridge Server يعمل على http://127.0.0.1:3000'); });
