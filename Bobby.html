// src/apify-fetcher.js - المرحلة 1: بناء category_map + منتجات أولية import fetch from 'node-fetch'; import crypto from 'crypto'; import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const APIFY_TOKEN = process.env.APIFY_TOKEN; const KEYWORDS_PER_RUN = 5; const MAX_ITEMS_PER_KEYWORD = 30; function generateImageHash(url) { if (!url) return null; const m = url.match(/\/([^/]+\.(jpg|jpeg|png|webp))/i); if (m) return crypto.createHash('md5').update(m[1]).digest('hex').substring(0, 16); return null; } function extractTokens(title) { if (!title) return ''; const stop = new Set(['for','the','and','with','from','new','hot','best','free','high','pro','sale','shipping','pcs','piece','set']); return title.toLowerCase() .replace(/[^\w\s-]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !stop.has(w) && !/^\d+$/.test(w)) .slice(0, 10) .join(' '); } async function runApifyActor(keyword) { console.log(` 🎭 Apify Actor...`); const actorId = 'apt_marble~aliexpress-search-scraper'; const runUrl = `https://api.apify.com/v2/acts/${actorId}/run-sync-get-dataset-items?token=${APIFY_TOKEN}`; const response = await fetch(runUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ keyword: keyword, country: 'US', currency: 'USD', sortOrder: 'best_match', maxItems: MAX_ITEMS_PER_KEYWORD, }), timeout: 300000, }); if (!response.ok) { const err = await response.text(); throw new Error(`Apify ${response.status}: ${err.substring(0, 200)}`); } return await response.json(); } // ⭐ استخراج الفئات من المسار الهرمي async function extractCategoriesFromProducts(products) { const uniqueCategories = new Map(); for (const p of products) { if (!p.category_path || !Array.isArray(p.category_path)) continue; const path = p.category_path; // خزّن كل مستوى في المسار for (let i = 0; i < path.length; i++) { const catId = String(path[i]); if (!uniqueCategories.has(catId)) { uniqueCategories.set(catId, { category_id: catId, level: i + 1, parent_id: i > 0 ? String(path[i - 1]) : null, }); } } } // احفظ في category_map let inserted = 0; for (const [catId, cat] of uniqueCategories) { const result = await db.execute({ sql: `INSERT INTO category_map (category_id, name, parent_id, level, last_scraped_at) VALUES (?, ?, ?, ?, NULL) ON CONFLICT (category_id) DO NOTHING`, args: [cat.category_id, '', cat.parent_id, cat.level], }); inserted += result.rowsAffected || 0; } return inserted; } async function saveToTurso(keyword, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { if (!p.product_id || !p.title) continue; if (p.title.length < 15) continue; try { const imageHash = generateImageHash(p.image_url); const tokens = extractTokens(p.title); // التصنيف من categoryPath let category = 'general'; if (p.category_path && p.category_path.length > 0) { const rootId = String(p.category_path[0]); const rootMap = { '44': 'electronics', '3': 'home-kitchen', '100001540': 'fashion-women', '100001541': 'fashion-men', '66': 'beauty', '18': 'sports', '26': 'toys-kids', '34': 'automotive', '36': 'pets', '1420': 'tools', '509': 'watches', '1511': 'jewelry', '200000348': 'phones', '7': 'computer', }; category = rootMap[rootId] || 'general'; } const trustScore = (p.rating >= 4.8 ? 40 : p.rating >= 4.5 ? 30 : p.rating >= 4.0 ? 20 : 10) + (p.orders_count >= 10000 ? 30 : p.orders_count >= 1000 ? 20 : p.orders_count >= 100 ? 10 : 0) + (p.is_free_shipping ? 10 : 0); const valueScore = (p.rating / 5.0) * 35 + Math.min(p.orders_count / 5000, 1.0) * 30 + Math.max(0, 1 - (p.current_price / 100.0)) * 20; await db.execute({ sql: `INSERT INTO products_current (product_id, keyword, language, title, current_price, original_price, orders_count, rating, image_url, seller_name, image_hash, title_tokens, category, is_free_shipping, trust_score, value_score, total_score, first_seen_at, last_updated_at, update_count) VALUES (?, ?, 'en', ?, ?, ?, ?, ?, ?, '', ?, ?, ?, ?, ?, ?, ?, ?, ?, 1) ON CONFLICT (product_id, keyword, language) DO UPDATE SET current_price = excluded.current_price, orders_count = excluded.orders_count, rating = excluded.rating, last_updated_at = excluded.last_updated_at, update_count = update_count + 1`, args: [ p.product_id, keyword, p.title, p.current_price, p.original_price, p.orders_count, p.rating, p.image_url, imageHash, tokens, category, p.is_free_shipping ? 1 : 0, trustScore, valueScore, Math.round(trustScore * 0.5 + valueScore * 0.5), now, now ], }); saved++; } catch (e) {} } return saved; } async function pickNextKeywords(count) { const res = await db.execute({ sql: `SELECT keyword, category FROM keyword_rotation WHERE enabled = 1 ORDER BY CASE WHEN last_run_at = 0 THEN 0 ELSE 1 END, priority ASC, run_count ASC, last_run_at ASC LIMIT ?`, args: [count], }); return res.rows; } async function markProcessed(keyword, success) { await db.execute({ sql: `UPDATE keyword_rotation SET last_run_at = ?, run_count = run_count + 1, success_count = success_count + ?, fail_count = fail_count + ? WHERE keyword = ?`, args: [Date.now(), success ? 1 : 0, success ? 0 : 1, keyword], }); } async function main() { const start = Date.now(); console.log('🗺️ المرحلة 1: Apify - بناء خريطة الفئات + منتجات أولية...\n'); if (!APIFY_TOKEN) { console.error('❌ APIFY_TOKEN غير موجود!'); process.exit(1); } const keywords = await pickNextKeywords(KEYWORDS_PER_RUN); console.log(`📋 سيتم معالجة ${keywords.length} كلمة\n`); let totalSaved = 0; let totalCategories = 0; let successCount = 0; let failCount = 0; for (const row of keywords) { console.log(`\n🔍 "${row.keyword}" [${row.category}]`); try { const items = await runApifyActor(row.keyword); console.log(` 📦 استُلم ${items.length} منتج`); if (items.length === 0) { failCount++; await markProcessed(row.keyword, false); continue; } const products = items.map(item => ({ product_id: item.productId ? String(item.productId) : null, title: item.title || '', current_price: parseFloat(item.price || 0), original_price: parseFloat(item.originalPrice || 0), image_url: item.imageUrl || '', orders_count: parseInt(item.soldCount || 0), rating: parseFloat(item.rating || 0), is_free_shipping: item.hasFreeShippingOffer === true, category_path: item.categoryPath || [], })).filter(p => p.product_id && p.title && p.title.length > 15); // 🎯 استخراج الفئات من المسار الهرمي const catsAdded = await extractCategoriesFromProducts(products); totalCategories += catsAdded; console.log(` 🏷️ ${catsAdded} فئة جديدة في category_map`); const saved = await saveToTurso(row.keyword, products); console.log(` ✅ ${saved} منتج مخزّن`); totalSaved += saved; if (saved > 0) successCount++; else failCount++; await markProcessed(row.keyword, saved > 0); await new Promise(r => setTimeout(r, 2000)); } catch (error) { console.log(` ❌ فشل: ${error.message}`); failCount++; await markProcessed(row.keyword, false); } } // عرض إجمالي الفئات const catCount = await db.execute('SELECT COUNT(*) as c FROM category_map'); const mins = ((Date.now() - start) / 60000).toFixed(1); console.log(`\n📊 تقرير المرحلة 1 (${mins} دقيقة):`); console.log(` ✅ كلمات ناجحة: ${successCount}/${keywords.length}`); console.log(` 📦 منتجات مخزّنة: ${totalSaved}`); console.log(` 🏷️ فئات جديدة: ${totalCategories}`); console.log(` 📚 إجمالي الفئات في category_map: ${catCount.rows[0].c}`); console.log(` 💰 تكلفة Apify: $${(totalSaved * 0.0007).toFixed(4)}`); console.log(`\n✅ الآن شغّل ScrapingAnt Fetcher لملء المنتجات من هذه الفئات!`); if (failCount > successCount) process.exit(1); } main().catch(err => { console.error('💥 خطأ قاتل:', err); process.exit(1); });
