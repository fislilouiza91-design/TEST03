let attachedTabId = null; async function attachDebugger(tabId) { return new Promise((resolve, reject) => { chrome.debugger.attach({ tabId }, "1.3", () => { if (chrome.runtime.lastError) { // إذا كان مرفقاً مسبقاً، لا مشكلة if (chrome.runtime.lastError.message.includes("Another debugger is already attached")) { resolve(); } else { reject(new Error(chrome.runtime.lastError.message)); } } else { attachedTabId = tabId; resolve(); } }); }); } async function detachDebugger() { if (attachedTabId) { return new Promise((resolve) => { chrome.debugger.detach({ tabId: attachedTabId }, () => { attachedTabId = null; resolve(); }); }); } } async function fetchSimilarProducts() { const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true }); if (!tab?.url?.match(/\/item\/(\d+)\.htm/)) return { error: 'NOT_PRODUCT_PAGE' }; try { await attachDebugger(tab.id); await chrome.debugger.sendCommand({ tabId: tab.id }, "Network.enable"); // إعداد مستمع لالتقاط الاستجابات const capturedData = await new Promise((resolve, reject) => { const timeout = setTimeout(() => { reject(new Error('TIMEOUT: لم يتم العثور على بيانات المنتجات. حاول التمرير لأسفل الصفحة.')); }, 10000); // 10 ثوانٍ كحد أقصى const onResponse = async (source, method, params) => { if (method === "Network.responseReceived") { const url = params.response.url; // نبحث عن الاستجابات التي تحتوي على قوائم منتجات (توصيات أو بحث) if (url.includes('recommend') || url.includes('itemList') || url.includes('aeglodetailweb')) { try { const body = await chrome.debugger.sendCommand( { tabId: tab.id }, "Network.getResponseBody", { requestId: params.requestId } ); if (body && body.body) { let json = JSON.parse(body.body); // التحقق من وجود منتجات if (JSON.stringify(json).includes('productId')) { clearTimeout(timeout); chrome.debugger.onEvent.removeListener(onResponse); resolve(json); } } } catch (e) {} } } }; chrome.debugger.onEvent.addListener(onResponse); // تحفيز الموقع لتحميل البيانات عبر التمرير chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => { window.scrollBy(0, 1500); } }); }); await detachDebugger(); // استخراج المنتجات من البيانات الملتقطة const findProductsArray = (obj) => { if (!obj) return null; if (Array.isArray(obj)) { if (obj.length > 0 && (obj[0].productId || obj[0].itemId || obj[0].id)) return obj; for (let item of obj) { const found = findProductsArray(item); if (found) return found; } } else if (typeof obj === 'object') { for (let key in obj) { const found = findProductsArray(obj[key]); if (found) return found; } } return null; }; let rawProducts = findProductsArray(capturedData) || []; if (rawProducts.length === 0) { return { error: 'NO_PRODUCTS_FOUND_IN_NETWORK' }; } const products = rawProducts.map(p => { const priceNum = p?.prices?.salePrice?.minPrice || p?.prices?.salePrice?.formattedPrice || p?.price?.formattedPrice || p?.price?.minPrice || p?.price || '0'; return { productId: p.productId || p.id || p.itemId, title: { displayTitle: p.title?.displayTitle || p.title || p.titleText || 'Product' }, image: { imgUrl: p.image?.imgUrl || p.imageUrl || p.imgUrl || p.image || '' }, prices: { salePrice: { formattedPrice: typeof priceNum === 'string' && (priceNum.includes('₽') || priceNum.includes('руб')) ? priceNum : `${priceNum} ₽`, minPrice: p?.prices?.salePrice?.minPrice || p?.price?.minPrice || 0 } }, trade: { tradeCount: p.trade?.tradeCount || p.tradeCount || p.sold || 0 }, evaluation: { starRating: p.evaluation?.starRating || p.starRating || p.rating || 0 } }; }); return { products, streamId: null, ret: 'SUCCESS', host: location.host }; } catch (error) { await detachDebugger(); return { error: error.message }; } } chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (msg.action === 'FETCH_SIMILAR') { fetchSimilarProducts() .then(sendResponse) .catch(err => sendResponse({ error: err.message })); return true; } });
