/* Smart Shopper - background service worker (Fast Streaming Version) */ const MAX_PAGES = 8; // عدد الصفحات الإجمالي const PARALLEL_BATCH = 3; // جلب 3 صفحات في نفس الوقت const DELAY_BETWEEN_BATCHES = 400; // انتظار قصير بين كل مجموعة chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg || msg.type !== 'SMART_SEARCH') return; const tabId = sender.tab?.id; (async () => { try { const q = encodeURIComponent((msg.title || '').slice(0, 90)); const baseUrl = 'https://www.aliexpress.com/wholesale?SearchText=' + q + '&catId=0'; const seenIds = new Set(); let firstBatchSent = false; // جلب الصفحات على دفعات متوازية for (let batch = 0; batch < MAX_PAGES; batch += PARALLEL_BATCH) { const pages = []; for (let i = 0; i < PARALLEL_BATCH && (batch + i) < MAX_PAGES; i++) { pages.push(batch + i + 1); } // جلب كل صفحات الدفعة في نفس الوقت const results = await Promise.all(pages.map(p => fetchPage(baseUrl, p))); // تجميع النتائج الجديدة const newItems = []; let emptyPages = 0; for (const items of results) { if (!items || !items.length) { emptyPages++; continue; } for (const it of items) { if (!it || !it.id || seenIds.has(it.id)) continue; seenIds.add(it.id); newItems.push(it); } } // إرسال النتائج الجديدة فوراً للـ content script (Streaming) if (newItems.length > 0 && tabId && !firstBatchSent) { // الدفعة الأولى: نرد على الرسالة الأصلية sendResponse({ ok: true, items: newItems, searchUrl: baseUrl, done: false, count: newItems.length }); firstBatchSent = true; } else if (newItems.length > 0 && tabId) { // الدفعات التالية: نرسل تحديث try { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_UPDATE', items: newItems, searchUrl: baseUrl }); } catch (e) { /* tab closed */ } } // إذا كانت كل الصفحات فارغة، أوقف if (emptyPages === pages.length) break; // انتظار قصير قبل الدفعة التالية if (batch + PARALLEL_BATCH < MAX_PAGES) { await new Promise(r => setTimeout(r, DELAY_BETWEEN_BATCHES)); } } // إعلام النهاية if (tabId) { try { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_DONE' }); } catch (e) {} } // إذا لم يُرسل شيء بعد (حالة نادرة)، نرد فارغاً if (!firstBatchSent) { sendResponse({ ok: true, items: [], searchUrl: baseUrl, done: true }); } } catch (e) { sendResponse({ ok: false, error: String(e) }); } })(); return true; // async response }); /* ---------- جلب صفحة واحدة ---------- */ async function fetchPage(baseUrl, page) { try { const url = baseUrl + '&page=' + page; const res = await fetch(url, { headers: { 'Accept-Language': 'en-US,en;q=0.9', 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36' } }); if (!res.ok) return []; const html = await res.text(); let items = parseItems(html); if (!items.length) items = fallbackIds(html); return items; } catch (e) { return []; } } /* ---------- استخراج المنتجات ---------- */ function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { if (!o || typeof o !== 'object') continue; const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const title = clean(o.title || (o.item && o.item.title) || ''); let img = (o.image && (o.image.imgUrl || o.image.imageUrl)) || o.picUrl || o.imageUrl || ''; if (typeof img === 'object') img = img.imgUrl || img.imageUrl || ''; if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const orig = p.originalPrice || {}; const price = sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || ''; const oldPrice = orig.formatedPrice || orig.minPrice || ''; const discount = o.discount || sale.discount || ''; const store = (o.store && o.store.storeName) || o.storeName || 'AliExpress'; const rating = (o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''; const sold = o.tradeDesc || o.sold || o.soldCount || ''; out.push({ id: String(id), title, img, price, oldPrice, discount, store, rating, sold, link: 'https://www.aliexpress.com/item/' + id + '.html' }); } catch (e) {} } return out; } function fallbackIds(html) { const re = /"productId"\s*:\s*"(\d{6,})"/g; const seen = new Set(); const out = []; let m; while ((m = re.exec(html))) { if (seen.has(m[1])) continue; seen.add(m[1]); out.push({ id: m[1], title: 'Produit similaire', img: '', price: '', oldPrice: '', discount: '', store: 'AliExpress', rating: '', sold: '', link: 'https://www.aliexpress.com/item/' + m[1] + '.html' }); } return out; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function clean(s) { return String(s).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); }
