import { createClient } from '@libsql/client'; import fetch from 'node-fetch'; import dotenv from 'dotenv'; dotenv.config(); const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const ANT_API_KEY = process.env.SCRAPINGANT_API_KEY; const MARKET = { lang: 'en', code: 'US' }; // ⚙️ الإعدادات (يمكنك تعديلها) const BATCH_SIZE = 10; // عدد الكلمات التي يسحبها في الدورة الواحدة const PAGES_PER_KEYWORD = 2; // عدد الصفحات لكل كلمة (كل صفحة ~60 منتج) const MIN_RUN_TIME_MS = 25 * 60 * 1000; // السكريبت سيعمل لمدة 25 دقيقة كحد أقصى ثم يتوقف // ═══════════════════════════════════════════ // 1. إدارة الفئة النشطة والتدوير // ═══════════════════════════════════════════ async function getActiveCategory() { try { const stateRes = await db.execute(`SELECT value FROM system_state WHERE key = 'active_category'`); let activeCat = stateRes.rows.length > 0 ? String(stateRes.rows[0].value) : null; if (activeCat) { // هل هناك كلمات متبقية في هذه الفئة؟ const pending = await db.execute({ sql: `SELECT 1 FROM keywords_pipeline WHERE category = ? AND last_run_at = 0 AND enabled = 1 LIMIT 1`, args: [activeCat] }); if (pending.rows.length > 0) return activeCat; } // البحث عن فئة أخرى بها كلمات متبقية const catsRes = await db.execute(` SELECT category, MIN(created_at) as oldest_created FROM keywords_pipeline WHERE last_run_at = 0 AND enabled = 1 AND category IS NOT NULL AND category != '' GROUP BY category ORDER BY oldest_created ASC `); if (catsRes.rows.length > 0) { activeCat = String(catsRes.rows[0].category); console.log(`\n🔄 الانتقال إلى فئة جديدة: [${activeCat}]`); } else { // إذا انتهت جميع الكلمات في جميع الفئات.. نصفر الطابور لدورة جديدة console.log(`\n🚨 اكتملت جميع الكلمات في القاعدة! جاري تصفير الطابور لدورة جديدة...`); await db.execute(`UPDATE keywords_pipeline SET last_run_at = 0 WHERE enabled = 1`); const resetCatsRes = await db.execute(` SELECT category FROM keywords_pipeline WHERE enabled = 1 GROUP BY category ORDER BY MIN(created_at) ASC LIMIT 1 `); if (resetCatsRes.rows.length === 0) return null; activeCat = String(resetCatsRes.rows[0].category); } // حفظ الفئة النشطة await db.execute({ sql: `INSERT OR REPLACE INTO system_state (key, value, updated_at) VALUES ('active_category', ?, ?)`, args: [activeCat, Date.now()] }); return activeCat; } catch (e) { console.error('❌ خطأ في إدارة الفئة:', e.message); return null; } } async function getKeywordsForCategory(category, limit) { const res = await db.execute({ sql: `SELECT keyword FROM keywords_pipeline WHERE category = ? AND last_run_at = 0 AND enabled = 1 ORDER BY RANDOM() LIMIT ?`, args: [category, limit] }); return res.rows.map(r => String(r.keyword)); } async function markKeywordProcessed(keyword) { await db.execute({ sql: `UPDATE keywords_pipeline SET last_run_at = ?, run_count = run_count + 1 WHERE keyword = ?`, args: [Date.now(), keyword] }); } // ═══════════════════════════════════════════ // 2. دالة الجلب من ScrapingAnt (اقتصادية) // ═══════════════════════════════════════════ async function fetchSearchPage(keyword, page) { // رابط البحث في علي إكسبريس const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; // استخدام return_page_source لجلب الـ HTML بتكلفة نقطة واحدة تقريباً const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${ANT_API_KEY}&browser=true&return_page_source=true&proxy_type=datacenter&proxy_country=${MARKET.code}`; const response = await fetch(antUrl); if (!response.ok) throw new Error(`HTTP Error: ${response.status}`); return await response.text(); } // ═══════════════════════════════════════════ // 3. اللب الخاص بك لاستخراج البيانات // ═══════════════════════════════════════════ function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { if (!o || typeof o !== 'object') continue; const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const orig = p.originalPrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const oldPriceStr = extractText(orig.formatedPrice || orig.minPrice || ''); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img, priceStr, oldPriceStr, rating: parseFloat(rating) || 0, soldStr: sold }); } catch (e) {} } return out; } function extractPriceNum(s) { const m = String(s || '').replace(/,/g, '').match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractSoldCount(s) { if (!s) return 0; const str = String(s).toLowerCase(); if (str.includes('k')) return Math.floor(parseFloat(str.replace(/[^0-9.]/g, '')) * 1000); const m = str.match(/(\d+)/); return m ? parseInt(m[1]) : 0; } // ═══════════════════════════════════════════ // 4. الحفظ في قاعدة البيانات // ═══════════════════════════════════════════ async function saveToDatabase(products, keyword, language, category) { if (products.length === 0) return 0; const now = Date.now(); let savedCount = 0; for (const p of products) { try { const price = extractPriceNum(p.priceStr); const oldPrice = extractPriceNum(p.oldPriceStr); const orders = extractSoldCount(p.soldStr); const productUrl = `https://www.aliexpress.com/item/${p.id}.html`; // حفظ المنتج await db.execute({ sql: `INSERT OR REPLACE INTO products (product_id, title, product_url, main_image, category, created_at, updated_at) VALUES (?, ?, ?, ?, ?, COALESCE((SELECT created_at FROM products WHERE product_id = ?), ?), ?)`, args: [p.id, p.title, productUrl, p.img, category, p.id, now, now] }); // حفظ التفاصيل await db.execute({ sql: `INSERT OR REPLACE INTO product_details (product_id, current_price, original_price, rating, orders_count, last_updated) VALUES (?, ?, ?, ?, ?, ?)`, args: [p.id, price, oldPrice, p.rating, orders, now] }); // تاريخ السعر await db.execute({ sql: `INSERT INTO price_history (product_id, price, recorded_at) VALUES (?, ?, ?)`, args: [p.id, price, now] }); // ربط المنتج بالكلمة await db.execute({ sql: `INSERT OR IGNORE INTO keywords_mapping (product_id, keyword, language) VALUES (?, ?, ?)`, args: [p.id, keyword, language] }); savedCount++; } catch (dbError) { // تجاهل أخطاء الإدخال المكرر } } return savedCount; } // ═══════════════════════════════════════════ // 5. التشغيل الرئيسي (Main) // ═══════════════════════════════════════════ async function main() { const START_TIME = Date.now(); console.log('🚀 بدء محرك التدوير (سحب الكلمات من القاعدة وجلب المنتجات)...\n'); // حلقة تعمل حتى تنقضي مدة الـ 25 دقيقة while (Date.now() - START_TIME < MIN_RUN_TIME_MS) { const category = await getActiveCategory(); if (!category) { console.log('⚠️ لا توجد كلمات مفتاحية في قاعدة البيانات. تأكد من تشغيل سكريبت الاستيراد أولاً.'); break; } const keywords = await getKeywordsForCategory(category, BATCH_SIZE); if (!keywords || keywords.length === 0) { await new Promise(r => setTimeout(r, 2000)); continue; } console.log(`\n📋 سحب ${keywords.length} كلمة من فئة [${category}]`); for (const kw of keywords) { // التحقق من الوقت if (Date.now() - START_TIME >= MIN_RUN_TIME_MS) { console.log(`\n⏱️ انتهى وقت التشغيل المخصص. جاري الإغلاق بأمان...`); return; } console.log(`\n🔍 البحث عن: "${kw}"`); let keywordSaved = 0; for (let page = 1; page <= PAGES_PER_KEYWORD; page++) { try { const html = await fetchSearchPage(kw, page); if (html.length < 5000) break; // صفحة فارغة const products = parseItems(html); if (products.length === 0) break; const saved = await saveToDatabase(products, kw, MARKET.lang, category); console.log(` ✅ صفحة ${page}: تم استخراج ${products.length} منتج (جديد/محدث: ${saved})`); keywordSaved += saved; await new Promise(r => setTimeout(r, 3000)); // تأخير لتجنب الحظر } catch (error) { console.error(` ⚠️ خطأ في الصفحة ${page}:`, error.message); } } // وضع علامة على الكلمة بأنها تم جلبها بنجاح await markKeywordProcessed(kw); console.log(` 🏁 تم إنهاء الكلمة وتحديث حالتها في القاعدة.`); } } console.log(`\n🏁 انتهت العملية لهذه الدورة.`); } main().catch(console.error);
