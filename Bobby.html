/* ============================================ Smart Shopper - Background Service Worker (AI Filter is handled by Cloudflare Worker) ============================================ */ const WORKER_URL = 'https://competitors-api.smailsuit11.workers.dev'; const MAX_PAGES_TO_FETCH = 3; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { fetch(`${WORKER_URL}/api/competitors?productId=${msg.productId}`) .then(res => { if(res.ok) return res.json(); throw new Error('Not found'); }) .then(data => sendResponse({ ok: true, data: data.data })) .catch(() => sendResponse({ ok: false })); return true; } if (msg.type === 'SMART_SEARCH') { // نرسل العنوان الأصلي لكي يستخدمه الذكاء الاصطناعي في المقارنة const q = encodeURIComponent(msg.title); const originalTitle = encodeURIComponent(msg.title); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&page=1&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { sendResponse({ ok: true, items: data.items, hasMore: data.items.length > 0 }); if (msg.productId && data.items.length > 0) { saveToCloudflare(msg.productId, data.items); } } else { sendResponse({ ok: false, error: data.error }); } }).catch(err => sendResponse({ ok: false, error: err.message })); return true; } if (msg.type === 'SMART_LOAD_MORE') { const nextPage = Math.max(2, parseInt(msg.startPage) || 2); if (nextPage > MAX_PAGES_TO_FETCH) return sendResponse({ ok: true, items: [], hasMore: false }); const q = encodeURIComponent(msg.title); const originalTitle = encodeURIComponent(msg.title); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&page=${nextPage}&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { let existingIds = new Set(msg.existingIds); let unique = data.items.filter(i => !existingIds.has(i.id)); const hasMore = unique.length > 0 && nextPage < MAX_PAGES_TO_FETCH; sendResponse({ ok: true, items: unique, hasMore: hasMore }); if (msg.productId && unique.length > 0) { saveToCloudflare(msg.productId, unique); } } else { sendResponse({ ok: false, error: data.error, hasMore: false }); } }).catch(err => sendResponse({ ok: false, error: err.message, hasMore: false })); return true; } }); function saveToCloudflare(productId, items) { const payload = { base_product_id: productId, all_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })), relevant_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })) }; fetch(`${WORKER_URL}/api/competitors`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).then(res => res.json()).then(data => console.log('💾 Saved:', data)).catch(console.error); }
