function createCard(p) { let rawImg = p?.image?.imgUrl || ''; if (Array.isArray(rawImg)) rawImg = rawImg[0] || ''; const img = rawImg.startsWith('//') ? 'https:' + rawImg : rawImg; const price = p?.prices?.salePrice?.formattedPrice || '—'; const priceClass = getPriceClass(price); const title = p?.title?.displayTitle || 'منتج'; const rating = getRating(p); const directUrl = `https://${globalHost}/item/${p.productId}.html`; // 🔥 استخراج بيانات الشحن let isFreeShipping = false; let shippingText = ''; // 1. البحث في الحقول المباشرة if (p?.shipping) { if (p.shipping.shippingFee === 'Free' || p.shipping.shippingFee === 0 || p.shipping.freeShipping) { isFreeShipping = true; } else if (p.shipping.shippingFee) { shippingText = p.shipping.shippingFee; } } // 2. البحث في الوسوم (Selling Points) if (p?.sellingPoints && Array.isArray(p.sellingPoints)) { const freeTag = p.sellingPoints.find(sp => sp.toLowerCase().includes('free shipping') || sp.includes('شحن مجاني') || sp.includes('مجاني') ); if (freeTag) isFreeShipping = true; } // 3. البحث في وصف التجارة (TradeDesc) if (p?.trade?.tradeDesc && (p.trade.tradeDesc.includes('Free Shipping') || p.trade.tradeDesc.includes('مجاني'))) { isFreeShipping = true; } // بناء شكل الشحن let shippingHtml = ''; if (isFreeShipping) { shippingHtml = `
🚚 شحن مجاني
`; } else if (shippingText) { shippingHtml = `
🚚 ${shippingText}
`; } const a = document.createElement('a'); a.href = directUrl; a.target = '_blank'; a.className = 'product-card'; a.setAttribute('dir', 'auto'); a.innerHTML = `
${title}
${getStarsHTML(rating)}
${rating ? rating.toFixed(1) : '—'}
${price}
${shippingHtml}
`; return a; }
