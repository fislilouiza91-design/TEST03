// src/fetcher.js import fetch from 'node-fetch'; export class OmkarFetcher { constructor(apiKey, monitor) { this.apiKey = apiKey; this.monitor = monitor; } async fetch(keyword, countryCode = 'US') { const url = `https://aliexpress-scraper-api.omkar.cloud/aliexpress/v2/search?` + `query=${encodeURIComponent(keyword)}&` + `sort_by=most_orders&` + `country_code=${countryCode}&` + `page=1`; const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 20000); const startTime = Date.now(); try { const response = await fetch(url, { method: 'GET', headers: { 'API-Key': this.apiKey, 'Accept': 'application/json', }, signal: controller.signal, }); clearTimeout(timeoutId); const duration = Date.now() - startTime; if (!response.ok) { const errText = await response.text(); this.monitor.recordFailure(`HTTP ${response.status}: ${errText.substring(0, 80)}`); throw new Error(`Omkar ${response.status}`); } const data = await response.json(); this.monitor.recordSuccess(duration); return data.results || []; } catch (error) { clearTimeout(timeoutId); if (error.name === 'AbortError') { this.monitor.recordFailure('Timeout'); throw new Error('Timeout'); } throw error; } } }
