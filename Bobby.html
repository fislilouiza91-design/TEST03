/* ============================================ Smart Shopper - Background v2.0 مع التحكم الكامل في chrome.proxy ============================================ */ import { refreshProxyPool, getNextProxy, applyProxyForAliExpress, setupProxyAuth, getStatus } from './proxy-manager.js'; // ⚙️ الإعدادات const MAX_PAGES = 10; const PARALLEL_BATCH = 1; const DELAY_BETWEEN_BATCHES = 1500; /* ============================================ التهيئة عند بدء الإضافة ============================================ */ chrome.runtime.onInstalled.addListener(async () => { setupProxyAuth(); await refreshProxyPool(); console.log('[Init] الإضافة جاهزة'); }); chrome.runtime.onStartup.addListener(() => { setupProxyAuth(); }); /* ============================================ مستمع الرسائل الرئيسي ============================================ */ chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; // حالة البروكسي (للتشخيص) if (msg.type === 'PROXY_STATUS') { sendResponse(getStatus()); return true; } // تحديث قائمة البروكسي يدوياً if (msg.type === 'REFRESH_PROXIES') { refreshProxyPool().then(count => sendResponse({ ok: true, count })); return true; } // إلغاء البروكسي (تصفح مباشر) if (msg.type === 'CLEAR_PROXY') { applyProxyForAliExpress(null).then(() => sendResponse({ ok: true })); return true; } // البحث الرئيسي if (msg.type !== 'SMART_SEARCH') return; const tabId = sender.tab && sender.tab.id; (async () => { try { // ⭐ 1. الحصول على بروكسي جديد واختباره console.log('[Search] 🔄 جلب بروكسي جديد...'); const proxy = await getNextProxy(); // ⭐ 2. تطبيقه على AliExpress فقط (PAC) await applyProxyForAliExpress(proxy); // ⭐ 3. الانتظار حتى يستقر الاتصال await new Promise(r => setTimeout(r, 2000)); // ⭐ 4. بدء الزحف const q = encodeURIComponent((msg.title || '').slice(0, 90)); const baseUrl = 'https://www.aliexpress.com/wholesale?SearchText=' + q + '&catId=0'; const seenIds = new Set(); let firstBatchSent = false; let totalFetched = 0; for (let batch = 0; batch < MAX_PAGES; batch += PARALLEL_BATCH) { const pages = []; for (let i = 0; i < PARALLEL_BATCH && (batch + i) < MAX_PAGES; i++) { pages.push(batch + i + 1); } const results = await Promise.all(pages.map(p => fetchPage(baseUrl, p))); const newItems = []; let emptyPages = 0; for (const items of results) { if (!items || !items.length) { emptyPages++; continue; } for (const it of items) { if (!it || !it.id || seenIds.has(it.id)) continue; seenIds.add(it.id); newItems.push(it); } } totalFetched += newItems.length; if (newItems.length > 0 && tabId) { if (!firstBatchSent) { sendResponse({ ok: true, items: newItems, searchUrl: baseUrl, done: false, count: newItems.length }); firstBatchSent = true; } else { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_UPDATE', items: newItems, searchUrl: baseUrl }).catch(() => {}); } } // إذا فشلت صفحة (حظر)، نغير البروكسي فوراً if (emptyPages === pages.length && batch === 0) { console.warn('[Search] ⚠️ حظر محتمل، تبديل البروكسي...'); const newProxy = await getNextProxy(); await applyProxyForAliExpress(newProxy); await new Promise(r => setTimeout(r, 2000)); } if (emptyPages === pages.length) break; if (batch + PARALLEL_BATCH < MAX_PAGES) { await new Promise(r => setTimeout(r, DELAY_BETWEEN_BATCHES)); } } if (tabId) { chrome.tabs.sendMessage(tabId, { type: 'SMART_SEARCH_DONE', total: totalFetched }).catch(() => {}); } if (!firstBatchSent) { sendResponse({ ok: true, items: [], searchUrl: baseUrl, done: true }); } } catch (e) { console.error('[Search] فشل:', e); sendResponse({ ok: false, error: String(e) }); } })(); return true; }); /* ============================================ دالة جلب صفحة (بدون تعديل) ============================================ */ async function fetchPage(baseUrl, page) { try { const url = baseUrl + '&page=' + page; const res = await fetch(url, { credentials: 'include', headers: { 'Accept-Language': 'en-US,en;q=0.9', 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8' }, signal: AbortSignal.timeout(15000) }); if (!res.ok) { console.warn(`[Fetch] صفحة ${page} → HTTP ${res.status}`); return []; } const html = await res.text(); if (html.includes('captcha') || html.includes('punish')) { console.warn(`[Fetch] صفحة ${page} → حظر مكتشف`); return []; } let items = parseItems(html); if (!items.length) items = fallbackIds(html); return items; } catch (e) { console.warn(`[Fetch] صفحة ${page} فشلت:`, e.message); return []; } } /* ============================================ دوال التحليل (كما هي من كودك الأصلي) ============================================ */ function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const title = clean(o.title?.displayTitle || o.title || ''); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const price = sale.formatedActivityPrice || sale.formatedPrice || p.formatedPrice || ''; const discount = o.discount || sale.discount || ''; const rating = (o.evaluation && o.evaluation.starRating) || o.averageStarRate || ''; const sold = o.tradeDesc || o.sold || ''; out.push({ id: String(id), title: title || 'Produit similaire', img, price, discount, rating, sold, link: 'https://www.aliexpress.com/item/' + id + '.html' }); } catch {} } return out; } function fallbackIds(html) { const re = /"productId"\s*:\s*"(\d{6,})"/g; const seen = new Set(); const out = []; let m; while ((m = re.exec(html))) { if (seen.has(m[1])) continue; seen.add(m[1]); out.push({ id: m[1], title: 'Produit', img: '', price: '', discount: '', rating: '', sold: '', link: 'https://www.aliexpress.com/item/' + m[1] + '.html' }); } return out; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function clean(s) { return String(s).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); }
