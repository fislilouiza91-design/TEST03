/* ============================================ Smart Shopper - Background Service Worker (Clean & Fast Edition) ============================================ */ const WORKER_URL = 'https://shopper.smailsuit11.workers.dev'; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; const extractPid = (url) => { if (!url) return ''; const objectIdMatch = url.match(/x_object_id(?:%3A|:)(\d+)/i); if (objectIdMatch) return objectIdMatch[1]; const match = url.match(/(?:\/item\/|\/i\/|productId=)(\d{8,})/i); return match ? match[1] : ''; }; if (msg.type === 'SMART_SEARCH') { const pid = extractPid(sender.tab?.url); handleSearch(msg, pid, sendResponse); return true; } if (msg.type === 'SMART_LOAD_MORE') { const pid = extractPid(sender.tab?.url); handleLoadMore(msg, pid, sendResponse); return true; } return true; }); async function handleSearch(msg, pid, sendResponse) { const searchTitle = msg.title || ''; const url = `${WORKER_URL}/search-db?q=${encodeURIComponent(searchTitle)}&pid=${pid}&page=1`; try { const res = await fetch(url, { cache: 'no-store' }); const data = await res.json(); if (!res.ok || !data.ok) throw new Error(data.error || `HTTP ${res.status}`); sendResponse({ ok: true, items: data.items || [], lastPage: 1, hasMore: data.hasMore || false, totalItems: data.items ? data.items.length : 0 }); } catch (e) { sendResponse({ ok: false, error: String(e) }); } } async function handleLoadMore(msg, pid, sendResponse) { const title = encodeURIComponent(msg.title || ''); const nextPage = Math.max(2, parseInt(msg.startPage) || 2); const existingIds = (msg.existingIds || []).join(','); const url = `${WORKER_URL}/search-db?q=${title}&pid=${pid}&page=${nextPage}&existingIds=${existingIds}`; try { const res = await fetch(url, { cache: 'no-store' }); const data = await res.json(); if (!res.ok || !data.ok) throw new Error(data.error || `HTTP ${res.status}`); sendResponse({ ok: true, items: data.items || [], lastPage: nextPage, hasMore: data.hasMore || false }); } catch (e) { sendResponse({ ok: false, error: String(e), hasMore: false }); } }
