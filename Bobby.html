/* Nezzer History panel — Per-Card Skeleton Loading */ const grid = document.getElementById('grid'); const currentCards = new Map(); const ALLOWED_ORIGIN = /^https:\/\/([a-z0-9-]+\.)*aliexpress\.com$/i; const PLACEHOLDER = 'data:image/svg+xml;utf8,' + encodeURIComponent( '' ); function createCardEl() { const card = document.createElement('a'); card.className = 'product-card'; card.target = '_blank'; card.rel = 'noopener noreferrer'; const imgBox = document.createElement('div'); imgBox.className = 'img-box'; // النبض (Skeleton) يظهر افتراضياً const skeleton = document.createElement('div'); skeleton.className = 'skeleton-pulse'; const img = document.createElement('img'); img.style.display = 'none'; // مخفية حتى تُحمل img.loading = 'lazy'; // عندما يتم تحميل الصورة، نخفي النبض ونظهر الصورة img.onload = () => { img.style.display = 'block'; skeleton.style.display = 'none'; }; // في حال فشل التحميل، نظهر صورة بديلة ونخفي النبض img.onerror = () => { img.onerror = null; img.src = PLACEHOLDER; img.style.display = 'block'; skeleton.style.display = 'none'; }; imgBox.append(skeleton, img); const info = document.createElement('div'); info.className = 'info'; const title = document.createElement('div'); title.className = 'title'; const price = document.createElement('div'); price.className = 'price'; info.append(title, price); card.append(imgBox, info); card._img = img; card._skeleton = skeleton; card._title = title; card._price = price; return card; } function updateCardEl(card, item) { let href = '#'; try { const u = new URL(item.url); if (u.protocol === 'https:' || u.protocol === 'http:') href = item.url; } catch (e) {} card.href = href; card._title.textContent = item.title || ''; card._price.textContent = item.price || '—'; let imgSrc = item.img || ''; if (imgSrc.startsWith('//')) imgSrc = 'https:' + imgSrc; if (imgSrc) { // إذا كان هناك رابط صورة جديد، قم بتحميله if (card._img.getAttribute('src') !== imgSrc) { card._img.style.display = 'none'; card._skeleton.style.display = 'block'; // إعادة تشغيل النبض حتى يتم التحميل card._img.src = imgSrc; } } else { // إذا لم توجد صورة، أبقِ النبض يعمل card._img.style.display = 'none'; card._skeleton.style.display = 'block'; } } function renderCards(products) { if (!products || !products.length) { grid.innerHTML = '
No products found.
'; currentCards.clear(); return; } const empty = grid.querySelector('.empty'); if (empty) empty.remove(); const seen = new Set(); const frag = document.createDocumentFragment(); for (const item of products) { if (!item || !item.productId) continue; seen.add(item.productId); const fp = item.productId + '|' + (item.price || '') + '|' + (item.title || '') + '|' + (item.img || ''); const existing = currentCards.get(item.productId); if (existing && existing.fp === fp) { frag.appendChild(existing.el); } else { const card = existing ? existing.el : createCardEl(); updateCardEl(card, item); currentCards.set(item.productId, { el: card, fp }); frag.appendChild(card); } } for (const id of Array.from(currentCards.keys())) { if (!seen.has(id)) currentCards.delete(id); } grid.replaceChildren(frag); } window.addEventListener('message', (event) => { if (!ALLOWED_ORIGIN.test(event.origin)) return; const data = event.data; if (!data || typeof data !== 'object') return; if (data.action === 'RENDER_DATA') { renderCards(data.products); } else if (data.action === 'RENDER_LOADING') { grid.innerHTML = '
⏳ Loading…
'; currentCards.clear(); } else if (data.action === 'CAPTCHA_FOUND') { grid.innerHTML = '
⚠️ AliExpress is checking your browser (CAPTCHA).
Please solve it on the page first.
'; currentCards.clear(); } });
