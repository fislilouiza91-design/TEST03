/* ============================================ Smart Shopper - Background Service Worker Local Search (no AI) - 8 keywords optimized ============================================ */ const WORKER_URL = 'https://extension.smailsuit11.workers.dev/'; const MAX_PAGES_TO_FETCH = 3; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } console.log(`🎯 [${msg.type}] Query: [${searchQuery}] | Page: ${page}`); console.log(`📝 Original: "${msg.title}"`); const q = encodeURIComponent(searchQuery); // 🔥 لا نرسل originalTitle (البحث محلي بالكامل) fetch(`${WORKER_URL}?q=${q}&page=${page}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: 100 })); let hasMore = items.length >= 15; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } if (msg.type === 'SORT_ITEMS') { fetch(`${WORKER_URL}sort`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ originalTitle: msg.title, items: msg.items }) }) .then(res => res.json()) .then(data => sendResponse({ ok: data.success, items: data.data })) .catch(err => sendResponse({ ok: false, error: err.message })); return true; } }); // 🔥 استخراج الكلمة الدقيقة (محسّن - 8 كلمات، يحافظ على Brand/Model) function buildSmartSearchQuery(title) { if (!title) return ''; const STOP_WORDS = new Set([ 'the', 'and', 'for', 'with', 'new', 'hot', 'sale', 'free', 'shipping', 'original', 'high', 'quality', 'best', 'wholesale', 'dropshipping', 'fast', 'quick', 'good', 'perfect', 'gift', 'fashion', 'luxury', 'premium', '2020', '2021', '2022', '2023', '2024', '2025', '2026', 'pcs', 'piece', 'pieces', 'set', 'pack', 'items', 'item', 'lot', 'bundle', 'waterproof', 'shockproof', 'multifunctional', 'portable', 'heavy', 'duty', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'gratuit', 'livraison', 'nouveau', 'promo', 'solde', 'chez', 'plus' ]); // كلمات قصيرة مهمة (تُحفظ رغم قصرها) const KEEP_SHORT = new Set([ 'usb', 'led', 'lcd', 'tv', 'pc', 'hd', 'uhd', '4k', '8k', '2k', 'hdmi', 'sd', 'sim', 'ip', 'ai', 'pro', 'max', 'air', 'mini', 'gen' ]); const cleanTitle = title .replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ') .replace(/\s+/g, ' ') .trim(); const words = cleanTitle.split(' '); const usefulWords = words.filter(w => { if (!w) return false; const lower = w.toLowerCase(); if (STOP_WORDS.has(lower)) return false; if (KEEP_SHORT.has(lower)) return true; if (w.length < 2) return false; // احتفظ بالأرقام الطويلة (موديلات مثل 22mm, 1080p) if (/^\d{1,2}$/.test(w) && w.length < 3) return false; return true; }); // 🔥 الحفاظ على 8 كلمات (بدلاً من 4) const finalQuery = usefulWords.slice(0, 8).join(' '); if (!finalQuery) { return words.slice(0, 4).join(' '); } return finalQuery; }
