import { createClient } from '@libsql/client'; import fetch from 'node-fetch'; import dotenv from 'dotenv'; dotenv.config(); const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const ANT_API_KEY = process.env.SCRAPINGANT_API_KEY; const MARKET = { lang: 'en', code: 'US' }; const BATCH_SIZE = 10; // كلمات في الدورة الواحدة // ⚠️ التعديل هنا: قراءة الوقت من GitHub Actions (355 دقيقة) أو استخدام 355 كافتراضي const MAX_RUNTIME_MINUTES = parseInt(process.env.MAX_RUNTIME_MINUTES || 355); const MAX_RUNTIME_MS = MAX_RUNTIME_MINUTES * 60 * 1000; const MAX_SAFETY_PAGES = 50; // حد أقصى للأمان (لن يسحب أكثر من 50 صفحة للكلمة الواحدة مهما حدث) // ========================================== // إدارة الفئة العشوائية والاستكمال من حيث انتهى // ========================================== async function getActiveCategory() { try { const stateRes = await db.execute(`SELECT value FROM system_state WHERE key = 'active_category'`); let activeCat = stateRes.rows.length > 0 ? String(stateRes.rows[0].value) : null; if (activeCat) { const pending = await db.execute({ sql: `SELECT 1 FROM keywords_pipeline WHERE category = ? AND last_run_at = 0 AND enabled = 1 LIMIT 1`, args: [activeCat] }); if (pending.rows.length > 0) return activeCat; } const catsRes = await db.execute(` SELECT DISTINCT category FROM keywords_pipeline WHERE last_run_at = 0 AND enabled = 1 AND category IS NOT NULL AND category != '' ORDER BY RANDOM() LIMIT 1 `); if (catsRes.rows.length > 0) { activeCat = String(catsRes.rows[0].category); console.log(`\n🎲 الانتقال عشوائياً إلى فئة جديدة: [${activeCat}]`); } else { console.log(`\n🚨 اكتملت جميع الكلمات! جاري تصفير الطابور لدورة جديدة...`); await db.execute(`UPDATE keywords_pipeline SET last_run_at = 0 WHERE enabled = 1`); const resetCatsRes = await db.execute(` SELECT DISTINCT category FROM keywords_pipeline WHERE enabled = 1 ORDER BY RANDOM() LIMIT 1 `); if (resetCatsRes.rows.length === 0) return null; activeCat = String(resetCatsRes.rows[0].category); console.log(`🔄 بدء دورة جديدة من فئة عشوائية: [${activeCat}]`); } await db.execute({ sql: `INSERT OR REPLACE INTO system_state (key, value, updated_at) VALUES ('active_category', ?, ?)`, args: [activeCat, Date.now()] }); return activeCat; } catch (e) { return null; } } async function getKeywordsForCategory(category, limit) { const res = await db.execute({ sql: `SELECT keyword FROM keywords_pipeline WHERE category = ? AND last_run_at = 0 AND enabled = 1 ORDER BY RANDOM() LIMIT ?`, args: [category, limit] }); return res.rows.map(r => String(r.keyword)); } async function markKeywordProcessed(keyword) { await db.execute({ sql: `UPDATE keywords_pipeline SET last_run_at = ?, run_count = run_count + 1 WHERE keyword = ?`, args: [Date.now(), keyword] }); } // ========================================== // الجلب والاستخراج // ========================================== async function fetchSearchPage(keyword, page) { const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${ANT_API_KEY}&browser=true&return_page_source=true&proxy_type=datacenter&proxy_country=${MARKET.code}`; const response = await fetch(antUrl); if (!response.ok) throw new Error(`HTTP Error: ${response.status}`); return await response.text(); } function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const orig = p.originalPrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const oldPriceStr = extractText(orig.formatedPrice || orig.minPrice || ''); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img, priceStr, oldPriceStr, rating: parseFloat(rating) || 0, soldStr: sold }); } catch (e) {} } return out; } function extractPriceNum(s) { const m = String(s || '').replace(/,/g, '').match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractSoldCount(s) { if (!s) return 0; const str = String(s).toLowerCase(); if (str.includes('k')) return Math.floor(parseFloat(str.replace(/[^0-9.]/g, '')) * 1000); const m = str.match(/(\d+)/); return m ? parseInt(m[1]) : 0; } async function saveToDatabase(products, keyword, language, category) { if (products.length === 0) return 0; const now = Date.now(); let savedCount = 0; for (const p of products) { try { const price = extractPriceNum(p.priceStr); const oldPrice = extractPriceNum(p.oldPriceStr); const orders = extractSoldCount(p.soldStr); const productUrl = `https://www.aliexpress.com/item/${p.id}.html`; // نتحقق مما إذا كان المنتج جديداً فعلاً (لمنع التكرار) // نستخدم INSERT OR IGNORE للمنتجات، ونحدث التفاصيل فقط إذا كانت جديدة const insertRes = await db.execute({ sql: `INSERT OR IGNORE INTO products (product_id, title, product_url, main_image, category, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?)`, args: [p.id, p.title, productUrl, p.img, category, now, now] }); // إذا تم إدخال صف جديد (rowsAffected > 0)، فهذا يعني أنه منتج جديد if (insertRes.rowsAffected > 0) { savedCount++; await db.execute({ sql: `INSERT OR REPLACE INTO product_details (product_id, current_price, original_price, rating, orders_count, last_updated) VALUES (?, ?, ?, ?, ?, ?)`, args: [p.id, price, oldPrice, p.rating, orders, now] }); await db.execute({ sql: `INSERT INTO price_history (product_id, price, recorded_at) VALUES (?, ?, ?)`, args: [p.id, price, now] }); } await db.execute({ sql: `INSERT OR IGNORE INTO keywords_mapping (product_id, keyword, language) VALUES (?, ?, ?)`, args: [p.id, keyword, language] }); } catch (dbError) {} } return savedCount; } // ========================================== // التشغيل الرئيسي // ========================================== async function main() { const START_TIME = Date.now(); console.log(`🚀 بدء محرك التدوير العشوائي... (مدة التشغيل المحددة: ${MAX_RUNTIME_MINUTES} دقيقة)\n`); while (Date.now() - START_TIME < MAX_RUNTIME_MS) { const category = await getActiveCategory(); if (!category) break; const keywords = await getKeywordsForCategory(category, BATCH_SIZE); if (!keywords || keywords.length === 0) { await new Promise(r => setTimeout(r, 2000)); continue; } console.log(`\n📋 سحب ${keywords.length} كلمة من فئة [${category}]`); for (const kw of keywords) { if (Date.now() - START_TIME >= MAX_RUNTIME_MS) { console.log(`\n⏰ وصلنا إلى الحد الأقصى للوقت. جاري الخروج بأمان...`); return; } console.log(`\n🔍 البحث عن: "${kw}"`); let page = 1; let consecutiveDuplicatePages = 0; // عداد للصفحات التي لا تحتوي على منتجات جديدة while (true) { // التحقق من الوقت قبل كل صفحة if (Date.now() - START_TIME >= MAX_RUNTIME_MS) { console.log(`\n⏰ وصلنا إلى الحد الأقصى للوقت. جاري الخروج بأمان...`); return; } // حد الأمان لمنع الحلقة اللانهائية if (page > MAX_SAFETY_PAGES) { console.log(` 🛑 وصلنا لحد الأمان (${MAX_SAFETY_PAGES} صفحة). الانتقال للكلمة التالية.`); break; } try { const html = await fetchSearchPage(kw, page); if (html.length < 5000) { console.log(` 🏁 لا توجد نتائج إضافية. نهاية البحث لهذه الكلمة.`); break; } const products = parseItems(html); if (products.length === 0) { console.log(` 🏁 لا توجد منتجات في الصفحة ${page}. نهاية البحث.`); break; } const saved = await saveToDatabase(products, kw, MARKET.lang, category); console.log(` ✅ صفحة ${page}: استخراج ${products.length} (جديد: ${saved})`); // إذا لم يتم حفظ أي منتج جديد if (saved === 0) { consecutiveDuplicatePages++; // إذا تكررت النتائج في صفحتين متتاليتين، نتوقف (الموقع بدأ يعيد نفس المنتجات) if (consecutiveDuplicatePages >= 2) { console.log(` ⏹️ المنتجات بدأت تتكرر. التوقف عن هذه الكلمة لتوفير الرصيد.`); break; } } else { consecutiveDuplicatePages = 0; // تصفير العداد لأننا وجدنا منتجات جديدة } page++; // الانتقال للصفحة التالية await new Promise(r => setTimeout(r, 3000)); // تأخير بشري } catch (error) { console.error(` ⚠️ خطأ في صفحة ${page}:`, error.message); break; // نتوقف عن هذه الكلمة إذا حدث خطأ } } // وضع علامة "تمت" فوراً بعد الكلمة await markKeywordProcessed(kw); } } console.log(`\n🏁 انتهى وقت التشغيل.`); } main().catch(console.error);
