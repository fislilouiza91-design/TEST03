/* ============================================ Smart Shopper - Background Service Worker (Global ID Edition) ============================================ */ const WORKER_URL = 'https://shopper.smailsuit11.workers.dev'; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; // 🔥 الدالة الذكية: تستخرج المعرف العالمي (1005...) من أي رابط في العالم const extractPid = (url) => { if (!url) return ''; // 1. البحث عن المعرف الحقيقي المخفي في أكواد التتبع (الخاص بالموقع الأمريكي) const objectIdMatch = url.match(/x_object_id(?:%3A|:)(\d+)/i); if (objectIdMatch) return objectIdMatch[1]; // 2. البحث عن المعرف العادي في المسار (الخاص بالمواقع العربية والأوروبية) const match = url.match(/(?:\/item\/|\/i\/|productId=)(\d{8,})/i); return match ? match[1] : ''; }; if (msg.type === 'SMART_SEARCH') { const pid = extractPid(sender.tab?.url); handleSearch(msg, pid, sendResponse); return true; } if (msg.type === 'SMART_LOAD_MORE') { const pid = extractPid(sender.tab?.url); handleLoadMore(msg, pid, sendResponse); return true; } return true; }); async function handleSearch(msg, pid, sendResponse) { const t = Date.now(); const url = `${WORKER_URL}/search-db?q=${encodeURIComponent(msg.title || '')}&pid=${pid}&page=1&t=${t}`; try { const res = await fetch(url, { cache: 'no-store' }); if (!res.ok) throw new Error(`HTTP Error: ${res.status}`); const data = await res.json(); sendResponse({ ok: true, items: data.items || [], lastPage: 1, hasMore: data.hasMore || false, totalItems: data.items ? data.items.length : 0 }); } catch (e) { sendResponse({ ok: false, error: String(e) }); } } async function handleLoadMore(msg, pid, sendResponse) { const nextPage = Math.max(2, parseInt(msg.startPage) || 2); const existingIds = (msg.existingIds || []).join(','); const t = Date.now(); const url = `${WORKER_URL}/search-db?q=${encodeURIComponent(msg.title || '')}&pid=${pid}&page=${nextPage}&existingIds=${existingIds}&t=${t}`; try { const res = await fetch(url, { cache: 'no-store' }); if (!res.ok) throw new Error(`HTTP Error: ${res.status}`); const data = await res.json(); sendResponse({ ok: true, items: data.items || [], lastPage: nextPage, hasMore: data.hasMore || false }); } catch (e) { sendResponse({ ok: false, error: String(e), hasMore: false }); } }
