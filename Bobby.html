/* ============================================ Smart Shopper - Background Service Worker (Dynamic Keywords) ============================================ */ const WORKER_URL = 'https://ali-scraper.your-username.workers.dev'; const MAX_PAGES_TO_FETCH = 2; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } console.log(`🎯 كلمة البحث الديناميكية: [${searchQuery}]`); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalPrice = msg.productPrice || 0; fetch(`${WORKER_URL}?q=${q}&page=${page}&originalTitle=${originalTitle}&originalPrice=${originalPrice}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: item.score || 100 })); let hasMore = items.length >= 15; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } }); // 🔥 الاستخراج الديناميكي: نأخذ أول 3-4 كلمات مهمة من العنوان function buildSmartSearchQuery(title) { if (!title) return ''; // تنظيف العنوان وإزالة الرموز const cleanTitle = title.replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ').replace(/\s+/g, ' ').trim(); const rawWords = cleanTitle.split(' '); // كلمات الحشو التي لا تفيد البحث const stopWords = new Set([ 'the', 'and', 'for', 'with', 'new', 'hot', 'sale', 'free', 'shipping', 'original', 'high', 'quality', 'best', 'pcs', 'set', 'pack', 'piece', 'pieces', 'items', 'item', 'wholesale', 'dropshipping', 'compatible', 'fits', 'fit', 'suitable', 'accessories', 'accessory', 'universal', 'dual', 'purpose', 'large', 'screen', 'plug', 'creative', 'dedicated', 'student', 'led', 'portable', 'multifunctional', 'design', 'style', 'luxury', 'fast', 'quick', 'smart', 'waterproof', 'shockproof' ]); const usefulWords = rawWords.filter(w => { const lower = w.toLowerCase(); // تجاهل الكلمات القصيرة، الأرقام، المقاسات، وكلمات الحشو return w.length > 1 && !stopWords.has(lower) && !/^\d+$/.test(lower) && !/^\d+(mm|cm|inch)$/.test(lower); }); // نأخذ أول 3 كلمات مفيدة. هذا سيضمن البحث عن "المنتج + الصفة الأساسية" // مثال: "Curved Electronic Clock" بدلاً من "Battery" return usefulWords.slice(0, 3).join(' '); }
