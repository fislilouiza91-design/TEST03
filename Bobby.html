import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); async function main() { console.log('📊 عرض أحدث 20 منتج:\n'); const res = await db.execute(` SELECT product_id, title, current_price, category, trust_score FROM products_current ORDER BY last_updated_at DESC LIMIT 20 `); res.rows.forEach((r, i) => { console.log(`${i+1}. [${r.category || 'NULL'}] ${(r.title || '').substring(0, 60)}`); console.log(` 💰 ${r.current_price}$ | 🛡️ Trust: ${r.trust_score || 'N/A'}`); console.log(''); }); console.log('\n📊 إحصائيات:'); const total = await db.execute('SELECT COUNT(*) as c FROM products_current'); console.log(` إجمالي المنتجات: ${total.rows[0].c}`); const newOnes = await db.execute({ sql: `SELECT COUNT(*) as c FROM products_current WHERE last_updated_at > ?`, args: [Date.now() - 3600000] }); console.log(` محدّثة في آخر ساعة: ${newOnes.rows[0].c}`); // إحصائيات حسب الفئة console.log('\n📂 التوزيع حسب الفئة:'); const cats = await db.execute(` SELECT category, COUNT(*) as c FROM products_current GROUP BY category ORDER BY c DESC `); cats.rows.forEach(r => { console.log(` ${r.category || 'NULL'}: ${r.c}`); }); } main().catch(console.error);
