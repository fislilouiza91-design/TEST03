export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } if (!env.AI || items.length === 0) { return new Response(JSON.stringify({ success: true, data: items }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const modelName = '@cf/baai/bge-m3'; const originalIsAccessory = accessoryKeywords.test(cleanOriginalTitle); // 🔥 دالة تجزئة بسيطة const STOP_WORDS = new Set([ 'the', 'and', 'for', 'with', 'new', 'hot', 'sale', 'free', 'shipping', 'original', 'high', 'quality', 'best', 'pcs', 'set', 'pack', 'piece', 'pieces', 'items', 'item', 'wholesale', 'dropshipping', 'compatible', 'fits', 'fit', 'suitable', 'accessories', 'accessory', 'universal' ]); const tokenize = (text) => { return text.toLowerCase() .replace(/[^a-z0-9\u0600-\u06FF\u0400-\u04FF\s]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !STOP_WORDS.has(w)); }; const originalTokensList = tokenize(originalTitle); const originalTokens = new Set(originalTokensList); const originalTokenCount = Math.max(originalTokensList.length, 1); // ========================================== // المرحلة 1: فرز محلي لاختيار أفضل 5 مرشحين // ========================================== const preScoredItems = items.map(it => { const itemTokens = new Set(tokenize(it.title)); if (itemTokens.size === 0) return { ...it, preScore: 0 }; const intersection = [...itemTokens].filter(w => originalTokens.has(w)).length; const union = new Set([...itemTokens, ...originalTokens]).size; const jaccard = union > 0 ? intersection / union : 0; return { ...it, preScore: jaccard }; }); preScoredItems.sort((a, b) => b.preScore - a.preScore); const aiItems = preScoredItems.slice(0, 5); const remainingItems = preScoredItems.slice(5); // ========================================== // المرحلة 2: الذكاء الاصطناعي لأفضل 5 منتجات // ========================================== const originalEmbedding = await env.AI.run(modelName, { text: [originalTitle] }); const baseVector = originalEmbedding.data[0]; const aiTitles = aiItems.map(it => it.title); const aiEmbeddings = await env.AI.run(modelName, { text: aiTitles }); // 🔥 منطق جديد: افتراضياً "مطابق" إلا إذا كان ملحقاً let scoredAIItems = aiItems.map((it, index) => { const aiScore = cosineSimilarity(baseVector, aiEmbeddings.data[index]); const itemTitleLower = it.title.toLowerCase(); const itemIsAccessory = accessoryKeywords.test(itemTitleLower); const isActualAccessory = itemIsAccessory && !originalIsAccessory; // المنتجات الـ 5 الأولى تعتبر "مطابقة" ما لم تكن ملحقاً const isExact = !isActualAccessory; let finalScore = aiScore * 100; if (isExact) finalScore += 50; return { ...it, finalScore, isExact }; }); scoredAIItems.sort((a, b) => b.finalScore - a.finalScore); // ========================================== // المرحلة 3: ترتيب محلي للباقي // ========================================== const referenceItems = scoredAIItems; const totalRefScore = referenceItems.reduce((sum, r) => sum + Math.max(r.finalScore, 1), 0) || 1; const referenceData = referenceItems.map(ref => ({ tokens: new Set(tokenize(ref.title)), weight: Math.max(ref.finalScore, 1) / totalRefScore })); const scoredRemaining = remainingItems.map(it => { const itemTokens = new Set(tokenize(it.title)); const itemTitleLower = it.title.toLowerCase(); if (itemTokens.size === 0) { return { ...it, finalScore: 0, isExact: false }; } // حساب درجات التشابه let weightedSimilarity = 0; for (const ref of referenceData) { const intersection = [...itemTokens].filter(w => ref.tokens.has(w)).length; const union = new Set([...itemTokens, ...ref.tokens]).size; const jaccard = union > 0 ? intersection / union : 0; weightedSimilarity += jaccard * ref.weight; } const origIntersection = [...itemTokens].filter(w => originalTokens.has(w)).length; const origUnion = new Set([...itemTokens, ...originalTokens]).size; const origJaccard = origUnion > 0 ? origIntersection / origUnion : 0; const keyTermsRatio = origIntersection / originalTokenCount; const combinedScore = (weightedSimilarity * 0.5 + origJaccard * 0.2 + keyTermsRatio * 0.3) * 150; const finalScore = Math.min(combinedScore, 85); // 🔥 منطق جديد: افتراضياً "مطابق" إلا إذا كان ملحقاً واضحاً const itemIsAccessory = accessoryKeywords.test(itemTitleLower); const isActualAccessory = itemIsAccessory && !originalIsAccessory; // فقط الملحقات الواضحة أو المنتجات المختلفة تماماً تُصنّف كمشابهة const isExact = !isActualAccessory; return { ...it, finalScore, isExact }; }); scoredRemaining.sort((a, b) => b.finalScore - a.finalScore); // ========================================== // المرحلة 4: الدمج // ========================================== const finalItems = [...scoredAIItems, ...scoredRemaining]; return new Response(JSON.stringify({ success: true, data: finalItems }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } // ========================================== // الكود القديم (جلب المنتجات) // ========================================== let keyword = url.searchParams.get("q"); const originalTitle = url.searchParams.get("originalTitle") || ""; const page = parseInt(url.searchParams.get("page") || "1"); if (page > 3) { return new Response(JSON.stringify({ success: true, data: [], hasMore: false }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } try { if (originalTitle && env.AI) { keyword = await extractKeywordsWithAI(originalTitle, env); } if (!keyword) { return new Response(JSON.stringify({ success: false, message: "Missing keyword" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } const searchUrl = `https://www.aliexpress.com/wholesale?SearchText=${encodeURIComponent(keyword)}&page=${page}`; const antUrl = `https://api.scrapingant.com/v2/general?url=${encodeURIComponent(searchUrl)}&x-api-key=${env.SCRAPINGANT_API_KEY}&return_page_source=true&proxy_type=datacenter&browser=false`; const antResponse = await fetch(antUrl); if (!antResponse.ok) throw new Error(`ScrapingAnt Error: ${antResponse.status}`); const html = await antResponse.text(); if (html.includes('punish') || html.includes('captcha') || html.length < 5000) { return new Response(JSON.stringify({ success: false, message: "كابتشا أو حظر مؤقت." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } let products = parseItems(html); if (products.length === 0) { return new Response(JSON.stringify({ success: false, message: "لا توجد منتجات." }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } return new Response(JSON.stringify({ success: true, data: products, aiQuery: keyword }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } }; // ========================================== // 🧠 دوال مساعدة // ========================================== async function extractKeywordsWithAI(title, env) { try { const response = await env.AI.run('@cf/meta/llama-3-8b-instruct', { messages: [ { role: 'system', content: 'You are an e-commerce search expert. Extract the main product type, brand, and model from the title. Output ONLY 3 to 4 English keywords separated by spaces. Do NOT include colors, sizes, numbers, or promotional words like "free shipping". Do NOT write sentences.' }, { role: 'user', content: title } ], max_tokens: 20 }); if (response && response.response) { let cleanQuery = response.response.trim().replace(/[^a-zA-Z0-9\s]/g, ''); return cleanQuery.split(/\s+/).slice(0, 4).join(' '); } return ''; } catch (e) { return ''; } } function cosineSimilarity(vecA, vecB) { let dotProduct = 0, normA = 0, normB = 0; for (let i = 0; i < vecA.length; i++) { dotProduct += vecA[i] * vecB[i]; normA += vecA[i] * vecA[i]; normB += vecB[i] * vecB[i]; } return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB)); } function extractText(v) { if (v == null) return ''; if (typeof v === 'string') return v; if (typeof v === 'number') return String(v); if (typeof v === 'object') { const keys = ['displayTitle', 'text', 'title', 'value', 'formatedActivityPrice', 'formatedPrice', 'minActivityPrice', 'minPrice', 'price']; for (const k of keys) if (v[k] != null && typeof v[k] !== 'object') return String(v[k]); for (const k in v) if (typeof v[k] === 'string' && v[k].length > 0) return v[k]; return ''; } return String(v); } function extractPriceNumber(s) { if (!s) return 0; const str = String(s).replace(/,/g, ''); const m = str.match(/(\d+(?:\.\d+)?)/); return m ? parseFloat(m[1]) : 0; } function extractJsonArray(html, marker) { const i = html.indexOf(marker); if (i < 0) return null; const j = html.indexOf('[', i); if (j < 0) return null; let depth = 0, inStr = false, esc = false; for (let k = j; k < html.length; k++) { const c = html[k]; if (inStr) { if (esc) esc = false; else if (c === '\\') esc = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '[') depth++; else if (c === ']') { depth--; if (depth === 0) return html.slice(j, k + 1); } } return null; } function parseItems(html) { const arrText = extractJsonArray(html, '"itemList"') || extractJsonArray(html, '"gridItems"'); if (!arrText) return []; let arr; try { arr = JSON.parse(arrText); } catch (e) { return []; } if (!Array.isArray(arr)) return []; const out = []; for (const o of arr) { try { const id = o.productId || o.id || (o.item && o.item.productId); if (!id) continue; const rawTitle = o.title || (o.item && o.item.title) || o.displayTitle || ''; const title = String(extractText(rawTitle)).replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim(); let img = ''; if (o.image) img = typeof o.image === 'string' ? o.image : (o.image.imgUrl || o.image.imageUrl || o.image.url || ''); if (!img && o.picUrl) img = typeof o.picUrl === 'string' ? o.picUrl : (o.picUrl.imgUrl || o.picUrl.url || ''); if (img && img.startsWith('//')) img = 'https:' + img; const p = o.prices || {}; const sale = p.salePrice || {}; const priceStr = extractText(sale.formatedActivityPrice || sale.formatedPrice || sale.minActivityPrice || sale.minPrice || p.formatedPrice || p.minPrice || ''); const priceNum = extractPriceNumber(priceStr); const rating = extractText((o.evaluation && (o.evaluation.starRating || o.evaluation.averageStar)) || o.averageStarRate || o.starRating || ''); const sold = extractText(o.tradeDesc || o.sold || o.soldCount || ''); out.push({ id: String(id), title: title || 'Produit', img: img, price: priceNum, rating: parseFloat(rating) || 0, soldStr: sold, url: `https://www.aliexpress.com/item/${id}.html` }); } catch (e) {} } return out; }
