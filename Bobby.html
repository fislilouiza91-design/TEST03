const sleep = (ms) => new Promise(r => setTimeout(r, ms)); async function fetchSimilarProducts({ page = 1, streamId = null }) { const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true }); if (!tab?.url?.match(/\/item\/(\d+)\.htm/)) return { error: 'NOT_PRODUCT_PAGE' }; const productId = tab.url.match(/\/item\/(\d+)\.htm/)[1]; // جلب صورة المنتج الأساسي (مطلوبة للبحث في الواجهة الروسية) let pageData; try { const [result] = await chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => { const ogImageElement = document.querySelector('meta[property="og:image"]'); return { ogImage: ogImageElement ? ogImageElement.content : '', host: location.host }; } }); pageData = result?.result; } catch (e) { return { error: 'PAGE_DATA_FAILED' }; } if (!pageData || !pageData.ogImage) return { error: 'NO_IMAGE_FOUND' }; try { // استخدام الواجهة الداخلية لموقع aliexpress.ru لجلب الأسعار الحقيقية const url = 'https://aliexpress.ru/aeglodetailweb/api/recommend/v2/query'; const payload = { productId: productId, imageUrl: pageData.ogImage.split('_')[0], page: page, pageSize: 20, searchBizScene: "imageSearch", shipToCountry: "RU", currency: "RUB", locale: "ru_RU", platform: "pc" }; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json", "Accept": "application/json, text/plain, */*" }, body: JSON.stringify(payload) }); if (!response.ok) { return { error: 'RU_API_ERROR: ' + response.status }; } const json = await response.json(); // استخراج المنتجات من استجابة الموقع الروسي let rawProducts = json?.data?.mods?.itemList?.content || json?.result?.mods?.itemList?.content || json?.data?.itemList || []; // تحويل البيانات لتتوافق مع واجهة الإضافة const products = rawProducts.map(p => { const priceNum = p?.prices?.salePrice?.minPrice || p?.prices?.salePrice?.formattedPrice || p?.price?.formattedPrice || p?.price?.minPrice || '0'; return { productId: p.productId || p.id, title: { displayTitle: p.title?.displayTitle || p.title || 'Product' }, image: { imgUrl: p.image?.imgUrl || p.imageUrl || '' }, prices: { salePrice: { formattedPrice: typeof priceNum === 'string' && priceNum.includes('₽') ? priceNum : `${priceNum} ₽`, minPrice: p?.prices?.salePrice?.minPrice || p?.price?.minPrice || 0 } }, trade: { tradeCount: p.trade?.tradeCount || p.tradeCount || 0 }, evaluation: { starRating: p.evaluation?.starRating || p.starRating || 0 } }; }); return { products: products, streamId: null, // الواجهة الروسية لا تستخدم streamId بنفس الطريقة ret: 'SUCCESS', host: pageData.host }; } catch (error) { return { error: 'FETCH_FAILED: ' + error.message }; } } chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (msg.action === 'FETCH_SIMILAR') { fetchSimilarProducts({ page: msg.page || 1, streamId: msg.streamId || null }) .then(sendResponse) .catch(err => sendResponse({ error: err.message })); return true; } });
