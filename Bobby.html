/* ============================================ Smart Shopper - Background Service Worker (Broad Search to get more exact matches) ============================================ */ const WORKER_URL = 'https://competitors-api.smailsuit11.workers.dev'; const MAX_PAGES_TO_FETCH = 4; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { fetch(`${WORKER_URL}/api/competitors?productId=${msg.productId}`) .then(res => { if(res.ok) return res.json(); throw new Error('Not found'); }) .then(data => sendResponse({ ok: true, data: data.data })) .catch(() => sendResponse({ ok: false })); return true; } if (msg.type === 'SMART_SEARCH') { const searchQuery = buildSearchQuery(msg.title); console.log('🔍 Broad Search Query:', searchQuery); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=1&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { sendResponse({ ok: true, items: data.items, hasMore: data.items.length > 0 }); if (msg.productId && data.items.length > 0) saveToCloudflare(msg.productId, data.items); } else { sendResponse({ ok: false, error: data.error }); } }).catch(err => sendResponse({ ok: false, error: err.message })); return true; } if (msg.type === 'SMART_LOAD_MORE') { const nextPage = Math.max(2, parseInt(msg.startPage) || 2); if (nextPage > MAX_PAGES_TO_FETCH) return sendResponse({ ok: true, items: [], hasMore: false }); const searchQuery = buildSearchQuery(msg.title); const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&originalImage=${originalImage}&page=${nextPage}&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { let existingIds = new Set(msg.existingIds); let unique = data.items.filter(i => !existingIds.has(i.id)); const hasMore = unique.length > 0 && nextPage < MAX_PAGES_TO_FETCH; sendResponse({ ok: true, items: unique, hasMore: hasMore }); if (msg.productId && unique.length > 0) saveToCloudflare(msg.productId, unique); } else { sendResponse({ ok: false, error: data.error, hasMore: false }); } }).catch(err => sendResponse({ ok: false, error: err.message, hasMore: false })); return true; } }); // 🔥 بحث أوسع لجلب منتجات أكثر function buildSearchQuery(title) { if (!title) return ''; const cleanTitle = title.toLowerCase().replace(/[^a-z0-9\s]/g, ' '); const words = cleanTitle.split(/\s+/).filter(w => w.length > 1); // استخراج الموديل (مثل h3, w68) const models = words.filter(w => /[a-z]/.test(w) && /[0-9]/.test(w) && !w.includes('.')); const brand = words[0] || ''; if (models.length > 0) { // البحث بالماركة + الموديل + كلمة مساعدة return `${brand} ${models[0]} bluetooth`; } return `${brand} bluetooth`; } function saveToCloudflare(productId, items) { const payload = { base_product_id: productId, all_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })), relevant_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })) }; fetch(`${WORKER_URL}/api/competitors`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).then(res => res.json()).then(data => console.log('💾 Saved:', data)).catch(console.error); }
