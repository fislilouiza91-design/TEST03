// src/monitor.js export class PerformanceMonitor { constructor(options = {}) { this.slowThresholdMs = options.slowThresholdMs || 15000; this.maxConsecutiveFails = options.maxConsecutiveFails || 3; this.maxConsecutiveSlows = options.maxConsecutiveSlows || 5; this.consecutiveFails = 0; this.consecutiveSlows = 0; this.totalRequests = 0; this.totalFails = 0; this.totalSlows = 0; this.circuitOpen = false; this.reason = ''; } recordSuccess(durationMs) { this.totalRequests++; if (durationMs > this.slowThresholdMs) { this.totalSlows++; this.consecutiveSlows++; this.consecutiveFails = 0; console.log(` ⚠️ بطيء: ${(durationMs/1000).toFixed(1)}ث`); if (this.consecutiveSlows >= this.maxConsecutiveSlows) { this.openCircuit(`بطء متتالي (${this.consecutiveSlows})`); } } else { this.consecutiveSlows = 0; this.consecutiveFails = 0; } } recordFailure(errorMessage) { this.totalRequests++; this.totalFails++; this.consecutiveFails++; this.consecutiveSlows = 0; console.log(` ❌ فشل (${this.consecutiveFails}/${this.maxConsecutiveFails})`); if (this.consecutiveFails >= this.maxConsecutiveFails) { this.openCircuit(`فشل متتالي (${this.consecutiveFails})`); } } openCircuit(reason) { this.circuitOpen = true; this.reason = reason; console.log(`\n🛑 تم فتح قاطع الدائرة: ${reason}\n`); } canProceed() { return !this.circuitOpen; } getReport() { return { totalRequests: this.totalRequests, totalFails: this.totalFails, totalSlows: this.totalSlows, circuitOpen: this.circuitOpen, reason: this.reason, }; } }
