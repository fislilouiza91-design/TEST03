/* ============================================ Smart Shopper - Background Service Worker (Hidden Tab Fetch Edition) ============================================ */ const WORKER_URL = 'https://shopper.smailsuit11.workers.dev'; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; const extractPid = (url) => { if (!url) return ''; const objectIdMatch = url.match(/x_object_id(?:%3A|:)(\d+)/i); if (objectIdMatch) return objectIdMatch[1]; const match = url.match(/(?:\/item\/|\/i\/|productId=)(\d{8,})/i); return match ? match[1] : ''; }; if (msg.type === 'SMART_SEARCH') { const pid = extractPid(sender.tab?.url); handleSearch(msg, pid, sendResponse); return true; } if (msg.type === 'SMART_LOAD_MORE') { const pid = extractPid(sender.tab?.url); handleLoadMore(msg, pid, sendResponse); return true; } return true; }); // 🔥 فتح تبويب مخفي وجلب العنوان الإنجليزي (مثل الإنسان تماماً) function getEnglishTitleInBackground(pid) { return new Promise((resolve) => { if (!pid) return resolve(null); // فتح الصفحة الأمريكية في تبويب غير نشط (مخفي) chrome.tabs.create({ url: `https://www.aliexpress.us/item/${pid}.html`, active: false }, (tab) => { const tabId = tab.id; let timeout = setTimeout(() => { chrome.tabs.onUpdated.removeListener(listener); chrome.tabs.remove(tabId); resolve(null); }, 15000); // 15 ثانية كحد أقصى const listener = (updatedTabId, changeInfo) => { if (updatedTabId === tabId && changeInfo.status === 'complete') { chrome.tabs.onUpdated.removeListener(listener); clearTimeout(timeout); // سحب العنوان من الصفحة المحملة chrome.scripting.executeScript({ target: { tabId: tabId }, func: () => { const meta = document.querySelector('meta[property="og:title"]'); return meta ? meta.content : document.title; } }, (results) => { chrome.tabs.remove(tabId); // إغلاق التبويب if (results && results[0] && results[0].result) { let title = results[0].result; // تنظيف العنوان من اسم المتجر title = title.replace(/ - AliExpress.*$/, '').trim(); resolve(title); } else { resolve(null); } }); } }; chrome.tabs.onUpdated.addListener(listener); }); }); } async function handleSearch(msg, pid, sendResponse) { let searchTitle = msg.title || ''; // 🔥 جلب الكلمات الإنجليزية عبر التبويب المخفي if (pid) { const englishTitle = await getEnglishTitleInBackground(pid); if (englishTitle) { searchTitle = englishTitle; console.log('[Smart Shopper] Hidden Tab Success:', searchTitle); } } const url = `${WORKER_URL}/search-db?q=${encodeURIComponent(searchTitle)}&pid=${pid}&page=1`; try { const res = await fetch(url, { cache: 'no-store' }); if (!res.ok) throw new Error(`HTTP Error: ${res.status}`); const data = await res.json(); sendResponse({ ok: true, items: data.items || [], lastPage: 1, hasMore: data.hasMore || false, totalItems: data.items ? data.items.length : 0 }); } catch (e) { sendResponse({ ok: false, error: String(e) }); } } async function handleLoadMore(msg, pid, sendResponse) { const title = encodeURIComponent(msg.title || ''); const nextPage = Math.max(2, parseInt(msg.startPage) || 2); const existingIds = (msg.existingIds || []).join(','); const url = `${WORKER_URL}/search-db?q=${title}&pid=${pid}&page=${nextPage}&existingIds=${existingIds}`; try { const res = await fetch(url, { cache: 'no-store' }); if (!res.ok) throw new Error(`HTTP Error: ${res.status}`); const data = await res.json(); sendResponse({ ok: true, items: data.items || [], lastPage: nextPage, hasMore: data.hasMore || false }); } catch (e) { sendResponse({ ok: false, error: String(e), hasMore: false }); } }
