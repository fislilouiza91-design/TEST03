// engine.js - Background Script chrome.runtime.onMessage.addListener((request, sender, sendResponse) => { if (request.action === 'FETCH_ALTERNATIVES') { // بما أن تيمو يمنع الاتصال بالـ API، سنعتمد على البيانات المستخرجة من الصفحة // ونقوم بإرسالها مباشرة للواجهة (Panel) const topProducts = request.products || []; // إرسال الدفعة الأولى للواجهة chrome.tabs.sendMessage(sender.tab.id, { action: 'ALTERNATIVE_BATCH', alternatives: topProducts }); // إعلام الواجهة بانتهاء الجلب chrome.tabs.sendMessage(sender.tab.id, { action: 'ALTERNATIVES_FINISHED' }); sendResponse({ success: true }); } return true; });
