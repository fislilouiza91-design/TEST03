/* ============================================ Smart Shopper - Background Service Worker (Direct Product Page Fetch) ============================================ */ const WORKER_URL = 'https://competitors-api.smailsuit11.workers.dev'; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { fetch(`${WORKER_URL}/api/competitors?productId=${msg.productId}`) .then(res => { if(res.ok) return res.json(); throw new Error('Not found'); }) .then(data => sendResponse({ ok: true, data: data.data })) .catch(() => sendResponse({ ok: false })); return true; } if (msg.type === 'SMART_SEARCH') { // نرسل الـ productId والعنوان للخادم ليقوم بسحب صفحة المنتج مباشرة const productId = msg.productId; const originalTitle = encodeURIComponent(msg.title); const originalImage = encodeURIComponent(msg.originalImage || ''); fetch(`${WORKER_URL}/api/fetch-page?productId=${productId}&originalTitle=${originalTitle}&originalImage=${originalImage}&country=${msg.country}`) .then(res => res.json()) .then(data => { if (data.ok) { sendResponse({ ok: true, items: data.items, hasMore: data.hasMore }); if (msg.productId && data.items.length > 0) saveToCloudflare(msg.productId, data.items); } else { sendResponse({ ok: false, error: data.error }); } }).catch(err => sendResponse({ ok: false, error: err.message })); return true; } if (msg.type === 'SMART_LOAD_MORE') { // بما أننا نسحب صفحة منتج واحدة، لا يوجد "تحميل المزيد" من صفحات البحث return sendResponse({ ok: true, items: [], hasMore: false }); } }); function saveToCloudflare(productId, items) { const payload = { base_product_id: productId, all_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })), relevant_items: items.map(c => ({ competitor_id: c.id, title: c.title, price: c.price, currency: 'USD', seller_rating: c.rating, product_url: c.link })) }; fetch(`${WORKER_URL}/api/competitors`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).then(res => res.json()).then(data => console.log('💾 Saved:', data)).catch(console.error); }
