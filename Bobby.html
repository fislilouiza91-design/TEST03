/* ============================================ Smart Shopper - Background Service Worker ============================================ */ // ⚠️ ضع رابط الـ Worker الخاص بك هنا const WORKER_URL = 'https://your-worker.workers.dev'; const MAX_PAGES_TO_FETCH = 3; const STOP_WORDS = new Set([ 'free', 'shipping', 'hot', 'sale', 'new', 'original', 'quality', 'wholesale', 'dropshipping', 'fast', 'delivery', 'cheap', 'best', 'top', 'high', 'good', 'perfect', '2023', '2024', '2025', '2026', 'pcs', 'set', 'pack', 'piece', 'pieces', 'lot', 'items', 'for', 'and', 'with', 'the', 'a', 'of', 'to', 'in', 'on', 'is', 'by', 'at', 'from', 'or', 'an', 'as', 'be', 'this', 'that', 'it', 'not', 'are', 'was', 'if', 'but', 'no', 'yes', 'you', 'your', 'our', 'sans', 'fil', 'sur', 'pour', 'avec', 'les', 'des', 'une', 'dans', 'par', 'et', 'est', 'un', 'le', 'la', 'для', 'и', 'в', 'на', 'с', 'по', 'из' ]); chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const searchQuery = buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } const q = encodeURIComponent(searchQuery); const originalTitle = encodeURIComponent(msg.title); // 🔥 استخدام المسار الجديد واحتساب الدولة fetch(`${WORKER_URL}/api/fetch-page?q=${q}&originalTitle=${originalTitle}&page=${page}&country=US`) .then(res => res.json()) .then(data => { if (data.ok && data.items) { const items = data.items.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.sold, link: item.link, isExact: item.isExact !== undefined ? item.isExact : true, finalScore: item.finalScore || 100 })); let hasMore = data.hasMore; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.error || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } }); // استخراج كلمات مفتاحية دقيقة وقصيرة (للبحث الشامل ثم الفلترة بالذكاء الاصطناعي) function buildSmartSearchQuery(title) { if (!title) return ''; const cleanTitle = title.replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ').replace(/\s+/g, ' ').trim(); const words = cleanTitle.split(' '); const usefulWords = words.filter(w => { const lower = w.toLowerCase(); return w.length > 1 && !STOP_WORDS.has(lower) && !/^\d+$/.test(lower); }); // يأخذ أول 4 كلمات مفيدة كأساس للبحث let result = usefulWords.slice(0, 4); if (result.length === 0) { return words.slice(0, 3).join(' '); } return result.join(' '); }
