// src/apify-fetcher.js import fetch from 'node-fetch'; import crypto from 'crypto'; import { createClient } from '@libsql/client'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const APIFY_TOKEN = process.env.APIFY_TOKEN; const KEYWORDS_PER_RUN = 3; // ===== دوال مساعدة ===== function generateImageHash(url) { if (!url) return null; const m = url.match(/\/([^/]+\.(jpg|jpeg|png|webp))/i); if (m) return crypto.createHash('md5').update(m[1]).digest('hex').substring(0, 16); return null; } function extractTokens(title) { if (!title) return ''; const stop = new Set(['for','the','and','with','from','new','hot','best','free','high','pro','sale','shipping','pcs','piece','set']); return title.toLowerCase() .replace(/[^\w\s-]/g, ' ') .split(/\s+/) .filter(w => w.length > 2 && !stop.has(w) && !/^\d+$/.test(w)) .slice(0, 10) .join(' '); } function detectCategory(title) { if (!title) return 'general'; const t = ' ' + title.toLowerCase().replace(/[^\w\s]/g, ' ') + ' '; const cats = { 'electronics': ['earphone','headphone','speaker','charger','cable','usb','camera','drone','smart watch','phone case','screen protector','phone holder','power bank','earbuds','bluetooth','smartwatch','headset','selfie stick','tripod','sim card'], 'fashion-women': ['women dress','women skirt','women blouse','women heels','women handbag','women bag','women jacket'], 'fashion-men': ['men shirt','men jean','men sneaker','men wallet','men tie','men jacket'], 'home-kitchen': ['kitchen','mug','knife','cookware','storage box','lamp','home decor','pillow','blanket'], 'beauty': ['makeup','skincare','perfume','lipstick','nail polish','hair care','face cream'], 'sports': ['yoga','gym','fitness','running','cycling','sports','dumbbell'], 'toys-kids': ['kids toy',' baby ','lego','puzzle game','action figure','kids dress'], 'automotive': ['car part','motorcycle','auto part','tire','engine','car '], 'pets': ['dog collar','cat toy','pet ','fish tank','bird cage'], 'tools': ['drill','wrench','hammer','screwdriver','tool set'], 'watches': ['watch band','watch case','quartz watch','automatic watch','wristwatch'], 'jewelry': ['necklace','bracelet',' earring','ring ',' jewelry'], 'phones': ['smartphone','phone case','phone screen','phone battery'], 'computer': ['laptop','keyboard','mouse','monitor','pc '], }; for (const [cat, keywords] of Object.entries(cats)) { for (const k of keywords) { const pattern = new RegExp(`\\b${k.trim()}\\b`, 'i'); if (pattern.test(t)) return cat; } } return 'general'; } // ===== استدعاء Apify Actor ===== async function runApifyActor(keyword) { console.log(` 🎭 Apify Actor: lukehunter/aliexpress-scraper`); const actorId = 'lukehunter~aliexpress-scraper'; const runUrl = `https://api.apify.com/v2/acts/${actorId}/run-sync-get-dataset-items?token=${APIFY_TOKEN}`; const inputBody = { searchQueries: [keyword], maxResults: 100, countryCode: 'US', currency: 'USD', }; console.log(` 📤 Input: ${JSON.stringify(inputBody)}`); const response = await fetch(runUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(inputBody), timeout: 300000, }); if (!response.ok) { const err = await response.text(); throw new Error(`Apify ${response.status}: ${err.substring(0, 200)}`); } const items = await response.json(); console.log(` 📥 Apify response: ${Array.isArray(items) ? items.length : 'غير مصفوفة'} عنصر`); // اطبع عينة من أول عنصر لفهم البنية if (Array.isArray(items) && items.length > 0) { console.log(` 🔍 نموذج البيانات:`); console.log(` ${JSON.stringify(items[0]).substring(0, 300)}...`); } return items; } // ===== تحويل بيانات Apify إلى صيغة موحدة ===== function normalizeApifyItem(item) { // Apify Actors المختلفة تُرجع حقولاً بأسماء مختلفة // نحاول استخراج البيانات من أي بنية محتملة const productId = item.productId || item.itemId || item.id || item.product_id || (item.url ? item.url.match(/\/item\/(\d+)/)?.[1] : null); const title = item.title || item.name || item.productTitle || item.product_title || ''; const price = parseFloat(item.price || item.salePrice || item.currentPrice || item.pricing?.sale_price || item.pricing?.current_price || 0); const originalPrice = parseFloat(item.originalPrice || item.listPrice || item.pricing?.original_price || 0); const imageUrl = item.image || item.imageUrl || item.mainImage || item.image_url || (Array.isArray(item.images) ? item.images[0] : '') || ''; const ordersCount = parseInt(item.orders || item.ordersCount || item.sales || item.trade?.orders || 0); const rating = parseFloat(item.rating || item.stars || item.avgRating || item.evaluation?.rating || 0); const sellerName = item.sellerName || item.storeName || item.store?.name || item.seller?.name || ''; const isFreeShipping = item.freeShipping === true || item.shipping?.free === true || item.isFreeShipping === true; return { product_id: productId ? String(productId) : null, title: title, current_price: price, original_price: originalPrice, image_url: imageUrl, orders_count: ordersCount, rating: rating, seller_name: sellerName, is_free_shipping: isFreeShipping, }; } // ===== حفظ في Turso ===== async function saveToTurso(keyword, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { if (!p.product_id || !p.title) continue; if (p.title.length < 15) continue; if (!/^\d+$/.test(p.product_id)) continue; try { const imageHash = generateImageHash(p.image_url); const tokens = extractTokens(p.title); const category = detectCategory(p.title); const trustScore = (p.rating >= 4.8 ? 40 : p.rating >= 4.5 ? 30 : p.rating >= 4.0 ? 20 : 10) + (p.orders_count >= 10000 ? 30 : p.orders_count >= 1000 ? 20 : p.orders_count >= 100 ? 10 : 0) + (p.is_free_shipping ? 10 : 0); const valueScore = (p.rating / 5.0) * 35 + Math.min(p.orders_count / 5000, 1.0) * 30 + Math.max(0, 1 - (p.current_price / 100.0)) * 20; await db.execute({ sql: `INSERT INTO products_current (product_id, keyword, language, title, current_price, original_price, orders_count, rating, image_url, seller_name, image_hash, title_tokens, category, is_free_shipping, trust_score, value_score, total_score, first_seen_at, last_updated_at, update_count) VALUES (?, ?, 'en', ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1) ON CONFLICT (product_id, keyword, language) DO UPDATE SET current_price = excluded.current_price, orders_count = excluded.orders_count, rating = excluded.rating, trust_score = excluded.trust_score, value_score = excluded.value_score, total_score = excluded.total_score, last_updated_at = excluded.last_updated_at, update_count = update_count + 1`, args: [ p.product_id, keyword, p.title, p.current_price, p.original_price || 0, p.orders_count || 0, p.rating || 0, p.image_url || '', p.seller_name || '', imageHash, tokens, category, p.is_free_shipping ? 1 : 0, trustScore, valueScore, Math.round(trustScore * 0.5 + valueScore * 0.5), now, now ], }); saved++; } catch (e) { console.error(` ❌ ${e.message.substring(0, 60)}`); } } return saved; } // ===== اختيار الكلمات التالية ===== async function pickNextKeywords(count) { const res = await db.execute({ sql: `SELECT keyword, category FROM keyword_rotation WHERE enabled = 1 ORDER BY CASE WHEN last_run_at = 0 THEN 0 ELSE 1 END, priority ASC, run_count ASC, last_run_at ASC LIMIT ?`, args: [count], }); return res.rows; } async function markProcessed(keyword, success) { await db.execute({ sql: `UPDATE keyword_rotation SET last_run_at = ?, run_count = run_count + 1, success_count = success_count + ?, fail_count = fail_count + ? WHERE keyword = ?`, args: [Date.now(), success ? 1 : 0, success ? 0 : 1, keyword], }); } // ===== الدالة الرئيسية ===== async function main() { const start = Date.now(); console.log('🎭 بدء الجلب من Apify (lukehunter Actor)...\n'); if (!APIFY_TOKEN) { console.error('❌ APIFY_TOKEN غير موجود!'); process.exit(1); } const keywords = await pickNextKeywords(KEYWORDS_PER_RUN); console.log(`📋 سيتم معالجة ${keywords.length} كلمة\n`); let totalSaved = 0; let successCount = 0; let failCount = 0; for (const row of keywords) { console.log(`\n🔍 "${row.keyword}" [${row.category}]`); try { const items = await runApifyActor(row.keyword); if (!Array.isArray(items) || items.length === 0) { console.log(` ⚠️ لم يرجع Actor أي نتائج`); failCount++; await markProcessed(row.keyword, false); continue; } // تحويل كل عنصر const products = items .map(normalizeApifyItem) .filter(p => p.product_id && p.title && p.title.length > 15); console.log(` 🔧 بعد التطبيع: ${products.length} منتج صالح`); const saved = await saveToTurso(row.keyword, products); console.log(` ✅ ${saved} منتج مخزّن`); totalSaved += saved; if (saved > 0) successCount++; else failCount++; await markProcessed(row.keyword, saved > 0); await new Promise(r => setTimeout(r, 2000)); } catch (error) { console.log(` ❌ فشل: ${error.message}`); failCount++; await markProcessed(row.keyword, false); } } const mins = ((Date.now() - start) / 60000).toFixed(1); console.log(`\n📊 تقرير الدورة (${mins} دقيقة):`); console.log(` ✅ نجح: ${successCount}/${keywords.length}`); console.log(` ❌ فشل: ${failCount}`); console.log(` 📦 مخزّن: ${totalSaved}`); if (failCount > successCount) process.exit(1); } main().catch(err => { console.error('💥 خطأ قاتل:', err); process.exit(1); });
