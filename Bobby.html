import { createClient } from '@libsql/client'; import fetch from 'node-fetch'; const db = createClient({ url: process.env.TURSO_URL, authToken: process.env.TURSO_TOKEN, }); const KEYWORDS = [ 'bluetooth earphones', 'smart watch', 'wireless charger', 'phone case', 'laptop stand', 'usb c cable', 'power bank', 'gaming mouse', 'mechanical keyboard', 'webcam', ]; const API_KEY = process.env.OMKAR_API_KEY; const DELAY_BETWEEN_KEYWORDS = 2000; async function fetchFromOmkar(keyword) { const url = `https://aliexpress-scraper-api.omkar.cloud/aliexpress/v2/search?` + `query=${encodeURIComponent(keyword)}&` + `sort_by=most_orders&` + `country_code=US&` + `page=1`; const response = await fetch(url, { method: 'GET', headers: { 'API-Key': API_KEY, 'Accept': 'application/json', }, }); if (!response.ok) { const errText = await response.text(); throw new Error(`Omkar API error ${response.status}: ${errText}`); } const data = await response.json(); return data.results || []; } async function saveToTurso(keyword, products) { if (products.length === 0) return 0; const now = Date.now(); let saved = 0; for (const p of products) { try { const productId = p.id || p.product_id; const title = p.title || ''; const price = parseFloat(p.pricing?.sale_price || p.sale_price || 0); const originalPrice = parseFloat(p.pricing?.original_price || p.original_price || 0); const orders = parseInt(p.orders_count || p.orders || 0); const rating = parseFloat(p.rating || 0); const image = p.image || ''; const sellerName = p.store?.name || p.seller?.name || ''; const sellerRating = parseFloat(p.store?.rating || p.seller?.rating || 0); const isFreeShipping = (p.shipping?.free === true || p.shipping?.free_shipping === true) ? 1 : 0; await db.execute({ sql: `INSERT OR REPLACE INTO products_current (product_id, keyword, title, current_price, original_price, orders_count, rating, image_url, seller_name, seller_rating, is_free_shipping, first_seen_at, last_updated_at, update_count) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1)`, args: [productId, keyword, title, price, originalPrice, orders, rating, image, sellerName, sellerRating, isFreeShipping, now, now] }); await db.execute({ sql: `INSERT INTO price_history (product_id, keyword, price, orders_count, recorded_at, source) VALUES (?, ?, ?, ?, ?, ?)`, args: [productId, keyword, price, orders, now, 'omkar'] }); saved++; } catch (error) { console.error(` ❌ فشل حفظ ${p.id}:`, error.message); } } return saved; } async function main() { console.log('🚀 بدء الجلب من Omkar Cloud...'); console.log(`📋 عدد الكلمات المفتاحية: ${KEYWORDS.length}`); let totalSaved = 0; let successCount = 0; let failCount = 0; for (const keyword of KEYWORDS) { try { console.log(`\n🔍 جلب: ${keyword}`); const products = await fetchFromOmkar(keyword); console.log(` 📦 استُلم ${products.length} منتج`); const saved = await saveToTurso(keyword, products); console.log(` ✅ خُزّن ${saved} منتج`); totalSaved += saved; successCount++; await new Promise(r => setTimeout(r, DELAY_BETWEEN_KEYWORDS)); } catch (error) { console.error(` ❌ فشل "${keyword}":`, error.message); failCount++; } } console.log(`\n📊 ملخص:`); console.log(` ✅ كلمات ناجحة: ${successCount}/${KEYWORDS.length}`); console.log(` ❌ كلمات فاشلة: ${failCount}`); console.log(` 📦 إجمالي المنتجات المخزنة: ${totalSaved}`); if (failCount > successCount) { process.exit(1); } } main().catch(error => { console.error('💥 خطأ قاتل:', error); process.exit(1); });
