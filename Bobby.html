/* ============================================ Smart Shopper - Background Service Worker Search is done on Cloudflare Worker (with AI) ============================================ */ const WORKER_URL = 'https://extension.smailsuit11.workers.dev/'; const MAX_PAGES_TO_FETCH = 3; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } // 🔥 إرسال العنوان الأصلي كاملاً (بدون أي تصفية) // الـ Cloudflare Worker هو من يقرر ماذا يفعل const originalTitle = encodeURIComponent(msg.title || ''); console.log(`🎯 [${msg.type}] Title: "${msg.title}" | Page: ${page}`); fetch(`${WORKER_URL}?originalTitle=${originalTitle}&page=${page}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: 100 })); let hasMore = items.length >= 15; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; console.log(`✅ Got ${items.length} items (query: "${data.aiQuery || '?'}")`); sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } if (msg.type === 'SORT_ITEMS') { fetch(`${WORKER_URL}sort`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ originalTitle: msg.title, items: msg.items }) }) .then(res => res.json()) .then(data => sendResponse({ ok: data.success, items: data.data })) .catch(err => sendResponse({ ok: false, error: err.message })); return true; } });
