/* ============================================ Smart Shopper - Background Service Worker ============================================ */ const WORKER_URL = 'https://extension.smailsuit11.workers.dev/'; const MAX_PAGES_TO_FETCH = 3; chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (!msg) return; if (msg.type === 'CHECK_CACHE') { sendResponse({ ok: false }); return true; } if (msg.type === 'SMART_SEARCH' || msg.type === 'SMART_LOAD_MORE') { // 🔥 استخدام customQuery إذا وُجد (من content.js) const searchQuery = msg.customQuery || buildSmartSearchQuery(msg.title); const page = msg.type === 'SMART_SEARCH' ? 1 : (parseInt(msg.startPage) || 1); if (page > MAX_PAGES_TO_FETCH) { return sendResponse({ ok: true, items: [], hasMore: false }); } console.log(`🎯 [${msg.type}] Query: [${searchQuery}] | Page: ${page}`); const q = encodeURIComponent(searchQuery); fetch(`${WORKER_URL}?q=${q}&page=${page}`) .then(res => res.json()) .then(data => { if (data.success && data.data) { const items = data.data.map(item => ({ id: item.id, title: item.title, img: item.img, price: item.price, rating: item.rating, sold: item.soldStr, link: item.url, isExact: true, finalScore: 100 })); let hasMore = items.length >= 15; if (page >= MAX_PAGES_TO_FETCH) hasMore = false; sendResponse({ ok: true, items: items, hasMore: hasMore }); } else { sendResponse({ ok: false, error: data.message || 'No results', hasMore: false }); } }).catch(err => { sendResponse({ ok: false, error: err.message, hasMore: false }); }); return true; } if (msg.type === 'SORT_ITEMS') { fetch(`${WORKER_URL}sort`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ originalTitle: msg.title, items: msg.items }) }) .then(res => res.json()) .then(data => sendResponse({ ok: data.success, items: data.data })) .catch(err => sendResponse({ ok: false, error: err.message })); return true; } }); function buildSmartSearchQuery(title) { if (!title) return ''; const cleanTitle = title.replace(/[^\w\s\-\u0600-\u06FF\u0400-\u04FF\u00C0-\u017F]/g, ' ').replace(/\s+/g, ' ').trim(); const rawWords = cleanTitle.split(' '); const stopWords = new Set([ 'the', 'and', 'for', 'with', 'new', 'hot', 'sale', 'free', 'shipping', 'original', 'high', 'quality', 'best', 'pcs', 'set', 'pack', 'piece', 'pieces', 'items', 'item', 'wholesale', 'dropshipping', 'compatible', 'fits', 'fit', 'suitable', 'accessories', 'accessory', 'universal', 'dual', 'purpose', 'large', 'screen', 'plug', 'creative', 'dedicated', 'student', 'portable', 'multifunctional', 'design', 'style', 'luxury', 'fast', 'quick', 'smart', 'waterproof', 'shockproof', 'gift', 'fashion', 'good', 'perfect', '2023', '2024', '2025' ]); const usefulWords = rawWords.filter(w => { const lower = w.toLowerCase(); const isNumberOrSize = /^\d+$/.test(lower) || /^\d+(mm|cm|inch|in|v|w|mah)$/i.test(lower); return w.length > 1 && !stopWords.has(lower) && !isNumberOrSize; }); const finalQuery = usefulWords.slice(0, 4).join(' '); if (!finalQuery) { return rawWords.slice(0, 3).join(' '); } return finalQuery; }
