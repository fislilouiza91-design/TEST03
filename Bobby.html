async function fetchSimilarProducts({ page = 1, streamId = null }) { const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true }); if (!tab?.url?.match(/\/item\/(\d+)\.htm/)) return { error: 'NOT_PRODUCT_PAGE' }; // 1. حقن اعتراض الـ API داخل الصفحة await chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => { if (window.__NEZZER_INTERCEPTED__) return; window.__NEZZER_INTERCEPTED__ = true; window.__NEZZER_API_DATA__ = null; const originalFetch = window.fetch; window.fetch = async function(...args) { const response = await originalFetch.apply(this, args); try { const url = typeof args[0] === 'string' ? args[0] : (args[0] && args[0].url); if (url && (url.includes('recommend') || url.includes('similar')) && response.ok) { const clone = response.clone(); clone.json().then(data => { // التحقق من وجود بيانات المنتجات في الاستجابة if (data && (data.data || data.result || data.mods)) { window.__NEZZER_API_DATA__ = data; } }).catch(e => {}); } } catch (e) {} return response; }; } }); // 2. تحريك الصفحة للأسفل لإجبار الموقع على جلب منتجات جديدة await chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => { window.scrollTo(0, document.body.scrollHeight * 0.8); } }); // 3. الانتظار 3 ثوانٍ حتى يقوم الموقع بطلب البيانات await new Promise(r => setTimeout(r, 3000)); // 4. استرجاع البيانات التي تم اعتراضها const [result] = await chrome.scripting.executeScript({ target: { tabId: tab.id }, world: 'MAIN', func: () => window.__NEZZER_API_DATA__ }); const apiData = result?.result; if (!apiData) { return { error: 'API_DATA_NOT_FOUND: يرجى التمرير لأسفل الصفحة قليلاً ثم إعادة المحاولة.' }; } // 5. استخراج المنتجات من الاستجابة (دعم الهياكل المختلفة لموقع .ru) let rawProducts = apiData?.data?.mods?.itemList?.content || apiData?.result?.mods?.itemList?.content || apiData?.data?.itemList || apiData?.mods?.itemList?.content || []; // إذا لم نجد المنتجات، نبحث في المصفوفات الأخرى if (rawProducts.length === 0) { const findArray = (obj) => { for (let key in obj) { if (Array.isArray(obj[key]) && obj[key].length > 0 && obj[key][0].productId) return obj[key]; if (typeof obj[key] === 'object' && obj[key] !== null) { const found = findArray(obj[key]); if (found) return found; } } return null; }; rawProducts = findArray(apiData) || []; } // 6. تحويل البيانات لتتوافق مع واجهة الإضافة const products = rawProducts.map(p => { const priceNum = p?.prices?.salePrice?.minPrice || p?.prices?.salePrice?.formattedPrice || p?.price?.formattedPrice || p?.price?.minPrice || '0'; return { productId: p.productId || p.id || p.itemId, title: { displayTitle: p.title?.displayTitle || p.title || 'Product' }, image: { imgUrl: p.image?.imgUrl || p.imageUrl || p.imgUrl || '' }, prices: { salePrice: { formattedPrice: typeof priceNum === 'string' && priceNum.includes('₽') ? priceNum : `${priceNum} ₽`, minPrice: p?.prices?.salePrice?.minPrice || p?.price?.minPrice || 0 } }, trade: { tradeCount: p.trade?.tradeCount || p.tradeCount || 0 }, evaluation: { starRating: p.evaluation?.starRating || p.starRating || 0 } }; }); if (products.length === 0) { return { error: 'NO_PRODUCTS_EXTRACTED' }; } return { products: products, streamId: null, ret: 'SUCCESS', host: location.host }; } chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => { if (msg.action === 'FETCH_SIMILAR') { fetchSimilarProducts({ page: msg.page || 1, streamId: msg.streamId || null }) .then(sendResponse) .catch(err => sendResponse({ error: err.message })); return true; } });
