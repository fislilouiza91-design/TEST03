name: Unzip and Extract on: workflow_dispatch: jobs: unzip: runs-on: ubuntu-latest permissions: contents: write steps: - name: Checkout uses: actions/checkout@v4 - name: Find and extract ZIP run: | echo "🔍 البحث عن ملف ZIP..." # ابحث عن أي ملف zip ZIP_FILE=$(find . -maxdepth 1 -name "*.zip" -type f | head -1) if [ -z "$ZIP_FILE" ]; then echo "❌ لم أجد أي ملف ZIP" exit 1 fi echo "📦 وجدت: $ZIP_FILE" # فك الضغط unzip -o "$ZIP_FILE" -d ./extracted echo "✅ تم فك الضغط" # اعرض المحتوى ls -la ./extracted - name: Move files to root run: | # إذا كان هناك مجلد داخل extracted، انقل محتواه EXTRACTED_DIR="./extracted" # إذا كان هناك مجلد واحد فقط داخل extracted SUBDIRS=$(find "$EXTRACTED_DIR" -maxdepth 1 -type d | tail -n +2 | wc -l) if [ "$SUBDIRS" -eq 1 ]; then # هناك مجلد واحد — انقل محتواه SUBDIR=$(find "$EXTRACTED_DIR" -maxdepth 1 -type d | tail -n +2 | head -1) echo "📁 نقل محتوى: $SUBDIR" # انقل كل شيء mv "$SUBDIR"/* . 2>/dev/null || true mv "$SUBDIR"/.[!.]* . 2>/dev/null || true else # عدة عناصر — انقلها مباشرة mv "$EXTRACTED_DIR"/* . 2>/dev/null || true mv "$EXTRACTED_DIR"/.[!.]* . 2>/dev/null || true fi # احذف المجلدات المؤقتة rm -rf ./extracted # احذف ملف ZIP الأصلي rm -f *.zip echo "✅ تم التنظيف" echo "📁 المحتوى النهائي:" ls -la - name: Commit changes run: | git config user.name "github-actions[bot]" git config user.email "github-actions[bot]@users.noreply.github.com" git add -A if git diff --staged --quiet; then echo "لا يوجد تغييرات" else git commit -m "Extract ZIP contents" git push fi
